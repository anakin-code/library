
int maxMonths = totalMonthsMap.values()
        .stream()
        .max(Integer::compareTo)
        .orElse(1);

// ここに追加：経験月数が多い上位5スキルだけ残す
Set<Long> topSkillIds =
        totalMonthsMap.entrySet().stream()
                .sorted(Map.Entry.<Long, Integer>comparingByValue().reversed())
                .limit(5)
                .map(Map.Entry::getKey)
                .collect(Collectors.toSet());

List<PersonalSkillNodeResponse> nodes =
        durationMap.entrySet()
                .stream()
                .filter(entry -> topSkillIds.contains(entry.getKey()))
                .map(entry -> {
                    Long skillId = entry.getKey();
                    CareerDuration duration = entry.getValue();
                    int totalMonths = toTotalMonths(duration);

                    return new PersonalSkillNodeResponse(
                            skillId,
                            skillNameMap.get(skillId),
                            duration.years(),
                            duration.months(),
                            maxMonths == 0
                                    ? 0.0
                                    : (double) totalMonths / maxMonths
                    );
                })
                .toList();

List<PersonalSkillEdgeResponse> edges =
        edgePeriodMap.entrySet()
                .stream()
                .filter(entry -> {
                    String[] ids = entry.getKey().split("-");

                    Long source = Long.valueOf(ids[0]);
                    Long target = Long.valueOf(ids[1]);

                    return topSkillIds.contains(source)
                            && topSkillIds.contains(target);
                })
                .map(entry -> {
                    String[] ids = entry.getKey().split("-");

                    Long source = Long.valueOf(ids[0]);
                    Long target = Long.valueOf(ids[1]);

                    CareerDuration duration =
                            calculateDurationFromPeriods(entry.getValue());

                    int edgeTotalMonths = toTotalMonths(duration);

                    int denominator = Math.max(
                            totalMonthsMap.getOrDefault(source, 0),
                            totalMonthsMap.getOrDefault(target, 0)
                    );

                    double normalizedWeight =
                            denominator == 0
                                    ? 0.0
                                    : (double) edgeTotalMonths / denominator;

                    return new PersonalSkillEdgeResponse(
                            source,
                            target,
                            duration.years(),
                            duration.months(),
                            normalizedWeight
                    );
                })
                .toList();

return new PersonalSkillGraphResponse(nodes, edges);
