public interface NeedRepository extends JpaRepository<Need, Long> {

    @Query(
            value = """
                    SELECT n
                    FROM Need n
                    LEFT JOIN NeedEmpathy ne ON ne.need = n
                    LEFT JOIN Tip t ON t.need = n AND t.deleteFlg = false
                    WHERE n.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(n.needTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(n.needDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(n.user.userName) LIKE LOWER(CONCAT('%', :keyword, '%'))
                          )
                      AND (:affiliatedId IS NULL OR n.user.affiliated.affiliatedId = :affiliatedId)
                      AND (:rankId IS NULL OR n.user.rank.rankId = :rankId)
                      AND (:categoryId IS NULL OR n.category.categoryId = :categoryId)
                    GROUP BY n
                    ORDER BY
                      CASE WHEN :orderBy = 'empathy' THEN COUNT(DISTINCT ne) END DESC,
                      n.createdTime DESC
                    """,
            countQuery = """
                    SELECT COUNT(n)
                    FROM Need n
                    WHERE n.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(n.needTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(n.needDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import simplex.pj148.skippy.server.controller.dto.response.NeedListResponseDto;
import simplex.pj148.skippy.server.controller.dto.search.NeedSearchConditionDto;
import simplex.pj148.skippy.server.entity.Need;
import simplex.pj148.skippy.server.repository.NeedEmpathyRepository;
import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.TipRepository;

@Service
@RequiredArgsConstructor
public class NeedService {

    private final NeedRepository needRepository;
    private final NeedEmpathyRepository needEmpathyRepository;
    private final TipRepository tipRepository;

    @Transactional(readOnly = true)
    public Page<NeedListResponseDto> searchNeeds(
            Long loginUserId,
            NeedSearchConditionDto condition,
            Pageable pageable
    ) {
        String keyword = normalizeKeyword(condition.getKeyword());
        Long affiliatedId = normalizeId(condition.getAffiliatedId());
        Long rankId = normalizeId(condition.getRankId());
        Long categoryId = normalizeId(condition.getCategoryId());
        String orderBy = normalizeOrderBy(condition.getOrderBy());

        Page<Need> needs = needRepository.searchNeeds(
                keyword,
                affiliatedId,
                rankId,
                categoryId,
                orderBy,
                pageable
        );

        return needs.map(need -> toNeedListResponseDto(need, loginUserId));
    }

    private NeedListResponseDto toNeedListResponseDto(Need need, Long loginUserId) {
        Long needId = need.getNeedId();

        Long empathyCount = needEmpathyRepository.countByNeedNeedId(needId);

        Long answerCount = tipRepository.countByNeedNeedIdAndDeleteFlgFalse(needId);

        Boolean empathyByCurrentUser =
                needEmpathyRepository.existsByNeedNeedIdAndUserUserId(
                        needId,
                        loginUserId
                );

        Boolean editFlg = need.getUser().getUserId().equals(loginUserId);

        return new NeedListResponseDto(
                need.getNeedId(),
                need.getNeedTitle(),
                need.getNeedDetail(),
                need.getCategory().getCategoryId(),
                need.getCategory().getCategoryName(),
                need.getUser().getUserId(),
                need.getUser().getUserName(),
                need.getUser().getAffiliated().getAffiliatedName(),
                need.getUser().getRank().getRankName(),
                empathyCount,
                answerCount,
                empathyByCurrentUser,
                editFlg,
                need.getCreatedTime()
        );
    }

    private String normalizeKeyword(String keyword) {
        if (keyword == null || keyword.isBlank()) {
            return null;
        }

        return keyword.trim();
    }

    private Long normalizeId(Long id) {
        if (id == null || id == 0) {
            return null;
        }

        return id;
    }

    private String normalizeOrderBy(String orderBy) {
        if ("empathy".equals(orderBy)) {
            return "empathy";
        }

        return "new";
    }
}

package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import simplex.pj148.skippy.server.controller.dto.response.NeedListResponseDto;
import simplex.pj148.skippy.server.controller.dto.search.NeedSearchConditionDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.NeedService;
import simplex.pj148.skippy.server.service.UserInfoService;

@RequestMapping("/api/needs")
@RequiredArgsConstructor
@RestController
public class NeedsController {

    private final NeedService needService;
    private final UserInfoService userInfoService;

    @GetMapping("/search")
    public Page<NeedListResponseDto> searchNeeds(
            @AuthenticationPrincipal Jwt jwt,
            @RequestParam(required = false) String keyword,
            @RequestParam(required = false) Long affiliatedId,
            @RequestParam(required = false) Long positionId,
            @RequestParam(required = false) Long categoryId,
            @RequestParam(required = false) String orderBy,
            Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        NeedSearchConditionDto condition = new NeedSearchConditionDto(
                keyword,
                affiliatedId,
                positionId,
                categoryId,
                orderBy
        );

        return needService.searchNeeds(
                loginUser.getUserId(),
                condition,
                pageable
        );
    }
}
