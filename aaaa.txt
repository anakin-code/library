package simplex.pj148.skippy.server.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import simplex.pj148.skippy.server.entity.Tip;

import java.util.List;

public interface TipRepository extends JpaRepository<Tip, Long> {

    @Query(
            value = """
                    SELECT t
                    FROM Tip t
                    LEFT JOIN Tipinterest ti ON ti.tip = t
                    LEFT JOIN Comment c ON c.tip = t AND c.deleteFlg = false
                    WHERE t.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(t.tipTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.tipDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.user.userName) LIKE LOWER(CONCAT('%', :keyword, '%'))
                          )
                      AND (:affiliatedId IS NULL OR t.user.affiliated.affiliatedId = :affiliatedId)
                      AND (:rankId IS NULL OR t.user.rank.rankId = :rankId)
                      AND (:categoryId IS NULL OR t.category.categoryId = :categoryId)
                    GROUP BY t
                    ORDER BY
                      CASE WHEN :orderBy = 'interestCount' THEN COUNT(DISTINCT ti) END DESC,
                      CASE WHEN :orderBy = 'commentCount' THEN COUNT(DISTINCT c) END DESC,
                      t.createdTime DESC
                    """,
            countQuery = """
                    SELECT COUNT(t)
                    FROM Tip t
                    WHERE t.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(t.tipTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.tipDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.user.userName) LIKE LOWER(CONCAT('%', :keyword, '%'))
                          )
                      AND (:affiliatedId IS NULL OR t.user.affiliated.affiliatedId = :affiliatedId)
                      AND (:rankId IS NULL OR t.user.rank.rankId = :rankId)
                      AND (:categoryId IS NULL OR t.category.categoryId = :categoryId)
                    """
    )
    Page<Tip> searchTips(
            @Param("keyword") String keyword,
            @Param("affiliatedId") Long affiliatedId,
            @Param("rankId") Long rankId,
            @Param("categoryId") Long categoryId,
            @Param("orderBy") String orderBy,
            Pageable pageable
    );

    List<Tip> findByNeedNeedIdAndDeleteFlgFalseOrderByCreatedTimeDesc(Long needId);

    Long countByNeedNeedIdAndDeleteFlgFalse(Long needId);
}

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.TipCommentCreateRequestDto;
import simplex.pj148.skippy.server.controller.dto.request.TipCreateRequestDto;
import simplex.pj148.skippy.server.controller.dto.request.TipUpdateRequestDto;
import simplex.pj148.skippy.server.controller.dto.response.TipCommentResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.TipDetailResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.TipListResponseDto;
import simplex.pj148.skippy.server.controller.dto.search.TipSearchConditionDto;
import simplex.pj148.skippy.server.entity.Category;
import simplex.pj148.skippy.server.entity.Comment;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.Tipinterest;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CategoryRepository;
import simplex.pj148.skippy.server.repository.CommentRepository;
import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.TipinterestRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional
public class TipService {

    private final TipRepository tipRepository;

    private final TipinterestRepository tipinterestRepository;

    private final CommentRepository commentRepository;

    private final UserRepository userRepository;

    private final CategoryRepository categoryRepository;

    @Transactional(readOnly = true)
    public Page<TipListResponseDto> searchTips(
            Long loginUserId,
            TipSearchConditionDto condition,
            Pageable pageable
    ) {
        Page<Tip> tips = tipRepository.searchTips(
                condition.getKeyword(),
                condition.getAffiliatedId(),
                condition.getRankId(),
                condition.getCategoryId(),
                condition.getOrderBy(),
                pageable
        );

        return tips.map(tip -> toTipListResponseDto(tip, loginUserId));
    }

    @Transactional(readOnly = true)
    public TipDetailResponseDto findTipDetail(Long tipId, Long loginUserId) {
        Tip tip = findActiveTipById(tipId);

        List<TipCommentResponseDto> comments = commentRepository
                .findByTipTipIdAndDeleteFlgFalseOrderByCreatedTimeAsc(tipId)
                .stream()
                .map(comment -> toTipCommentResponseDto(comment, loginUserId))
                .toList();

        Long interestCount = tipinterestRepository.countByTipTipId(tipId);
        Boolean interestByCurrentUser =
                tipinterestRepository.existsByTipTipIdAndUserUserId(tipId, loginUserId);
        Boolean editFlg = tip.getUser().getUserId().equals(loginUserId);

        return new TipDetailResponseDto(
                tip.getTipId(),
                tip.getTipTitle(),
                tip.getTipDetail(),
                tip.getTipUrl(),
                tip.getCategory().getCategoryId(),
                tip.getCategory().getCategoryName(),
                tip.getUser().getUserId(),
                tip.getUser().getUserName(),
                tip.getUser().getAffiliated().getAffiliatedName(),
                tip.getUser().getRank().getRankName(),
                interestCount,
                interestByCurrentUser,
                editFlg,
                tip.getCreatedTime(),
                comments
        );
    }

    public void createTip(TipCreateRequestDto input, Long loginUserId) {
        User user = findUserById(loginUserId);
        Category category = findCategoryById(input.getCategoryId());

        Tip tip = new Tip();
        tip.setUser(user);
        tip.setCategory(category);
        tip.setNeed(null);
        tip.setTipTitle(input.getTipTitle());
        tip.setTipDetail(input.getTipDetail());
        tip.setTipUrl(input.getTipUrl());
        tip.setDeleteFlg(false);
        tip.setCreatedTime(LocalDateTime.now());
        tip.setUpdatedTime(LocalDateTime.now());

        tipRepository.save(tip);
    }

    public void updateTip(Long tipId, TipUpdateRequestDto input, Long loginUserId) {
        Tip tip = findActiveTipById(tipId);

        checkTipOwner(tip, loginUserId);

        tip.setTipDetail(input.getTipDetail());
        tip.setUpdatedTime(LocalDateTime.now());

        tipRepository.save(tip);
    }

    public void deleteTip(Long tipId, Long loginUserId) {
        Tip tip = findActiveTipById(tipId);

        checkTipOwner(tip, loginUserId);

        tip.setDeleteFlg(true);
        tip.setUpdatedTime(LocalDateTime.now());

        tipRepository.save(tip);
    }

    public void createTipInterest(Long tipId, Long loginUserId) {
        Tip tip = findActiveTipById(tipId);
        User user = findUserById(loginUserId);

        boolean alreadyExists =
                tipinterestRepository.existsByTipTipIdAndUserUserId(tipId, loginUserId);

        if (alreadyExists) {
            return;
        }

        Tipinterest tipinterest = new Tipinterest();
        tipinterest.setTip(tip);
        tipinterest.setUser(user);
        tipinterest.setIsLooked(false);
        tipinterest.setIsMyBest(false);
        tipinterest.setCreatedTime(LocalDateTime.now());

        tipinterestRepository.save(tipinterest);
    }

    public void deleteTipInterest(Long tipId, Long loginUserId) {
        Tipinterest tipinterest = tipinterestRepository
                .findByTipTipIdAndUserUserId(tipId, loginUserId)
                .orElseThrow(() -> new RuntimeException("気になるが見つかりません"));

        if (Boolean.TRUE.equals(tipinterest.getIsMyBest())) {
            tipinterest.setIsMyBest(false);
            tipinterestRepository.save(tipinterest);
            return;
        }

        tipinterestRepository.delete(tipinterest);
    }

    public void createComment(
            Long tipId,
            TipCommentCreateRequestDto input,
            Long loginUserId
    ) {
        Tip tip = findActiveTipById(tipId);
        User user = findUserById(loginUserId);

        Comment comment = new Comment();
        comment.setTip(tip);
        comment.setUser(user);
        comment.setCommentBody(input.getCommentBody());
        comment.setIsLooked(false);
        comment.setDeleteFlg(false);
        comment.setCreatedTime(LocalDateTime.now());

        commentRepository.save(comment);
    }

    public void deleteComment(Long tipId, Long commentId, Long loginUserId) {
        Comment comment = commentRepository.findById(commentId)
                .orElseThrow(() -> new RuntimeException("コメントが見つかりません"));

        if (Boolean.TRUE.equals(comment.getDeleteFlg())) {
            throw new RuntimeException("削除済みのコメントです");
        }

        if (!comment.getTip().getTipId().equals(tipId)) {
            throw new RuntimeException("ノウハウとコメントの組み合わせが不正です");
        }

        if (!comment.getUser().getUserId().equals(loginUserId)) {
            throw new RuntimeException("このコメントを削除する権限がありません");
        }

        comment.setDeleteFlg(true);

        commentRepository.save(comment);
    }

    private TipListResponseDto toTipListResponseDto(Tip tip, Long loginUserId) {
        Long interestCount = tipinterestRepository.countByTipTipId(tip.getTipId());
        Long commentCount = commentRepository.countByTipTipIdAndDeleteFlgFalse(tip.getTipId());
        Boolean interestByCurrentUser =
                tipinterestRepository.existsByTipTipIdAndUserUserId(tip.getTipId(), loginUserId);
        Boolean editFlg = tip.getUser().getUserId().equals(loginUserId);

        return new TipListResponseDto(
                tip.getTipId(),
                tip.getTipTitle(),
                tip.getTipDetail(),
                tip.getTipUrl(),
                tip.getCategory().getCategoryId(),
                tip.getCategory().getCategoryName(),
                tip.getUser().getUserId(),
                tip.getUser().getUserName(),
                tip.getUser().getAffiliated().getAffiliatedName(),
                tip.getUser().getRank().getRankName(),
                interestCount,
                commentCount,
                interestByCurrentUser,
                editFlg,
                tip.getCreatedTime()
        );
    }

    private TipCommentResponseDto toTipCommentResponseDto(Comment comment, Long loginUserId) {
        Boolean editFlg = comment.getUser().getUserId().equals(loginUserId);

        return new TipCommentResponseDto(
                comment.getCommentId(),
                comment.getCommentBody(),
                comment.getUser().getUserId(),
                comment.getUser().getUserName(),
                editFlg,
                comment.getCreatedTime()
        );
    }

    private Tip findActiveTipById(Long tipId) {
        Tip tip = tipRepository.findById(tipId)
                .orElseThrow(() -> new RuntimeException("ノウハウが見つかりません"));

        if (Boolean.TRUE.equals(tip.getDeleteFlg())) {
            throw new RuntimeException("削除済みのノウハウです");
        }

        return tip;
    }

    private User findUserById(Long userId) {
        return userRepository.findById(userId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));
    }

    private Category findCategoryById(Long categoryId) {
        return categoryRepository.findById(categoryId)
                .orElseThrow(() -> new RuntimeException("カテゴリが見つかりません"));
    }

    private void checkTipOwner(Tip tip, Long loginUserId) {
        if (!tip.getUser().getUserId().equals(loginUserId)) {
            throw new RuntimeException("このノウハウを操作する権限がありません");
        }
    }
}

import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.TipinterestRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

import java.time.LocalDateTime;
import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional
public class MyBestService {

    private final TipinterestRepository tipinterestRepository;

    private final TipRepository tipRepository;

    private final UserRepository userRepository;

    private final CategoryRepository categoryRepository;

    @Transactional(readOnly = true)
    public List<MyBestResponseDto> findMyBestList(Long loginUserId) {
        List<Category> categories = categoryRepository.findAll();
        List<Tipinterest> myBests =
                tipinterestRepository.findByUserUserIdAndIsMyBestTrue(loginUserId);

        return categories.stream()
                .map(category -> {
                    Tipinterest myBest = myBests.stream()
                            .filter(tipinterest -> tipinterest.getTip().getCategory().getCategoryId()
                                    .equals(category.getCategoryId()))
                            .findFirst()
                            .orElse(null);

                    if (myBest == null) {
                        return new MyBestResponseDto(
                                category.getCategoryId(),
                                category.getCategoryName(),
                                null,
                                "未選択",
                                null,
                                false
                        );
                    }

                    Tip tip = myBest.getTip();

                    return new MyBestResponseDto(
                            category.getCategoryId(),
                            category.getCategoryName(),
                            tip.getTipId(),
                            tip.getTipTitle(),
                            tip.getTipDetail(),
                            true
                    );
                })
                .toList();
    }

    public void upsertMyBest(MyBestUpsertRequestDto input, Long loginUserId) {
        User user = userRepository.findById(loginUserId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Tip tip = tipRepository.findById(input.getTipId())
                .orElseThrow(() -> new RuntimeException("ノウハウが見つかりません"));

        if (Boolean.TRUE.equals(tip.getDeleteFlg())) {
            throw new RuntimeException("削除済みのノウハウです");
        }

        if (!tip.getCategory().getCategoryId().equals(input.getCategoryId())) {
            throw new RuntimeException("カテゴリとノウハウが一致していません");
        }

        tipinterestRepository
                .findByUserUserIdAndTipCategoryCategoryIdAndIsMyBestTrue(
                        loginUserId,
                        input.getCategoryId()
                )
                .ifPresent(oldMyBest -> {
                    oldMyBest.setIsMyBest(false);
                    tipinterestRepository.save(oldMyBest);
                });

        Tipinterest target = tipinterestRepository
                .findByTipTipIdAndUserUserId(input.getTipId(), loginUserId)
                .orElseGet(() -> {
                    Tipinterest newTipinterest = new Tipinterest();
                    newTipinterest.setTip(tip);
                    newTipinterest.setUser(user);
                    newTipinterest.setIsLooked(false);
                    newTipinterest.setCreatedTime(LocalDateTime.now());
                    return newTipinterest;
                });

        target.setIsMyBest(true);

        tipinterestRepository.save(target);
    }
}

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.ProfileUpdateRequestDto;
import simplex.pj148.skippy.server.controller.dto.response.MyBestResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.NeedListResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.ProfileResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.TipListResponseDto;
import simplex.pj148.skippy.server.entity.Affiliated;
import simplex.pj148.skippy.server.entity.Rank;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.AffiliatedRepository;
import simplex.pj148.skippy.server.repository.RankRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
@Transactional
public class ProfileService {

    private final UserRepository userRepository;

    private final AffiliatedRepository affiliatedRepository;

    private final RankRepository rankRepository;

    private final MyBestService myBestService;

    @Transactional(readOnly = true)
    public ProfileResponseDto findProfile(Long targetUserId, Long loginUserId) {
        User targetUser = userRepository.findById(targetUserId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Boolean editFlg = targetUser.getUserId().equals(loginUserId);

        List<MyBestResponseDto> myBestList = myBestService.findMyBestList(targetUserId);

        return new ProfileResponseDto(
                targetUser.getUserId(),
                targetUser.getUserName(),
                targetUser.getAffiliated().getAffiliatedId(),
                targetUser.getAffiliated().getAffiliatedName(),
                targetUser.getRank().getRankId(),
                targetUser.getRank().getRankName(),
                targetUser.getSelfIntroduction(),
                editFlg,
                myBestList,
                List.of(),
                List.of(),
                List.of(),
                List.of()
        );
    }

    public void updateProfile(Long loginUserId, ProfileUpdateRequestDto input) {
        User user = userRepository.findById(loginUserId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Affiliated affiliated = affiliatedRepository.findById(input.getAffiliatedId())
                .orElseThrow(() -> new RuntimeException("所属が見つかりません"));

        Rank rank = rankRepository.findById(input.getRankId())
                .orElseThrow(() -> new RuntimeException("職位が見つかりません"));

        user.setAffiliated(affiliated);
        user.setRank(rank);
        user.setSelfIntroduction(input.getSelfIntroduction());

        userRepository.save(user);
    }
}
