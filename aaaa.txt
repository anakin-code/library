public record QuestionDetailResponse(
    Long questionId,
    String questionTitle,
    String questionContent,

    String targetHrid,
    String targetName,

    String createdHrid,
    String createdName
) {}
@Query("""
SELECT new simplex.bn26.master.server.response.QuestionDetailResponse(
    q.questionId,
    q.questionTitle,
    q.questionContent,

    tp.hrid,
    tp.name,

    cp.hrid,
    cp.name
)
FROM Question q
JOIN User tu
    ON tu.userId = q.targetUserId
JOIN Profile tp
    ON tp.user = tu

JOIN User cu
    ON cu.userId = q.createdByUserId
JOIN Profile cp
    ON cp.user = cu
""")
List<QuestionDetailResponse> findQuestionDetails();
