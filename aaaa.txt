
import { useState, useEffect } from "react";
import {
  Container,
  Box,
  Typography,
  Fab,
} from "@mui/material";
import AddIcon from "@mui/icons-material/Add";

import TipForm from "../components/TipForm";
import type { Category } from "../models/category";
import { useApi } from "../hooks/useApi";

function Home(): JSX.Element {
  const [categories, setCategories] = useState<Category[]>([]);
  const [isPostFormOpen, setIsPostFormOpen] = useState<boolean>(false);

  const api = useApi();

  useEffect((): void => {
    api
      .fetch("/api/categories")
      .then((res: Response) => {
        if (!res.ok) {
          throw new Error("カテゴリ一覧の取得に失敗しました");
        }

        return res.json();
      })
      .then((categoryList: Category[]): void => {
        setCategories(categoryList);
      })
      .catch((error: unknown): void => {
        console.error(error);
      });
  }, [api]);

  const handleSubmit = (values: {
    categoryId: number;
    tipTitle: string;
    tipDetail: string;
    tipURL: string;
  }): void => {
    api
      .sendJson("/api/tips", values, {
        method: "POST",
        headers: {
          "Content-type": "application/json",
        },
      })
      .then((res: Response): void => {
        if (!res.ok) {
          throw new Error("投稿に失敗しました");
        }

        setIsPostFormOpen(false);
      })
      .catch((error: unknown): void => {
        console.error(error);
        alert("投稿に失敗しました。");
      });
  };

  return (
    <Container maxWidth={false}>
      <Box>
        <Typography
          sx={{
            color: "primary.main",
            fontWeight: 700,
            textAlign: "center",
            textTransform: "uppercase",
            letterSpacing: "0.15rem",
            display: "block",
            mb: 1,
          }}
        >
          Overview
        </Typography>

        <Typography
          component="h1"
          variant="h3"
          sx={{
            color: "black",
            fontWeight: 800,
            textAlign: "center",
            letterSpacing: "-0.03rem",
            fontSize: {
              xs: "2rem",
              sm: "2.5rem",
            },
            mb: 5,
          }}
        >
          投稿
        </Typography>

        <Fab
          color="primary"
          aria-label="add"
          onClick={() => setIsPostFormOpen(true)}
          sx={{
            position: "fixed",
            right: 32,
            bottom: 32,
          }}
        >
          <AddIcon />
        </Fab>

        {isPostFormOpen && (
          <TipForm
            initialValues={{
              categoryId: "",
              tipTitle: "",
              tipDetail: "",
              tipURL: "",
            }}
            categories={categories}
            onSubmit={handleSubmit}
            submitLabel="投稿する"
            onClose={() => setIsPostFormOpen(false)}
          />
        )}
      </Box>
    </Container>
  );
}

export default Home;
