2. NeedsList.tsx の共感処理だけ修正

今たぶんここが、

api.fetch(`/api/needEmpathy/${need.needId}`, {
  method,
});

みたいになっているはず。
これを下に変えて。

const handleEmpathyClick = async (need: Need): Promise<void> => {
  try {
    const method = need.empathyByCurrentUser ? "DELETE" : "POST";

    const response = await api.fetch(`/api/needs/${need.needId}/empathy`, {
      method,
    });

    if (!response.ok) {
      throw new Error("共感の更新に失敗しました");
    }

    const data: NeedEmpathyResponse = await response.json();

    setNeedList((prevList) =>
      prevList.map((item) =>
        item.needId === data.needId
          ? {
              ...item,
              empathyCount: data.empathyCount,
              empathyByCurrentUser: data.empathyByCurrentUser,
            }
          : item
      )
    );
  } catch (error) {
    console.error(error);
  }
};
3. Server側のController URLと合わせる

clientを上みたいにするなら、server側はこれ。

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/needs")
public class NeedEmpathyController {

    private final NeedEmpathyService needEmpathyService;
    private final UserInfoService userInfoService;

    @PostMapping("/{needId}/empathy")
    public NeedEmpathyResponseDto createNeedEmpathy(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long needId
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return needEmpathyService.createNeedEmpathy(needId, loginUser.getUserId());
    }

    @DeleteMapping("/{needId}/empathy")
    public NeedEmpathyResponseDto deleteNeedEmpathy(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long needId
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return needEmpathyService.deleteNeedEmpathy(needId, loginUser.getUserId());
    }
}
