// ===== スキル削除処理 追加ここから =====
const handleDeleteSkill = async (): Promise<void> => {
    if (!deleteSkillId) {
        setErrorMessage("削除するスキルを選択してください。");
        return;
    }

    const currentUserId = hrid;

    if (!currentUserId) {
        setErrorMessage("ユーザー情報を取得できませんでした。");
        return;
    }

    try {
        setIsDeletingSkill(true);
        setErrorMessage(null);
        setSuccessMessage(null);

        await api.delete(`/skill/${deleteSkillId}`, {
            params: {
                hrid: currentUserId,
            },
        });

        setAllAvailableSkills((prev) =>
            prev.filter(
                (skill) => String(skill.skillId) !== String(deleteSkillId)
            )
        );

        setSkillExperiences((prev) =>
            prev.filter(
                (skill) => String(skill.skillId) !== String(deleteSkillId)
            )
        );

        setCareers((prev) =>
            prev.map((career) => ({
                ...career,
                careerSkills: career.careerSkills.filter(
                    (skill) => String(skill.skillId) !== String(deleteSkillId)
                ),
            }))
        );

        setSuccessMessage("スキルを削除しました。");
        setDeleteSkillId("");
        setIsSkillDeleteModalOpen(false);
    } catch (error: unknown) {
        if (axios.isAxiosError(error)) {
            const errorData = error.response?.data;

            if (typeof errorData === "string") {
                setErrorMessage(errorData);
            } else if (
                typeof errorData === "object" &&
                errorData !== null &&
                "message" in errorData
            ) {
                setErrorMessage(
                    String((errorData as { message?: string }).message)
                );
            } else {
                setErrorMessage("スキルの削除に失敗しました。");
            }
        } else {
            setErrorMessage("スキルの削除に失敗しました。");
        }
    } finally {
        setIsDeletingSkill(false);
    }
};
