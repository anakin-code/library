package simplex.pj148.skippy.server.service;

import jakarta.persistence.EntityNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.TipsCreateDto;
import simplex.pj148.skippy.server.entity.Category;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CategoryRepository;
import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class TipService {

    private final TipRepository tipRepository;
    private final CategoryRepository categoryRepository;
    private final UserRepository userRepository;

    @Transactional
    public void createTip(TipsCreateDto input) {

        /*
         * 修正前：
         *
         * input.setCategoryId(1);
         * input.setTipTitle("title");
         * input.setTipDetail("detail");
         * input.setTipURL("url");
         *
         * これは絶対に消す。
         * フロントから来た値をサーバー側で上書きしてしまうため。
         */

        Category category = categoryRepository.findByCategoryId(input.getCategoryId())
                .orElseThrow(() -> new EntityNotFoundException(
                        "カテゴリDBに存在しないカテゴリ：" + input.getCategoryId() + " が選択されています。"
                ));

        /*
         * 仮ログインユーザー。
         * 認証をまだ作っていない/外しているなら、いったん userId=1 のユーザーで投稿させる。
         * data.sql に user_id = 1 のユーザーが必要。
         */
        User loginUser = userRepository.findByUserId(1L)
                .orElseThrow(() -> new EntityNotFoundException(
                        "ユーザーDBに存在しないユーザー：1 が選択されています。"
                ));

        LocalDateTime createdTime = LocalDateTime.now();

        Tip tip = Tip.builder()
                .user(loginUser)
                .category(category)
                .tipTitle(input.getTipTitle())
                .tipDetail(input.getTipDetail())
                .tipURL(input.getTipURL())
                .createdTime(createdTime)
                .deleteFlg(false)
                .build();

        tipRepository.save(tip);
    }
}

function TipForm({
  initialValues,
  onSubmit,
  submitLabel,
  onClose,
}: TipFormProps): JSX.Element {
  const [formData, setFormData] = useState<TipFormInputValues>(initialValues);

  const handleChange = (
    event: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>
  ): void => {
    const { name, value } = event.target;

    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  const handleSubmit = (event: React.FormEvent<HTMLFormElement>): void => {
    event.preventDefault();

    if (formData.categoryId === "") {
      alert("カテゴリを選択してください。");
      return;
    }

    if (formData.tipTitle.trim() === "") {
      alert("タイトルを入力してください。");
      return;
    }

    if (formData.tipDetail.trim() === "") {
      alert("内容を入力してください。");
      return;
    }

    onSubmit({
      categoryId: Number(formData.categoryId),
      tipTitle: formData.tipTitle,
      tipDetail: formData.tipDetail,
      tipURL: formData.tipURL,
    });
  };

  return (
    <Modal open onClose={onClose}>
      <Box
        component="form"
        onSubmit={handleSubmit}
        sx={{
          position: "absolute",
          top: "50%",
          left: "50%",
          transform: "translate(-50%, -50%)",
          width: {
            xs: "90%",
            sm: 520,
          },
          bgcolor: "background.paper",
          borderRadius: 3,
          boxShadow: 24,
          p: 4,
        }}
      >
        <Typography
          component="h2"
          variant="h5"
          sx={{
            fontWeight: 700,
            mb: 3,
          }}
        >
          投稿フォーム
        </Typography>

        <Stack spacing={2}>
          <TextField
            required
            select
            fullWidth
            label="カテゴリ"
            name="categoryId"
            value={formData.categoryId}
            onChange={handleChange}
          >
            <MenuItem value="" disabled>
              カテゴリを選択
            </MenuItem>

            {categories.map((category) => (

                {category.categoryName}

            ))}
