public static boolean isValidSymbol(String str) {
    if (str == null) {
        return false;
    }

    return str.matches(
        "[0-9][0-9ACDFGHJKLMNPRSTUWXY][0-9][0-9ACDFGHJKLMNPRSTUWXY]"
    );
}

例：

Checks.isValidSymbol("7203"); // true
Checks.isValidSymbol("130A"); // true
Checks.isValidSymbol("1A0C"); // true

Checks.isValidSymbol("123");  // false
Checks.isValidSymbol("A123"); // false
Checks.isValidSymbol("12A3"); // false：3桁目に英字
Checks.isValidSymbol("130B"); // false：Bは使用しない
Checks.isValidSymbol("130a"); // false：小文字

少し読みやすくするなら、定数に分けてもいいです。

public final class Checks {

    private static final String DIGIT = "[0-9]";
    private static final String DIGIT_OR_ALLOWED_LETTER =
            "[0-9ACDFGHJKLMNPRSTUWXY]";

    private Checks() {
    }

    public static boolean isValidSymbol(String str) {
        if (str == null) {
            return false;
        }

        return str.matches(
                DIGIT
                + DIGIT_OR_ALLOWED_LETTER
                + DIGIT
                + DIGIT_OR_ALLOWED_LETTER
        );
    }
}
