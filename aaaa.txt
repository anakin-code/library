import {
  Box,
  Button,
  Dialog,
  DialogContent,
  MenuItem,
  TextField,
  Typography,
} from "@mui/material";
import { useState } from "react";
import type { CategoryOption } from "../models/Master";
import { useApi } from "../lib/api";

type Props = {
  open: boolean;
  onClose: () => void;
  categories: CategoryOption[];
  onCreated: () => void;
};

type NeedCreateForm = {
  categoryId: string;
  needTitle: string;
  needDetail: string;
};

export default function SubmitForm({
  open,
  onClose,
  categories,
  onCreated,
}: Props): JSX.Element {
  const api = useApi();

  const [formData, setFormData] = useState<NeedCreateForm>({
    categoryId: "",
    needTitle: "",
    needDetail: "",
  });

  const [isSubmitting, setIsSubmitting] = useState(false);

  const titleCount = formData.needTitle.length;
  const detailCount = formData.needDetail.length;

  const canSubmit =
    formData.categoryId !== "" &&
    formData.needTitle.trim() !== "" &&
    formData.needDetail.trim() !== "" &&
    titleCount <= 30 &&
    detailCount <= 1000 &&
    !isSubmitting;

  const resetForm = (): void => {
    setFormData({
      categoryId: "",
      needTitle: "",
      needDetail: "",
    });
  };

  const handleClose = (): void => {
    if (isSubmitting) return;
    onClose();
  };

  const handleSubmit = async (): Promise<void> => {
    if (!canSubmit) return;

    try {
      setIsSubmitting(true);

      const res = await api.fetch("/api/needs", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          categoryId: Number(formData.categoryId),
          needTitle: formData.needTitle.trim(),
          needDetail: formData.needDetail.trim(),
        }),
      });

      if (!res.ok) {
        throw new Error("お悩み投稿に失敗しました");
      }

      resetForm();
      onClose();
      onCreated();
    } catch (error) {
      console.error(error);
      alert("投稿に失敗しました。入力内容やサーバーを確認してください。");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Dialog
      open={open}
      onClose={handleClose}
      fullWidth
      maxWidth="sm"
      slotProps={{
        paper: {
          sx: {
            borderRadius: 4,
            width: 420,
            maxWidth: "calc(100% - 32px)",
          },
        },
      }}
    >
      <DialogContent
        sx={{
          px: 6,
          py: 4,
          backgroundColor: "#fff7e8",
        }}
      >
        <Typography
          component="h2"
          sx={{
            fontSize: 24,
            fontWeight: 800,
            textAlign: "center",
            mb: 3,
            color: "black",
          }}
        >
          お悩み投稿
        </Typography>

        <Box
          sx={{
            display: "flex",
            flexDirection: "column",
            gap: 2.5,
          }}
        >
          <Box>
            <Typography sx={{ fontWeight: 700, mb: 0.5 }}>*</Typography>
            <TextField
              select
              fullWidth
              size="small"
              value={formData.categoryId}
              onChange={(event) =>
                setFormData((prev) => ({
                  ...prev,
                  categoryId: event.target.value,
                }))
              }
              placeholder="悩みカテゴリ"
              SelectProps={{
                displayEmpty: true,
              }}
              sx={{
                backgroundColor: "white",
                borderRadius: 2,
              }}
            >
              <MenuItem value="" disabled>
                悩みカテゴリ
              </MenuItem>
              {categories.map((category) => (
                <MenuItem
                  key={category.categoryId}
                  value={String(category.categoryId)}
                >
                  {category.categoryName}
                </MenuItem>
              ))}
            </TextField>
          </Box>

          <Box>
            <Typography sx={{ fontWeight: 700, mb: 0.5 }}>*</Typography>
            <TextField
              fullWidth
              size="small"
              placeholder="タイトル"
              value={formData.needTitle}
              onChange={(event) =>
                setFormData((prev) => ({
                  ...prev,
                  needTitle: event.target.value,
                }))
              }
              inputProps={{
                maxLength: 30,
              }}
              sx={{
                backgroundColor: "white",
                borderRadius: 2,
              }}
            />
            <Typography
              sx={{
                mt: 0.3,
                textAlign: "right",
                fontSize: 12,
                color: titleCount > 30 ? "error.main" : "text.secondary",
              }}
            >
              {titleCount}/30
            </Typography>
          </Box>

          <Box>
            <TextField
              fullWidth
              multiline
              minRows={4}
              placeholder="詳細"
              value={formData.needDetail}
              onChange={(event) =>
                setFormData((prev) => ({
                  ...prev,
                  needDetail: event.target.value,
                }))
              }
              inputProps={{
                maxLength: 1000,
              }}
              sx={{
                backgroundColor: "white",
                borderRadius: 2,
              }}
            />
            <Typography
              sx={{
                mt: 0.3,
                textAlign: "right",
                fontSize: 12,
                color: detailCount > 1000 ? "error.main" : "text.secondary",
              }}
            >
              {detailCount}/1000
            </Typography>

            <Typography
              sx={{
                mt: 0.5,
                textAlign: "center",
                fontSize: 11,
                color: "text.secondary",
              }}
            >
              ※投稿後の編集はできません。
            </Typography>
          </Box>

          <Box
            sx={{
              display: "flex",
              justifyContent: "center",
              gap: 2,
              mt: 1,
            }}
          >
            <Button
              variant="contained"
              onClick={handleClose}
              disabled={isSubmitting}
              sx={{
                backgroundColor: "#ead9b8",
                color: "black",
                borderRadius: 3,
                px: 4,
                fontWeight: 700,
                "&:hover": {
                  backgroundColor: "#dfc99d",
                },

            <Button

                backgroundColor: "#0068a8",

import {
  Alert,
  Box,
  Button,
  CircularProgress,
  Fab,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import type { SelectChangeEvent } from "@mui/material";
import { useCallback, useEffect, useMemo, useState } from "react";
import NeedCard from "../components/NeedCard";
import SubmitForm from "./SubmitForm";
import type { Need, NeedPageResponse } from "../models/Need";
import type {
  AffiliatedOption,
  CategoryOption,
  PositionOption,
} from "../models/Master";
import { useApi } from "../lib/api";

type SortBy = "new" | "empathy";

type SearchCondition = {
  keyword: string;
  affiliatedId: string;
  positionId: string;
  categoryId: string;
};

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);
  const [postOpen, setPostOpen] = useState(false);

  const [sortBy, setSortBy] = useState<SortBy>("new");
  const [page, setPage] = useState(0);
  const [totalPage, setTotalPage] = useState(0);
  const [totalElements, setTotalElements] = useState(0);

  const [searchCondition, setSearchCondition] = useState<SearchCondition>({
    keyword: "",
    affiliatedId: "",
    positionId: "",
    categoryId: "",
  });

  const [appliedCondition, setAppliedCondition] = useState<SearchCondition>({
    keyword: "",
    affiliatedId: "",
    positionId: "",
    categoryId: "",
  });

  const [affiliateds, setAffiliateds] = useState<AffiliatedOption[]>([]);
  const [positions, setPositions] = useState<PositionOption[]>([]);
  const [categories, setCategories] = useState<CategoryOption[]>([]);

  const [isLoading, setIsLoading] = useState(true);
  const [isMasterLoading, setIsMasterLoading] = useState(true);
  const [isServerStopping, setIsServerStopping] = useState(false);

  const size = 18;

  const requestUrl = useMemo(() => {
    const params = new URLSearchParams();

    params.set("page", String(page));
    params.set("size", String(size));
    params.set("orderBy", sortBy);

    if (appliedCondition.keyword.trim() !== "") {
      params.set("keyword", appliedCondition.keyword.trim());
    }

    if (appliedCondition.affiliatedId !== "") {
      params.set("affiliatedId", appliedCondition.affiliatedId);
    }

    if (appliedCondition.positionId !== "") {
      params.set("positionId", appliedCondition.positionId);
    }

    if (appliedCondition.categoryId !== "") {
      params.set("categoryId", appliedCondition.categoryId);
    }

    return `/api/needs/search?${params.toString()}`;
  }, [page, sortBy, appliedCondition]);

  const fetchNeeds = useCallback(async (): Promise<void> => {
    try {
      setIsLoading(true);
      setIsServerStopping(false);

      const res = await api.fetch(requestUrl);

      if (!res.ok) {
        throw new Error("Error fetching needs");
      }

      const data: NeedPageResponse = await res.json();

      setNeedList(data.content ?? []);
      setTotalPage(data.totalPages ?? 0);
      setTotalElements(data.totalElements ?? 0);
    } catch (error) {
      console.error(error);
      setNeedList([]);
      setTotalPage(0);
      setTotalElements(0);
      setIsServerStopping(true);
    } finally {
      setIsLoading(false);
    }
  }, [api, requestUrl]);

  const fetchMasters = useCallback(async (): Promise<void> => {
    try {
      setIsMasterLoading(true);

      const [affiliatedRes, positionRes, categoryRes] = await Promise.all([
        api.fetch("/api/affiliateds"),
        api.fetch("/api/positions"),
        api.fetch("/api/categories"),
      ]);

      if (!affiliatedRes.ok || !positionRes.ok || !categoryRes.ok) {
        throw new Error("Error fetching master data");
      }

      const affiliatedData: AffiliatedOption[] = await affiliatedRes.json();
      const positionData: PositionOption[] = await positionRes.json();
      const categoryData: CategoryOption[] = await categoryRes.json();

      setAffiliateds(affiliatedData);
      setPositions(positionData);
      setCategories(categoryData);
    } catch (error) {
      console.error(error);
      setAffiliateds([]);
      setPositions([]);
      setCategories([]);
    } finally {
      setIsMasterLoading(false);
    }
  }, [api]);

  useEffect(() => {
    fetchMasters();
  }, [fetchMasters]);

  useEffect(() => {
    fetchNeeds();
  }, [fetchNeeds]);

  const handleSortChange = (event: SelectChangeEvent): void => {
    setSortBy(event.target.value as SortBy);
    setPage(0);
  };

  const handleSearchChange =
    (field: keyof SearchCondition) =>
    (event: React.ChangeEvent<HTMLInputElement>): void => {
      setSearchCondition((prev) => ({
        ...prev,
        [field]: event.target.value,
      }));
    };

  const handleSelectSearchChange =
    (field: keyof SearchCondition) =>
    (event: SelectChangeEvent): void => {
      setSearchCondition((prev) => ({
        ...prev,
        [field]: event.target.value,
      }));
    };

  const handleSearchSubmit = (): void => {
    setAppliedCondition(searchCondition);
    setPage(0);
  };

  const handleReset = (): void => {
    const emptyCondition: SearchCondition = {
      keyword: "",
      affiliatedId: "",
      positionId: "",
      categoryId: "",
    };

    setSearchCondition(emptyCondition);
    setAppliedCondition(emptyCondition);
    setSortBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  const handleCreated = (): void => {
    setPage(0);
    fetchNeeds();
  };

  return (
    <>
      <Box
        sx={{
          width: "100%",
          minHeight: "100vh",
          px: { xs: 2, md: 5 },
          py: 3,
          backgroundColor: "#fff7e8",
        }}
      >
        <Box
          sx={{
            display: "grid",
            gridTemplateColumns: { xs: "1fr", md: "1fr 230px" },
            gap: 3,
            alignItems: "start",
          }}
        >
          <Box>
            <Box
              sx={{
                display: "flex",
                justifyContent: "center",
                alignItems: "center",
                position: "relative",
                mb: 3,
              }}
            >
              <Typography
                component="h1"
                sx={{
                  fontSize: 30,
                  fontWeight: 800,
                  color: "black",
                  textAlign: "center",
                }}
              >
                お悩み一覧
              </Typography>

              <FormControl
                size="small"
                sx={{
                  position: { xs: "static", md: "absolute" },
                  right: 0,
                  minWidth: 110,
                  ml: { xs: 2, md: 0 },
                }}
              >
                <Select
                  value={sortBy}
                  onChange={handleSortChange}
                  sx={{
                    backgroundColor: "white",
                    borderRadius: 1,
                    fontSize: 13,
                    fontWeight: 700,
                  }}
                >
                  <MenuItem value="new">新着順</MenuItem>
                  <MenuItem value="empathy">共感数順</MenuItem>
                </Select>
              </FormControl>
            </Box>

            {isServerStopping && (
              <Alert
                severity="error"
                sx={{
                  mb: 3,
                  justifyContent: "center",
                  maxWidth: 600,
                  mx: "auto",
                }}
              >
                サーバーが起動していません。しばらくたってからお試しください。
              </Alert>
            )}

            {isLoading ? (
              <CircularProgress
                aria-label="Loading"
                sx={{
                  display: "block",
                  mx: "auto",
                  my: 8,
                }}
              />
            ) : (
              <>
                {needList.length === 0 ? (
                  <Box
                    sx={{
                      textAlign: "center",
                      py: 8,
                      color: "text.secondary",
                    }}
                  >
                    <Typography variant="h6" sx={{ fontWeight: 700 }}>
                      お悩みがありません
                    </Typography>
                  </Box>
                ) : (
                  <Grid container spacing={2}>
                    {needList.map((need, index) => (
                      <Grid
                        item
                        xs={12}
                        sm={6}
                        md={4}
                        key={need.needId ?? index}
                      >
                        <NeedCard need={need} />
                      </Grid>
                    ))}
                  </Grid>
                )}

                {totalPage > 1 && (
                  <Stack
                    direction="row"
                    sx={{
                      mt: 4,
                      justifyContent: "center",
                      alignItems: "center",
                      gap: 1,
                      flexWrap: "wrap",
                    }}
                  >
                    <Button disabled={page <= 0} onClick={() => setPage(0)}>
                      First
                    </Button>

                    <Button disabled={page <= 0} onClick={() => setPage(page - 1)}>

                      次へ

                      onClick={() => setPage(totalPage - 1)}
                    >
                      Last
                    </Button>
                  </Stack>
                )}
              </>
            )}
          </Box>

          <Box
            sx={{
              backgroundColor: "#e7e1c9",
              borderRadius: 3,
              px: 2,
              py: 2.5,
              position: { md: "sticky" },
              top: { md: 24 },
            }}
          >
            <Typography
              sx={{
                fontSize: 18,
                fontWeight: 800,
                textAlign: "center",
                mb: 2,
              }}
            >
              絞り込み
            </Typography>

            <Stack spacing={1.5}>
              <Box>
                <Typography sx={{ fontSize: 13, fontWeight: 700 }}>
                  所属
                </Typography>
                <FormControl fullWidth size="small">

                    value={searchCondition.affiliatedId}

6. serverから一覧を取るAPI

client側はこの3つを叩く想定にしてる。

GET /api/affiliateds
GET /api/positions
GET /api/categories

ただ、今のserverで名前が違う可能性はある。

たとえば既存がこうなら、

GET /api/affiliated
GET /api/ranks
GET /api/category

client側のここを変えればいい。

const [affiliatedRes, positionRes, categoryRes] = await Promise.all([
  api.fetch("/api/affiliateds"),
  api.fetch("/api/positions"),
  api.fetch("/api/categories"),
]);

もしまだserverに一覧APIがないなら、追加するならこういうイメージ。

@RestController
@RequiredArgsConstructor
@RequestMapping("/api")
public class MasterController {

    private final AffiliatedRepository affiliatedRepository;
    private final RankRepository rankRepository;
    private final CategoryRepository categoryRepository;

    @GetMapping("/affiliateds")
    public List<AffiliatedOptionDto> getAffiliateds() {
        return affiliatedRepository.findAll()
                .stream()
                .map(a -> new AffiliatedOptionDto(
                        a.getAffiliatedId(),
                        a.getAffiliatedName()
                ))
                .toList();
    }

    @GetMapping("/positions")
    public List<PositionOptionDto> getPositions() {
        return rankRepository.findAll()
                .stream()
                .map(r -> new PositionOptionDto(
                        r.getRankId(),
                        r.getRankName()
                ))
                .toList();
    }

    @GetMapping("/categories")
    public List<CategoryOptionDto> getCategories() {
        return categoryRepository.findAll()
                .stream()
                .map(c -> new CategoryOptionDto(
                        c.getCategoryId(),
                        c.getCategoryName()
                ))
                .toList();
    }
}

DTOはこれ。

public record AffiliatedOptionDto(
        Long affiliatedId,
        String affiliatedName
) {
}
public record PositionOptionDto(
        Long positionId,
        String positionName
) {
}
public record CategoryOptionDto(
        Long categoryId,
        String categoryName
) {
}

PositionOptionDto は中身では rankId / rankName を positionId / positionName に変換して返す。
こうしておくと、client側は「職位 = position」として扱えて分かりやすい。

結論としては、昔のコードをベースにするなら、NeedsListに右サイドバーを追加して、マスタ取得useEffectを追加するのが本命。
投稿モーダルは SubmitForm の中に Dialog を持たせる形に統一すれば、Fabを押す前からボタンが見える問題も消える。
