import { useEffect, useState } from 'react';
import { useNavigate, useParams } from 'react-router-dom';

type Collection = {
  circulationId: number;
  serialNumber: number;
  title: string;
  borrowerName: string;
  isOnLend: boolean;
};

const restBaseUrl = import.meta.env.VITE_REST_BASE_URL;

export default function Return() {
  const { circulationId } = useParams();
  const navigate = useNavigate();

  const [collection, setCollection] =
    useState<Collection | null>(null);

  const [errorMessage, setErrorMessage] =
    useState<string | null>(null);

  const [successMessage, setSuccessMessage] =
    useState<string | null>(null);

  useEffect(() => {
    async function fetchData(): Promise<void> {
      if (!circulationId) {
        setErrorMessage('貸出履歴がありません。');
        return;
      }

      const response = await fetch(
        `${restBaseUrl}/circulations/${circulationId}`
      );

      if (response.status === 404) {
        setCollection(null);
        setErrorMessage('貸出履歴がありません。');
        return;
      }

      if (!response.ok) {
        setCollection(null);
        setErrorMessage('貸出情報の取得に失敗しました。');
        return;
      }

      const data = await response.json() as Collection;

      setCollection(data);

      if (!data.isOnLend) {
        setErrorMessage(
          `No.${data.serialNumber}:「${data.title}」は、すでに返却されています。`
        );
      } else {
        setErrorMessage(null);
      }
    }

    void fetchData();
  }, [circulationId]);

  async function handleReturn(): Promise<void> {
    if (!circulationId || collection === null) {
      return;
    }

    const response = await fetch(
      `${restBaseUrl}/circulations/${circulationId}/return`,
      {
        method: 'POST',
      }
    );

    if (response.status === 409) {
      setErrorMessage(
        `No.${collection.serialNumber}:「${collection.title}」は、すでに返却されています。`
      );
      setSuccessMessage(null);
      setCollection({
        ...collection,
        isOnLend: false,
      });
      return;
    }

    if (!response.ok) {
      setErrorMessage('返却処理に失敗しました。');
      return;
    }

    setSuccessMessage('返却しました。');
    setErrorMessage(null);

    setCollection({
      ...collection,
      isOnLend: false,
    });
  }

  const canReturn =
    collection !== null && collection.isOnLend;

  return (
    <section className="return-page">
      <h1>返却確認</h1>

      {errorMessage && (
        <p className="error-message">
          {errorMessage}
        </p>
      )}

      {successMessage && (
        <p className="success-message">
          {successMessage}
        </p>
      )}

      {collection && (
        <div>
          <p>蔵書番号: {collection.serialNumber}</p>
          <p>書籍名: {collection.title}</p>
          <p>借りた人: {collection.borrowerName}</p>
        </div>
      )}

      {collection && (
        <button
          type="button"
          onClick={handleReturn}
          disabled={!canReturn}
        >
          返却
        </button>
      )}

      <button
        type="button"
        onClick={() => navigate(-1)}
      >
        戻る
      </button>
    </section>
  );
}
