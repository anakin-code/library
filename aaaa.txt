        setDeleteSkillId("");

        const filtered = skills.filter(skill =>
            skill.skillName
                .toLowerCase()
                .includes(keyword.toLowerCase())
        );

        setDeleteSkillCandidates(filtered);
        setShowDeleteSkillCandidates(true);
    }}
