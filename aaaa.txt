import { useCallback, useEffect, useMemo, useState } from "react";
import {
  Alert,
  Box,
  Button,
  CircularProgress,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";

import NeedCard from "../components/NeedCard";
import SubmitForm from "./SubmitForm";
import type { Need, NeedPageResponse } from "../models/Need";
import { useApi } from "../lib/api";

type Affiliated = {
  affiliatedId: number;
  affiliatedName: string;
};

type Rank = {
  rankId: number;
  rankName: string;
};

type Category = {
  categoryId: number;
  categoryName: string;
};

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);

  const [orderBy, setOrderBy] = useState<string>("new");
  const [keyword, setKeyword] = useState<string>("");

  const [affiliatedId, setAffiliatedId] = useState<string>("");
  const [positionId, setPositionId] = useState<string>("");
  const [categoryId, setCategoryId] = useState<string>("");

  const [affiliatedList, setAffiliatedList] = useState<Affiliated[]>([]);
  const [rankList, setRankList] = useState<Rank[]>([]);
  const [categoryList, setCategoryList] = useState<Category[]>([]);

  const [page, setPage] = useState<number>(0);
  const [totalPage, setTotalPage] = useState<number>(0);

  const [isLoading, setIsLoading] = useState<boolean>(true);
  const [isServerStopping, setIsServerStopping] = useState<boolean>(false);
  const [postOpen, setPostOpen] = useState<boolean>(false);

  const requestUrl = useMemo((): string => {
    const params = new URLSearchParams();

    params.append("page", String(page));
    params.append("size", "18");
    params.append("orderBy", orderBy);

    if (keyword.trim() !== "") {
      params.append("keyword", keyword.trim());
    }

    if (affiliatedId !== "") {
      params.append("affiliatedId", affiliatedId);
    }

    if (positionId !== "") {
      params.append("positionId", positionId);
    }

    if (categoryId !== "") {
      params.append("categoryId", categoryId);
    }

    return `/api/needs/search?${params.toString()}`;
  }, [page, orderBy, keyword, affiliatedId, positionId, categoryId]);

  const fetchNeeds = useCallback(async (): Promise<void> => {
    try {
      setIsLoading(true);
      setIsServerStopping(false);

      const response = await api.fetch(requestUrl);

      if (!response.ok) {
        throw new Error("お悩み一覧の取得に失敗しました");
      }

      const data: NeedPageResponse = await response.json();

      setNeedList(data.content ?? []);
      setTotalPage(data.totalPages ?? 0);
    } catch (error) {
      console.error(error);
      setNeedList([]);
      setTotalPage(0);
      setIsServerStopping(true);
    } finally {
      setIsLoading(false);
    }
  }, [api, requestUrl]);

  const fetchMasterData = useCallback(async (): Promise<void> => {
    try {
      const [affiliatedResponse, rankResponse, categoryResponse] =
        await Promise.all([
          api.fetch("/api/affiliated"),
          api.fetch("/api/rank"),
          api.fetch("/api/category"),
        ]);

      if (affiliatedResponse.ok) {
        const data: Affiliated[] = await affiliatedResponse.json();
        setAffiliatedList(data ?? []);
      }

      if (rankResponse.ok) {
        const data: Rank[] = await rankResponse.json();
        setRankList(data ?? []);
      }

      if (categoryResponse.ok) {
        const data: Category[] = await categoryResponse.json();
        setCategoryList(data ?? []);
      }
    } catch (error) {
      console.error(error);
    }
  }, [api]);

  useEffect((): void => {
    fetchMasterData();
  }, [fetchMasterData]);

  useEffect((): void => {
    fetchNeeds();
  }, [fetchNeeds]);

  const handleSearch = (): void => {
    setPage(0);
    fetchNeeds();
  };

  const handleReset = (): void => {
    setKeyword("");
    setAffiliatedId("");
    setPositionId("");
    setCategoryId("");
    setOrderBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  return (
    <>
      <Box
        sx={{
          maxWidth: 1060,
          mx: "auto",
          px: 2,
          py: 2.5,
          color: "#071a33",
        }}
      >
        <Box
          sx={{
            display: "grid",
            gridTemplateColumns: {
              xs: "1fr",
              md: "1fr 190px",
            },
            columnGap: 2,
            alignItems: "start",
          }}
        >
          {/* 左側：一覧 */}
          <Box>
            <Box
              sx={{
                position: "relative",
                mb: 2,
              }}
            >
              <Typography
                component="h1"
                sx={{
                  textAlign: "center",
                  fontSize: 28,
                  fontWeight: 900,
                  color: "#071a33",
                }}
              >
                お悩み一覧
              </Typography>

              <FormControl
                size="small"
                sx={{
                  position: {
                    xs: "static",
                    md: "absolute",
                  },
                  right: 0,
                  top: 2,
                  minWidth: 105,
                  mt: {
                    xs: 1.5,
                    md: 0,
                  },
                  display: "flex",
                  alignItems: {
                    xs: "flex-end",
                    md: "stretch",
                  },
                }}
              >
                <Select
                  value={orderBy}
                  onChange={(event) => {
                    setOrderBy(event.target.value);
                    setPage(0);
                  }}
                  sx={{
                    height: 34,
                    borderRadius: 1.3,
                    backgroundColor: "#fff",
                    fontWeight: 800,
                    fontSize: 13,
                    color: "#071a33",
                  }}
                >
                  <MenuItem value="new">新着順</MenuItem>
                  <MenuItem value="empathy">共感数順</MenuItem>
                </Select>
              </FormControl>
            </Box>

            {isServerStopping ? (
              <Alert
                severity="error"
                sx={{
                  mb: 2,
                  justifyContent: "center",
                  maxWidth: 600,
                  mx: "auto",
                }}
              >
                サーバーが起動していません。しばらくたってからお試しください。
              </Alert>
            ) : isLoading ? (
              <CircularProgress
                aria-label="Loading"
                sx={{
                  display: "block",
                  mx: "auto",
                  my: 6,
                }}
              />
            ) : (
              <>
                <Grid container spacing={2}>
                  {needList.map((need: Need, index: number) => (
                    <Grid
                      size={{
                        xs: 12,
                        sm: 6,
                        md: 4,
                      }}
                      key={need.needId ?? index}
                    >
                      <NeedCard need={need} />
                    </Grid>
                  ))}
                </Grid>

                {totalPage > 1 && (
                  <Stack
                    direction="row"
                    sx={{
                      pt: 3,
                      justifyContent: "center",
                      alignItems: "center",
                    }}
                  >
                    <Pagination
                      count={totalPage}
                      page={page + 1}
                      onChange={handlePageChange}
                      color="primary"
                      shape="rounded"
                    />
                  </Stack>
                )}
              </>
            )}
          </Box>

          {/* 右側：絞り込み + 投稿ボタン */}
          <Box
            sx={{
              mt: {
                xs: 2,
                md: 5,
              },
              width: {
                xs: "100%",
                md: 190,
              },
            }}
          >
            <Box
              sx={{
                borderRadius: 2,
                p: 1.7,
                backgroundColor: "#f3ecdc",
              }}
            >
              <Typography
                sx={{
                  fontWeight: 900,
                  textAlign: "center",
                  mb: 1.7,
                  fontSize: 17,
                  color: "#071a33",
                }}
              >
                絞り込み
              </Typography>

              <Box
                sx={{
                  display: "grid",
                  gridTemplateColumns: "52px 1fr",
                  rowGap: 1.1,
                  columnGap: 1,
                  alignItems: "center",
                  mb: 1.7,
                }}
              >
                <Typography sx={{ fontSize: 12, fontWeight: 900 }}>
                  所属
                </Typography>

                <FormControl fullWidth size="small">
                  <Select
                    value={affiliatedId}
                    displayEmpty
                    onChange={(event) => {
                      setAffiliatedId(event.target.value);
                      setPage(0);
                    }}
                    sx={{
                      height: 30,
                      borderRadius: 1,
                      backgroundColor: "#fff",
                      fontSize: 11,
                      fontWeight: 700,
                    }}
                  >
                    <MenuItem value="">指定なし</MenuItem>
                    {affiliatedList.map((affiliated) => (
                      <MenuItem
                        key={affiliated.affiliatedId}
                        value={String(affiliated.affiliatedId)}
                      >
                        {affiliated.affiliatedName}
                      </MenuItem>
                    ))}
                  </Select>
                </FormControl>

                <Typography sx={{ fontSize: 12, fontWeight: 900 }}>
                  職位
                </Typography>

                <FormControl fullWidth size="small">
                  <Select
                    value={positionId}
                    displayEmpty
                    onChange={(event) => {
                      setPositionId(event.target.value);
                      setPage(0);
                    }}
                    sx={{
                      height: 30,
                      borderRadius: 1,
                      backgroundColor: "#fff",
                      fontSize: 11,
                      fontWeight: 700,
                    }}
                  >
                    <MenuItem value="">指定なし</MenuItem>
                    {rankList.map((rank) => (
                      <MenuItem key={rank.rankId} value={String(rank.rankId)}>
                        {rank.rankName}
                      </MenuItem>
                    ))}
                  </Select>
                </FormControl>

                <Typography
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    lineHeight: 1.2,
                  }}
                >
                  悩み
                  <br />
                  カテゴリ
                </Typography>

                <FormControl fullWidth size="small">
                  <Select
                    value={categoryId}
                    displayEmpty
                    onChange={(event) => {
                      setCategoryId(event.target.value);
                      setPage(0);
                    }}
                    sx={{
                      height: 30,
                      borderRadius: 1,
                      backgroundColor: "#fff",
                      fontSize: 11,
                      fontWeight: 700,
                    }}
                  >
                    <MenuItem value="">指定なし</MenuItem>
                    {categoryList.map((category) => (
                      <MenuItem
                        key={category.categoryId}
                        value={String(category.categoryId)}
                      >
                        {category.categoryName}
                      </MenuItem>
                    ))}
                  </Select>
                </FormControl>
              </Box>

              <Box
                sx={{
                  borderTop: "1px solid rgba(0,0,0,0.15)",
                  pt: 1.4,
                }}
              >
                <Typography
                  sx={{
                    fontWeight: 900,
                    textAlign: "center",
                    mb: 1.1,
                    fontSize: 16,
                    color: "#071a33",
                  }}
                >
                  検索
                </Typography>

                <TextField
                  fullWidth
                  size="small"
                  placeholder="人名/キーワード"
                  value={keyword}
                  onChange={(event) => setKeyword(event.target.value)}
                  sx={{
                    mb: 1.2,
                    backgroundColor: "#fff",
                    borderRadius: 1,
                    "& .MuiOutlinedInput-root": {
                      height: 30,
                      borderRadius: 1,
                    },
                    "& input": {
                      fontSize: 10.5,
                    },
                  }}
                />

                <Stack direction="row" spacing={1}>
                  <Button
                    fullWidth
                    variant="outlined"
                    size="small"
                    onClick={handleReset}
                    sx={{
                      height: 28,
                      borderRadius: 1,
                      fontSize: 10.5,
                      fontWeight: 900,
                      color: "#071a33",
                      borderColor: "#d8d2c8",
                      backgroundColor: "#fff",
                    }}
                  >
                    リセット
                  </Button>

                  <Button
                    fullWidth
                    variant="contained"
                    size="small"
                    onClick={handleSearch}
                    sx={{
                      height: 28,
                      borderRadius: 1,
                      fontSize: 10.5,
                      fontWeight: 900,
                      backgroundColor: "#0054a6",
                      "&:hover": {
                        backgroundColor: "#004080",
                      },
                    }}
                  >
                    実行
                  </Button>
                </Stack>
              </Box>
            </Box>

            <Button
              variant="contained"
              onClick={() => setPostOpen(true)}
              disabled={isServerStopping}
              sx={{
                mt: 1.7,
                width: "100%",
                height: 58,
                borderRadius: "999px",
                backgroundColor: "#0054a6",
                color: "#fff",
                fontSize: 18,
                fontWeight: 900,
                lineHeight: 1.15,
                boxShadow: "0 8px 18px rgba(0,84,166,0.25)",
                "&:hover": {
                  backgroundColor: "#004080",
                },
                "&.Mui-disabled": {
                  backgroundColor: "#b0b0b0",
                  color: "#fff",
                },
              }}
            >
              お悩み
              <br />
              投稿
            </Button>
          </Box>
        </Box>
      </Box>

      <SubmitForm open={postOpen} onClose={() => setPostOpen(false)} />
    </>
  );
}
