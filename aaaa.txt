package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.*;
import simplex.pj148.skippy.server.controller.dto.auth.ProfileDto;
import simplex.pj148.skippy.server.controller.dto.response.UserInfoResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.UserInfoService;

@RestController
@RequestMapping("/api/me")
@RequiredArgsConstructor
public class AuthController {

    private final UserInfoService userInfoService;

    @GetMapping
    public ProfileDto getMe(@AuthenticationPrincipal Jwt jwt) {
        return switch (jwt) {
            case Jwt token -> {
                var cognitoResponse = userInfoService.retrieve(token);
                yield new ProfileDto(
                        cognitoResponse.name(),
                        cognitoResponse.email(),
                        cognitoResponse.givenName(),
                        cognitoResponse.familyName()
                );
            }
            default -> null;
        };
    }

    @GetMapping("/profile")
    public UserInfoResponseDto aboutMe(@AuthenticationPrincipal Jwt jwt) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return UserInfoResponseDto.builder()
                .userId(loginUser.getUserId())
                .userName(loginUser.getUserName())
                .selfIntroduction(loginUser.getSelfIntroduction())
                .affiliatedName(loginUser.getAffiliated().getAffiliatedName())
                .rankName(loginUser.getRank().getRankName())
                .mailAddress(loginUser.getMailAddress())
                .build();
    }

    @GetMapping("/id")
    public Long getUserId(@AuthenticationPrincipal Jwt jwt) {
        return userInfoService.getUserFromJwt(jwt).getUserId();
    }
}

package simplex.pj148.skippy.server.controller;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.data.web.PageableDefault;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.*;
import simplex.pj148.skippy.server.controller.dto.create.TipDto;
import simplex.pj148.skippy.server.controller.dto.response.TipDetailResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.TipShowResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.*;

@RestController
@RequestMapping("/api/tips")
@RequiredArgsConstructor
public class TipController {

    private final TipService tipService;
    private final TipCreateService tipCreateService;
    private final TipDetailService tipDetailService;
    private final TipUpdateService tipUpdateService;
    private final TipDeleteService tipDeleteService;
    private final UserInfoService userInfoService;

    @GetMapping
    public Page<TipShowResponseDto> listTips(
            @AuthenticationPrincipal Jwt jwt,
            @RequestParam(defaultValue = "createdTime") String orderBy,
            @RequestParam(defaultValue = "0") Long categoryId,
            @PageableDefault(
                    size = 20,
                    sort = "createdTime",
                    direction = Sort.Direction.DESC
            ) Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return tipService.findAllTips(
                loginUser,
                pageable,
                orderBy,
                categoryId
        );
    }

    @GetMapping("/search")
    public Page<TipShowResponseDto> searchTips(
            @AuthenticationPrincipal Jwt jwt,
            @RequestParam String context,
            @RequestParam String q,
            @PageableDefault(
                    size = 20,
                    sort = "createdTime",
                    direction = Sort.Direction.DESC
            ) Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return tipService.searchTips(
                q,
                context,
                loginUser,
                pageable
        );
    }

    @GetMapping("/{tipId}")
    public ResponseEntity<TipDetailResponseDto> getTipDetail(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long tipId
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        TipDetailResponseDto response =
                tipDetailService.getTipDetail(loginUser, tipId);

        return ResponseEntity.ok(response);
    }

    @PostMapping
    public ResponseEntity<Void> createTip(
            @AuthenticationPrincipal Jwt jwt,
            @RequestBody @Valid TipDto input
    ) {
        Long userId = userInfoService.getUserFromJwt(jwt).getUserId();

        input.setUserId(userId);

        tipCreateService.registerNewTip(input);

        return ResponseEntity.status(HttpStatus.CREATED).build();
    }

    @GetMapping("/{tipId}/edit")
    public ResponseEntity<TipDto> getEditTip(
            @PathVariable Long tipId
    ) {
        return ResponseEntity.ok(
                tipUpdateService.getUpdateTip(tipId)
        );
    }

    @PutMapping("/{tipId}")
    public ResponseEntity<Void> updateTip(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long tipId,
            @RequestBody @Valid TipDto input
    ) {
        Long userId = userInfoService.getUserFromJwt(jwt).getUserId();

        TipDetailResponseDto tipDetail =
                tipDetailService.getTipDetail(
                        userInfoService.getUserFromJwt(jwt),
                        tipId
                );

        if (!userId.equals(tipDetail.getUserId())) {
            throw new RuntimeException("投稿者以外は更新できません");
        }

        input.setUserId(userId);

        tipUpdateService.updateTip(tipId, input);

        return ResponseEntity.ok().build();
    }

    @DeleteMapping("/{tipId}")
    public ResponseEntity<Void> deleteTip(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long tipId
    ) {
        Long userId = userInfoService.getUserFromJwt(jwt).getUserId();

        TipDetailResponseDto tipDetail =
                tipDetailService.getTipDetail(
                        userInfoService.getUserFromJwt(jwt),
                        tipId
                );

        if (!userId.equals(tipDetail.getUserId())) {
            throw new RuntimeException("投稿者以外は削除できません");
        }

        tipDeleteService.deleteTip(tipId);

        return ResponseEntity.noContent().build();
    }
}

deleteFlgなしで変わるところ

旧Controllerでは、削除系がこういう形だった。

@PutMapping("/{contentsId}/delete")

今回の物理削除では、REST的にはこっちが自然。

@DeleteMapping("/{tipId}")

コメントも同じ。

@DeleteMapping("/{commentId}")

つまり今回のControllerでは、削除は全部 DELETE に寄せる。

重要な注意

NotificationController の findMyNotifications は今 List<Notification> を返しているけど、本当はEntityを直接返さずに NotificationResponseDto にした方がいい。
最初に動かすだけならEntity返しでも動くけど、循環参照が出る可能性があるから、あとでDTO変換にした方が安全。

今の完全版は、DB変更後にまずControllerを成立させるための構成。
次にやるなら、NotificationService.findMyNotifications の戻り値を List<NotificationResponseDto> に変えるのがいい。
