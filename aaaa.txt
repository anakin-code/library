投稿フォームボタンは表示するが、まだ投稿フォームへ遷移しない
ページングUIを下に表示する
検索・絞り込みしたら 1ページ目に戻す
import React, { useCallback, useEffect, useMemo, useState } from "react";
import {
  Box,
  Button,
  Card,
  CardContent,
  CircularProgress,
  FormControl,
  InputLabel,
  MenuItem,
  Select,
  SelectChangeEvent,
  TextField,
  Typography,
} from "@mui/material";
import { useNavigate } from "react-router-dom";

type Need = {
  needId: number;
  needTitle: string;
  needDetail: string;
  categoryId?: number;
  categoryName?: string;
  userId?: number;
  userName?: string;
  affiliatedName?: string;
  positionName?: string;
  empathyCount?: number;
  answerCount?: number;
  empathyByCurrentUser?: boolean;
  editFlg?: boolean;
  createdTime?: string;
};

type NeedPageResponse = {
  content: Need[];
  totalPages: number;
  totalElements: number;
  number: number;
  size: number;
};

type MasterItem = {
  id: number;
  name: string;
};

type CategoryResponse = {
  categoryId: number;
  categoryName: string;
};

type AffiliatedResponse = {
  affiliatedId: number;
  affiliatedName: string;
};

type PositionResponse = {
  positionId: number;
  positionName: string;
};

type NeedListProps = {
  api: {
    fetch: (url: string, options?: RequestInit) => Promise<Response>;
  };
};

const NeedList: React.FC<NeedListProps> = ({ api }) => {
  const navigate = useNavigate();

  const [needList, setNeedList] = useState<Need[]>([]);

  const [keywordInput, setKeywordInput] = useState<string>("");
  const [keyword, setKeyword] = useState<string>("");

  const [selectedAffiliatedId, setSelectedAffiliatedId] = useState<string>("");
  const [selectedPositionId, setSelectedPositionId] = useState<string>("");
  const [selectedCategoryId, setSelectedCategoryId] = useState<string>("");
  const [orderBy, setOrderBy] = useState<string>("createdTime");

  const [affiliatedList, setAffiliatedList] = useState<MasterItem[]>([]);
  const [positionList, setPositionList] = useState<MasterItem[]>([]);
  const [categoryList, setCategoryList] = useState<MasterItem[]>([]);

  const [currentPage, setCurrentPage] = useState<number>(0);
  const [totalPage, setTotalPage] = useState<number>(0);
  const [pageSize] = useState<number>(6);

  const [isLoading, setIsLoading] = useState<boolean>(false);
  const [isServerStopping, setIsServerStopping] = useState<boolean>(false);

  const requestUrl = useMemo(() => {
    const params = new URLSearchParams();

    params.append("page", String(currentPage));
    params.append("size", String(pageSize));

    if (keyword.trim() !== "") {
      params.append("keyword", keyword.trim());
    }

    if (selectedAffiliatedId !== "") {
      params.append("affiliatedId", selectedAffiliatedId);
    }

    if (selectedPositionId !== "") {
      params.append("positionId", selectedPositionId);
    }

    if (selectedCategoryId !== "") {
      params.append("categoryId", selectedCategoryId);
    }

    if (orderBy !== "") {
      params.append("orderBy", orderBy);
    }

    return `/api/needs?${params.toString()}`;
  }, [
    currentPage,
    pageSize,
    keyword,
    selectedAffiliatedId,
    selectedPositionId,
    selectedCategoryId,
    orderBy,
  ]);

  const fetchNeeds = useCallback(async (): Promise<void> => {
    try {
      setIsLoading(true);
      setIsServerStopping(false);

      const response = await api.fetch(requestUrl);

      if (!response.ok) {
        throw new Error("お悩み一覧の取得に失敗しました");
      }

      const data: NeedPageResponse = await response.json();

      setNeedList(data.content ?? []);
      setTotalPage(data.totalPages ?? 0);

      if (data.totalPages > 0 && currentPage >= data.totalPages) {
        setCurrentPage(data.totalPages - 1);
      }
    } catch (error) {
      console.error(error);
      setNeedList([]);
      setTotalPage(0);
      setIsServerStopping(true);
    } finally {
      setIsLoading(false);
    }
  }, [api, requestUrl, currentPage]);

  const fetchMasters = useCallback(async (): Promise<void> => {
    try {
      const [categoryResponse, affiliatedResponse, positionResponse] =
        await Promise.all([
          api.fetch("/api/categories"),
          api.fetch("/api/affiliateds"),
          api.fetch("/api/positions"),
        ]);

      if (categoryResponse.ok) {
        const categories: CategoryResponse[] = await categoryResponse.json();
        setCategoryList(
          categories.map((category) => ({
            id: category.categoryId,
            name: category.categoryName,
          }))
        );
      }

      if (affiliatedResponse.ok) {
        const affiliateds: AffiliatedResponse[] =
          await affiliatedResponse.json();
        setAffiliatedList(
          affiliateds.map((affiliated) => ({
            id: affiliated.affiliatedId,
            name: affiliated.affiliatedName,
          }))
        );
      }

      if (positionResponse.ok) {
        const positions: PositionResponse[] = await positionResponse.json();
        setPositionList(
          positions.map((position) => ({
            id: position.positionId,
            name: position.positionName,
          }))
        );
      }
    } catch (error) {
      console.error("マスタ取得に失敗しました", error);
    }
  }, [api]);

  useEffect(() => {
    fetchMasters();
  }, [fetchMasters]);

  useEffect(() => {
    fetchNeeds();
  }, [fetchNeeds]);

  const handleSearch = (): void => {
    setKeyword(keywordInput);
    setCurrentPage(0);
  };

  const handleReset = (): void => {
    setKeywordInput("");
    setKeyword("");
    setSelectedAffiliatedId("");
    setSelectedPositionId("");
    setSelectedCategoryId("");
    setOrderBy("createdTime");
    setCurrentPage(0);
  };

  const handleChangeAffiliated = (event: SelectChangeEvent): void => {
    setSelectedAffiliatedId(event.target.value);
    setCurrentPage(0);
  };

  const handleChangePosition = (event: SelectChangeEvent): void => {
    setSelectedPositionId(event.target.value);
    setCurrentPage(0);
  };

  const handleChangeCategory = (event: SelectChangeEvent): void => {
    setSelectedCategoryId(event.target.value);
    setCurrentPage(0);
  };

  const handleChangeOrderBy = (event: SelectChangeEvent): void => {
    setOrderBy(event.target.value);
    setCurrentPage(0);
  };

  const handleClickNeedCard = (needId: number): void => {
    navigate(`/needs/${needId}`);
  };

  const handleClickCreateNeed = (): void => {
    // まだ投稿フォームへ遷移しない
    console.log("お悩み投稿ボタンが押されました");
  };

  const handleClickEmpathy = async (
    event: React.MouseEvent<HTMLButtonElement>,
    need: Need
  ): Promise<void> => {
    event.stopPropagation();

    try {
      const response = await api.fetch(`/api/needEmpathy/${need.needId}`, {
        method: need.empathyByCurrentUser ? "DELETE" : "POST",
      });

      if (!response.ok) {
        throw new Error("共感更新に失敗しました");
      }

      setNeedList((prevNeedList) =>
        prevNeedList.map((currentNeed) => {
          if (currentNeed.needId !== need.needId) {
            return currentNeed;
          }

          const currentCount = currentNeed.empathyCount ?? 0;
          const alreadyEmpathy = currentNeed.empathyByCurrentUser ?? false;

          return {
            ...currentNeed,
            empathyByCurrentUser: !alreadyEmpathy,
            empathyCount: alreadyEmpathy
              ? Math.max(currentCount - 1, 0)
              : currentCount + 1,
          };
        })
      );
    } catch (error) {
      console.error(error);
    }
  };

  const isFirstPage = currentPage === 0;
  const isLastPage = totalPage === 0 || currentPage >= totalPage - 1;

  return (
    <Box sx={{ p: 4 }}>
      <Typography
        variant="h4"
        sx={{
          textAlign: "center",
          fontWeight: "bold",
          mb: 4,
        }}
      >
        お悩み一覧
      </Typography>

      <Box
        sx={{
          display: "flex",
          gap: 3,
          alignItems: "flex-start",
        }}
      >
        <Box sx={{ flex: 1 }}>
          {isLoading && (
            <Box sx={{ display: "flex", justifyContent: "center", mt: 6 }}>
              <CircularProgress />
            </Box>
          )}

          {!isLoading && isServerStopping && (
            <Typography sx={{ textAlign: "center", mt: 6 }}>
              サーバーに接続できません。
            </Typography>
          )}

          {!isLoading && !isServerStopping && needList.length === 0 && (
            <Typography sx={{ textAlign: "center", mt: 6 }}>
              表示できるお悩みがありません。
            </Typography>
          )}

          {!isLoading && !isServerStopping && needList.length > 0 && (
            <Box
              sx={{
                display: "grid",
                gridTemplateColumns: "repeat(3, 1fr)",
                gap: 3,
              }}
            >
              {needList.map((need) => (
                <Card
                  key={need.needId}
                  onClick={() => handleClickNeedCard(need.needId)}
                  sx={{
                    minHeight: 210,
                    borderRadius: 3,
                    cursor: "pointer",
                    boxShadow: 2,
                    transition: "0.2s",
                    "&:hover": {
                      boxShadow: 6,
                      transform: "translateY(-2px)",
                    },
                  }}
                >
                  <CardContent>
                    <Box sx={{ textAlign: "center", mb: 1 }}>
                      <Typography
                        sx={{
                          display: "inline-block",
                          px: 1.5,
                          py: 0.5,
                          borderRadius: 1,
                          backgroundColor: "#005bac",
                          color: "white",
                          fontSize: 12,
                          fontWeight: "bold",
                        }}
                      >
                        {need.categoryName ?? "悩みカテゴリ"}
                      </Typography>
                    </Box>

                    <Typography
                      variant="h6"
                      sx={{
                        fontWeight: "bold",
                        textAlign: "center",
                        mb: 2,
                        minHeight: 56,
                        display: "-webkit-box",
                        WebkitLineClamp: 2,
                        WebkitBoxOrient: "vertical",
                        overflow: "hidden",
                      }}
                    >
                      {need.needTitle}
                    </Typography>

                    <Typography
                      sx={{
                        fontSize: 14,
                        color: "text.secondary",
                        mb: 2,
                        minHeight: 40,
                        display: "-webkit-box",
                        WebkitLineClamp: 2,
                        WebkitBoxOrient: "vertical",
                        overflow: "hidden",
                      }}
                    >
                      {need.needDetail}
                    </Typography>

                    <Box
                      sx={{
                        display: "flex",
                        justifyContent: "space-between",
                        alignItems: "center",
                        fontSize: 13,
                        color: "text.secondary",
                        mb: 1,
                      }}
                    >
                      <Typography sx={{ fontSize: 13 }}>
                        {need.affiliatedName ?? ""}
                        {need.positionName ? ` ${need.positionName}` : ""}
                        {need.userName ? ` ${need.userName}` : ""}
                      </Typography>

                      <Typography sx={{ fontSize: 13 }}>
                        回答 {need.answerCount ?? 0}
                      </Typography>
                    </Box>

                    <Box
                      sx={{
                        display: "flex",
                        justifyContent: "flex-end",
                        alignItems: "center",
                      }}
                    >
                      <Button
                        onClick={(event) => handleClickEmpathy(event, need)}
                        sx={{
                          minWidth: 72,
                          borderRadius: 5,
                          color: "black",
                          backgroundColor: need.empathyByCurrentUser
                            ? "#ffe082"
                            : "#f5f5f5",
                          boxShadow: need.empathyByCurrentUser ? 3 : 1,
                          "&:hover": {
                            backgroundColor: "#ffe082",
                            boxShadow: 4,
                          },
                        }}
                      >
                        共感 👏 {need.empathyCount ?? 0}
                      </Button>
                    </Box>
                  </CardContent>
                </Card>
              ))}
            </Box>
          )}

          <Box
            sx={{
              display: "flex",
              justifyContent: "space-between",
              alignItems: "center",
              mt: 5,
              px: 4,
            }}
          >
            <Button
              disabled={isFirstPage}
              onClick={() => setCurrentPage(0)}
              sx={{
                color: isFirstPage ? "gray" : "black",
              }}
            >
              ＜ First
            </Button>

            <Button
              disabled={isFirstPage}
              onClick={() => setCurrentPage((prev) => Math.max(prev - 1, 0))}
              sx={{
                color: isFirstPage ? "gray" : "black",
              }}
            >
              ＜ 前へ
            </Button>

            <Typography sx={{ color: "text.secondary" }}>
              {totalPage === 0 ? 0 : currentPage + 1} / {totalPage}
            </Typography>

            <Button
              disabled={isLastPage}
              onClick={() =>
                setCurrentPage((prev) => Math.min(prev + 1, totalPage - 1))
              }
              sx={{
                color: isLastPage ? "gray" : "black",
              }}
            >
              次へ ＞
            </Button>

            <Button
              disabled={isLastPage}
              onClick={() => setCurrentPage(Math.max(totalPage - 1, 0))}
              sx={{
                color: isLastPage ? "gray" : "black",
              }}
            >
              Last ＞
            </Button>
          </Box>
        </Box>

        <Box
          sx={{
            width: 240,
            p: 2,
            borderRadius: 2,
            backgroundColor: "#f1eadf",
            position: "sticky",
            top: 24,
          }}
        >
          <Typography
            variant="h6"
            sx={{
              fontWeight: "bold",
              textAlign: "center",
              mb: 2,
            }}
          >
            絞り込み
          </Typography>

          <FormControl fullWidth size="small" sx={{ mb: 2 }}>
            <InputLabel>所属</InputLabel>
            <Select
              value={selectedAffiliatedId}
              label="所属"
              onChange={handleChangeAffiliated}
            >
              <MenuItem value="">指定なし</MenuItem>
              {affiliatedList.map((affiliated) => (
                <MenuItem key={affiliated.id} value={String(affiliated.id)}>
                  {affiliated.name}
                </MenuItem>
              ))}
            </Select>
          </FormControl>

          <FormControl fullWidth size="small" sx={{ mb: 2 }}>
            <InputLabel>職位</InputLabel>
            <Select
              value={selectedPositionId}
              label="職位"
              onChange={handleChangePosition}
            >
              <MenuItem value="">指定なし</MenuItem>
              {positionList.map((position) => (
                <MenuItem key={position.id} value={String(position.id)}>
                  {position.name}
                </MenuItem>
              ))}
            </Select>
          </FormControl>

          <FormControl fullWidth size="small" sx={{ mb: 2 }}>
            <InputLabel>悩みカテゴリ</InputLabel>
            <Select
              value={selectedCategoryId}
              label="悩みカテゴリ"
              onChange={handleChangeCategory}
            >
              <MenuItem value="">指定なし</MenuItem>
              {categoryList.map((category) => (
                <MenuItem key={category.id} value={String(category.id)}>
                  {category.name}
                </MenuItem>
              ))}
            </Select>
          </FormControl>

          <FormControl fullWidth size="small" sx={{ mb: 2 }}>
            <InputLabel>並び替え</InputLabel>
            <Select value={orderBy} label="並び替え" onChange={handleChangeOrderBy}>
              <MenuItem value="createdTime">新着順</MenuItem>
              <MenuItem value="empathyCount">共感が多い順</MenuItem>
              <MenuItem value="answerCount">回答が多い順</MenuItem>
            </Select>
          </FormControl>

          <Typography
            variant="subtitle1"
            sx={{
              fontWeight: "bold",
              textAlign: "center",
              mt: 3,
              mb: 1,
            }}
          >
            検索
          </Typography>

          <TextField
            fullWidth
            size="small"
            placeholder="キーワード"
            value={keywordInput}
            onChange={(event) => setKeywordInput(event.target.value)}
            onKeyDown={(event) => {
              if (event.key === "Enter") {
                handleSearch();
              }
            }}
            sx={{ mb: 2, backgroundColor: "white" }}
          />

          <Box sx={{ display: "flex", gap: 1, mb: 3 }}>
            <Button
              fullWidth
              variant="outlined"
              onClick={handleReset}
              sx={{
                backgroundColor: "white",
              }}
            >
              リセット
            </Button>

            <Button fullWidth variant="contained" onClick={handleSearch}>
              実行
            </Button>
          </Box>

          <Button
            fullWidth
            variant="contained"
            onClick={handleClickCreateNeed}
            sx={{
              borderRadius: 5,
              py: 1.5,
              fontWeight: "bold",
              backgroundColor: "#005bac",
            }}
          >
            お悩み
            <br />
            投稿
          </Button>
        </Box>
      </Box>
    </Box>
  );
};

export default NeedList;

  method: need.empathyByCurrentUser ? "DELETE" : "POST",
});

を

const response = await fetch(`/api/needEmpathy/${need.needId}`, {
  method: need.empathyByCurrentUser ? "DELETE" : "POST",
});
