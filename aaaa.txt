
import {
  Alert,
  Box,
  Button,
  CircularProgress,
  Dialog,
  DialogContent,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import type { SelectChangeEvent } from "@mui/material";
import { useEffect, useMemo, useState } from "react";
import NeedCard from "../components/NeedCard";
import SubmitNeeds from "../components/SubmitNeeds";
import { useApi } from "../lib/useApi";
import type { Affiliated } from "../models/Affiliated";
import type { Category } from "../models/Category";
import type { Need, NeedPageResponse } from "../models/Need";
import type { Rank } from "../models/Rank";

type SortBy = "new" | "empathy";

type SearchCondition = {
  keyword: string;
  affiliatedId: string;
  positionId: string;
  categoryId: string;
};

const emptySearchCondition: SearchCondition = {
  keyword: "",
  affiliatedId: "",
  positionId: "",
  categoryId: "",
};

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);
  const [postOpen, setPostOpen] = useState(false);

  const [sortBy, setSortBy] = useState<SortBy>("new");
  const [page, setPage] = useState(0);
  const [totalPage, setTotalPage] = useState(0);

  const [searchCondition, setSearchCondition] =
    useState<SearchCondition>(emptySearchCondition);

  const [appliedCondition, setAppliedCondition] =
    useState<SearchCondition>(emptySearchCondition);

  const [affiliateds, setAffiliateds] = useState<Affiliated[]>([]);
  const [ranks, setRanks] = useState<Rank[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);

  const [isLoading, setIsLoading] = useState(true);
  const [isMasterLoading, setIsMasterLoading] = useState(true);
  const [isServerStopping, setIsServerStopping] = useState(false);

  // 投稿成功後に一覧を再取得するためのキー
  const [reloadKey, setReloadKey] = useState(0);

  // 画像の理想形だと1ページ6件が近い
  const size = 6;

  const requestUrl = useMemo(() => {
    const params = new URLSearchParams();

    params.set("page", String(page));
    params.set("size", String(size));
    params.set("orderBy", sortBy);

    if (appliedCondition.keyword.trim() !== "") {
      params.set("keyword", appliedCondition.keyword.trim());
    }

    if (appliedCondition.affiliatedId !== "") {
      params.set("affiliatedId", appliedCondition.affiliatedId);
    }

    if (appliedCondition.positionId !== "") {
      params.set("positionId", appliedCondition.positionId);
    }

    if (appliedCondition.categoryId !== "") {
      params.set("categoryId", appliedCondition.categoryId);
    }

    return `/api/needs/search?${params.toString()}`;
  }, [page, sortBy, appliedCondition]);

  // 所属・職位・カテゴリ一覧を取得
  // 画面表示時に1回だけ動けばいいので、依存配列は [] にしている
  useEffect(() => {
    let ignore = false;

    const fetchMasters = async (): Promise<void> => {
      try {
        setIsMasterLoading(true);

        const [affiliatedResponse, rankResponse, categoryResponse] =
          await Promise.all([
            api.fetch("/api/affiliateds"),
            api.fetch("/api/ranks"),
            api.fetch("/api/categories"),
          ]);

        if (
          !affiliatedResponse.ok ||
          !rankResponse.ok ||
          !categoryResponse.ok
        ) {
          throw new Error("マスタ取得に失敗しました");
        }

        const affiliatedData: Affiliated[] = await affiliatedResponse.json();
        const rankData: Rank[] = await rankResponse.json();
        const categoryData: Category[] = await categoryResponse.json();

        if (ignore) return;

        setAffiliateds(affiliatedData);
        setRanks(rankData);
        setCategories(categoryData);
      } catch (error) {
        console.error(error);

        if (!ignore) {
          setAffiliateds([]);
          setRanks([]);
          setCategories([]);
        }
      } finally {
        if (!ignore) {
          setIsMasterLoading(false);
        }
      }
    };

    fetchMasters();

    return () => {
      ignore = true;
    };

    // api を依存配列に入れると、useApiの作りによっては無限fetchするため入れない
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, []);

  // お悩み一覧を取得
  // requestUrl が変わった時だけ再取得する
  // reloadKey は投稿成功後に再取得するため
  useEffect(() => {
    let ignore = false;

    const fetchNeeds = async (): Promise<void> => {
      try {
        setIsLoading(true);
        setIsServerStopping(false);

        const response = await api.fetch(requestUrl);

        if (!response.ok) {
          throw new Error("お悩み一覧の取得に失敗しました");
        }

        const data: NeedPageResponse = await response.json();

        if (ignore) return;

        setNeedList(data.content ?? []);
        setTotalPage(data.totalPages ?? 0);
      } catch (error) {
        console.error(error);

        if (!ignore) {
          setNeedList([]);
          setTotalPage(0);
          setIsServerStopping(true);
        }
      } finally {
        if (!ignore) {
          setIsLoading(false);
        }
      }
    };

    fetchNeeds();

    return () => {
      ignore = true;
    };

    // api を依存配列に入れると、useApiの作りによっては無限fetchするため入れない
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [requestUrl, reloadKey]);

  const handleSortChange = (event: SelectChangeEvent): void => {
    setSortBy(event.target.value as SortBy);
    setPage(0);
  };

  const handleKeywordChange = (
    event: React.ChangeEvent<HTMLInputElement>
  ): void => {
    setSearchCondition((prev) => ({
      ...prev,
      keyword: event.target.value,
    }));
  };

  const handleSelectSearchChange =
    (field: keyof SearchCondition) =>
    (event: SelectChangeEvent): void => {
      setSearchCondition((prev) => ({
        ...prev,
        [field]: event.target.value,
      }));
    };

  const handleSearchSubmit = (): void => {
    setAppliedCondition(searchCondition);
    setPage(0);
  };

  const handleReset = (): void => {
    setSearchCondition(emptySearchCondition);
    setAppliedCondition(emptySearchCondition);
    setSortBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  const handleCreated = (): void => {
    setPage(0);
    setReloadKey((prev) => prev + 1);
  };

  return (
    <>
      <Box
        sx={{
          minHeight: "100vh",
          backgroundColor: "#fff2dd",
          px: { xs: 2, md: 4 },
          py: 3,
        }}
      >
        <Box
          sx={{
            maxWidth: 1050,
            mx: "auto",
          }}
        >
          <Box
            sx={{
              display: "grid",
              gridTemplateColumns: { xs: "1fr", md: "1fr 190px" },
              gap: 2,
              alignItems: "start",
            }}
          >
            <Box>
              <Box
                sx={{
                  position: "relative",
                  mb: 2,
                  minHeight: 44,
                }}
              >
                <Typography
                  component="h1"
                  sx={{
                    fontSize: 30,
                    fontWeight: 800,
                    color: "black",
                    textAlign: "center",
                  }}
                >
                  お悩み一覧
                </Typography>

                <FormControl
                  size="small"
                  sx={{
                    position: "absolute",
                    right: 0,
                    top: 4,
                    width: 105,
                    display: { xs: "none", md: "block" },
                  }}
                >
                  <Select
                    value={sortBy}
                    onChange={handleSortChange}
                    sx={{
                      height: 34,
                      backgroundColor: "#fff7ec",
                      borderRadius: 1,
                      fontSize: 12,
                      fontWeight: 700,
                      "& .MuiSelect-select": {
                        py: 0.5,
                      },
                    }}
                  >
                    <MenuItem value="new">新着順</MenuItem>
                    <MenuItem value="empathy">共感数順</MenuItem>
                  </Select>
                </FormControl>
              </Box>

              {isServerStopping && (
                <Alert
                  severity="error"
                  sx={{
                    mb: 2,
                    justifyContent: "center",
                  }}
                >
                  サーバーが起動していません。しばらくたってからお試しください。
                </Alert>
              )}

              {isLoading ? (
                <CircularProgress
                  aria-label="Loading"
                  sx={{
                    display: "block",
                    mx: "auto",
                    my: 8,
                  }}
                />
              ) : (
                <>
                  {needList.length === 0 ? (
                    <Box
                      sx={{
                        textAlign: "center",
                        py: 8,
                        color: "text.secondary",
                      }}
                    >
                      <Typography variant="h6" sx={{ fontWeight: 700 }}>
                        お悩みがありません
                      </Typography>
                    </Box>
                  ) : (
                    <Grid container spacing={2}>
                      {needList.map((need, index) => (
                        <Grid
                          item
                          xs={12}
                          sm={6}
                          md={4}
                          key={need.needId ?? index}
                        >
                          <NeedCard need={need} />
                        </Grid>
                      ))}
                    </Grid>
                  )}

                  {totalPage > 1 && (
                    <Stack
                      direction="row"
                      sx={{
                        mt: 3,
                        justifyContent: "center",
                        alignItems: "center",
                        gap: 1,
                        flexWrap: "wrap",
                      }}
                    >
                      <Pagination
                        count={totalPage}
                        page={page + 1}
                        onChange={handlePageChange}
                        color="primary"
                        shape="rounded"
                        size="small"
                      />
                    </Stack>
                  )}
                </>
              )}
            </Box>

            <Box>
              <Box
                sx={{
                  backgroundColor: "#e4dcc4",
                  borderRadius: 1,
                  px: 1.5,
                  py: 2,
                  mb: 2,
                }}
              >
                <Typography
                  sx={{
                    fontSize: 18,
                    fontWeight: 800,
                    textAlign: "center",
                    mb: 2,
                    color: "black",
                  }}
                >
                  絞り込み
                </Typography>

                <Stack spacing={1.2}>
                  <Box
                    sx={{
                      display: "grid",
                      gridTemplateColumns: "50px 1fr",
                      alignItems: "center",
                      gap: 1,
                    }}
                  >
                    <Typography sx={{ fontSize: 13, fontWeight: 700 }}>
                      所属
                    </Typography>

                    <FormControl fullWidth size="small">
                      <Select
                        value={searchCondition.affiliatedId}
                        onChange={handleSelectSearchChange("affiliatedId")}
                        displayEmpty
                        disabled={isMasterLoading}
                        sx={{
                          height: 28,
                          backgroundColor: "#fff7ec",
                          fontSize: 11,
                          "& .MuiSelect-select": {
                            py: 0.2,
                          },
                        }}
                      >
                        <MenuItem value="">指定なし</MenuItem>

                        {affiliateds.map((affiliated) => (
                          <MenuItem
                            key={affiliated.affiliatedId}
                            value={String(affiliated.affiliatedId)}
                          >
                            {affiliated.affiliatedName}
                          </MenuItem>
                        ))}
                      </Select>
                    </FormControl>
                  </Box>

                  <Box
                    sx={{
                      display: "grid",
                      gridTemplateColumns: "50px 1fr",
                      alignItems: "center",
                      gap: 1,
                    }}
                  >
                    <Typography sx={{ fontSize: 13, fontWeight: 700 }}>
                      職位
                    </Typography>

                    <FormControl fullWidth size="small">
                      <Select
                        value={searchCondition.positionId}
                        onChange={handleSelectSearchChange("positionId")}
                        displayEmpty
                        disabled={isMasterLoading}
                        sx={{
                          height: 28,
                          backgroundColor: "#fff7ec",
                          fontSize: 11,
                          "& .MuiSelect-select": {
                            py: 0.2,
                          },
                        }}
                      >
                        <MenuItem value="">指定なし</MenuItem>

                        {ranks.map((rank) => (
                          <MenuItem
                            key={rank.rankId}
                            value={String(rank.rankId)}
                          >
                            {rank.rankName}
                          </MenuItem>
                        ))}
                      </Select>
                    </FormControl>
                  </Box>

                  <Box
                    sx={{
                      display: "grid",
                      gridTemplateColumns: "50px 1fr",
                      alignItems: "center",
                      gap: 1,
                    }}
                  >
                    <Typography
                      sx={{
                        fontSize: 13,
                        fontWeight: 700,
                        lineHeight: 1.2,
                      }}
                    >
                      悩み
                      <br />
                      カテゴリ
                    </Typography>

                    <FormControl fullWidth size="small">
                      <Select
                        value={searchCondition.categoryId}
                        onChange={handleSelectSearchChange("categoryId")}
                        displayEmpty
                        disabled={isMasterLoading}
                        sx={{
                          height: 28,
                          backgroundColor: "#fff7ec",
                          fontSize: 11,
                          "& .MuiSelect-select": {
                            py: 0.2,
                          },
                        }}
                      >
                        <MenuItem value="">指定なし</MenuItem>

                        {categories.map((category) => (
                          <MenuItem
                            key={category.categoryId}
                            value={String(category.categoryId)}
                          >
                            {category.categoryName}
                          </MenuItem>
                        ))}
                      </Select>
                    </FormControl>
                  </Box>

                  <Typography
                    sx={{
                      fontSize: 16,
                      fontWeight: 800,
                      textAlign: "center",
                      pt: 1,
                      color: "black",
                    }}
                  >
                    検索
                  </Typography>

                  <TextField
                    size="small"
                    fullWidth
                    placeholder="人名/キーワード/投稿タイトル"
                    value={searchCondition.keyword}
                    onChange={handleKeywordChange}
                    sx={{
                      backgroundColor: "#fff7ec",
                      "& .MuiInputBase-input": {
                        height: 20,
                        fontSize: 11,
                        py: 0.5,
                      },
                    }}
                  />

                  <Box
                    sx={{
                      display: "flex",
                      justifyContent: "center",
                      gap: 1,
                      pt: 1,
                    }}
                  >
                    <Button
                      variant="contained"
                      onClick={handleReset}
                      sx={{
                        minWidth: 58,
                        height: 28,
                        backgroundColor: "#d8ccb0",
                        color: "black",
                        fontSize: 11,
                        fontWeight: 700,
                        boxShadow: "none",
                        "&:hover": {
                          backgroundColor: "#c9b990",
                          boxShadow: "none",
                        },
                      }}
                    >
                      リセット
                    </Button>

                    <Button
                      variant="contained"
                      onClick={handleSearchSubmit}
                      sx={{
                        minWidth: 58,
                        height: 28,
                        backgroundColor: "#005b96",
                        color: "white",
                        fontSize: 11,
                        fontWeight: 700,
                        boxShadow: "none",
                        "&:hover": {
                          backgroundColor: "#004a7c",
                          boxShadow: "none",
                        },
                      }}
                    >
                      実行
                    </Button>
                  </Box>
                </Stack>
              </Box>

              <Button
                fullWidth
                variant="contained"
                onClick={() => setPostOpen(true)}
                disabled={isServerStopping}
                sx={{
                  height: 62,
                  borderRadius: 8,
                  backgroundColor: "#005b96",
                  color: "white",
                  fontSize: 19,
                  fontWeight: 800,
                  lineHeight: 1.2,
                  boxShadow: "none",
                  "&:hover": {
                    backgroundColor: "#004a7c",
                    boxShadow: "none",
                  },
                }}
              >
                お悩み
                <br />
                投稿
              </Button>
            </Box>
          </Box>
        </Box>
      </Box>

      <Dialog
        open={postOpen}
        onClose={() => setPostOpen(false)}
        fullWidth
        maxWidth="sm"
        slotProps={{
          paper: {
            sx: {
              borderRadius: 4,
              width: 410,
              maxWidth: "calc(100% - 32px)",
              backgroundColor: "#fff2dd",
            },
          },
        }}
      >
        <DialogContent
          sx={{
            px: 5,
            py: 4,
            backgroundColor: "#fff2dd",
          }}
        >
          <SubmitNeeds
            categories={categories}
            onClose={() => setPostOpen(false)}
            onCreated={handleCreated}
          />
        </DialogContent>
      </Dialog>
    </>
  );
}
