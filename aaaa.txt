private void addCompanyYearsCondition(SearchRequest request) {
    if (request.getCompanyName() == null
            || request.getCompanyName().isBlank()
            || request.getMinYears() == null) {
        return;
    }

    sql.append("""
            AND EXISTS (
                SELECT 1
                FROM career c2
                JOIN project pr2 ON pr2.project_id = c2.project_id
                JOIN division d2 ON d2.division_id = pr2.division_id
                JOIN company co2 ON co2.company_id = d2.company_id
                WHERE c2.profile_id = p.hrid
                  AND co2.company_name = ?
                GROUP BY co2.company_name
                HAVING COALESCE(SUM(
                    DATE_PART(
                        'year',
                        AGE(
                            COALESCE(c2.end_time, CURRENT_DATE),
                            c2.start_time
                        )
                    ) * 12
                    +
                    DATE_PART(
                        'month',
                        AGE(
                            COALESCE(c2.end_time, CURRENT_DATE),
                            c2.start_time
                        )
                    )
                ), 0) >= ?
            )
            """);

    params.add(request.getCompanyName().trim());
    params.add(request.getMinYears() * 12);
}
