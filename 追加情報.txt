##SearchRepository.java の考え方

public List<ProfileSearchResponse> searchProfiles(SearchRequest request) {

    SearchSqlBuilder builder = new SearchSqlBuilder();

    builder.applyConditions(request);
    builder.applyGroupByAndHaving(request);

    List<ProfileSearchResponse> profiles = jdbcTemplate.query(
            builder.getSql(),
            (rs, rowNum) -> new ProfileSearchResponse(
                    rs.getString("hrid"),
                    rs.getString("name"),
                    rs.getString("mail_address"),
                    rs.getString("rank_name"),
                    rs.getString("company_name")
            ),
            builder.getParams()
    );

    for (ProfileSearchResponse profile : profiles) {
        List<SkillExperienceResponse> skills =
                findSkillExperiencesByHrid(profile.getHrid());

        profile.setSkillExperiences(skills);
    }

    return profiles;
}

##追加する private メソッド

private List<SkillExperienceResponse> findSkillExperiencesByHrid(String hrid) {

    String sql = """
            SELECT
                s.skill_name,
                COALESCE(SUM(
                    COALESCE(c.end_time, NOW()::date) - c.start_time
                ), 0) AS total_days
            FROM profile p
            JOIN career c ON c.profile_id = p.hrid
            JOIN career_skill cs ON cs.career_id = c.career_id
            JOIN skill s ON s.skill_id = cs.skill_id
            WHERE p.hrid = ?
            GROUP BY s.skill_name
            ORDER BY s.skill_name
            """;

    return jdbcTemplate.query(sql, (rs, rowNum) -> {
        int totalDays = rs.getInt("total_days");

        int years = totalDays / 365;
        int months = (totalDays % 365) / 30;

        return new SkillExperienceResponse(
                rs.getString("skill_name"),
                years,
                months
        );
    }, hrid);
}

private void addSkillCondition(SearchRequest request) {
    if (request.getSkillNames() == null) {
        return;
    }

    List<String> skillNames = request.getSkillNames().stream()
            .filter(skillName -> skillName != null)
            .map(skillName -> skillName.replace("　", "").trim())
            .filter(skillName -> !skillName.isEmpty())
            .distinct()
            .toList();

    for (String skillName : skillNames) {
        sql.append("""
                AND EXISTS (
                    SELECT 1
                    FROM career c2
                    JOIN career_skill cs2 ON cs2.career_id = c2.career_id
                    JOIN skill s2 ON s2.skill_id = cs2.skill_id
                    WHERE c2.profile_id = p.hrid
                      AND s2.skill_name = ?
                )
                """);

        params.add(skillName);
    }
}
