
MyPage.tsx を大きく作り替えずに、今の構成を守るなら「stateを少し足す」「fetch URLにpageを足す」「表示下にPaginationを足す」だけでいい。

今のコードはすでに、

setMyContents(data.content ?? data);
setLikedContents(data.content ?? data);

みたいに、Page 形式にも普通の配列にも一応対応しようとしている。なので、ここを活かす。

import { useState, useEffect } from "react";
import {
  Container,
  Box,
  Avatar,
  Typography,
  Tabs,
  Tab,
  Button,
  TextField,
  CircularProgress,
  Pagination,
} from "@mui/material";
import EditIcon from "@mui/icons-material/Edit";
import ContentCard from "../components/ContentCard";
import type { Content } from "../models/content";
import type { User } from "../models/user";
import { useApi } from "../lib/useApi.ts";
import type { Category } from "../models/category.ts";
import { Link } from "react-router-dom";

type PageResponse<T> = {
  content: T[];
  totalPages: number;
  totalElements: number;
  number: number;
  size: number;
  first: boolean;
  last: boolean;
};

function MyBestSeries(
  myBestMap: Map<string, Content>,
  category: Category
): JSX.Element | undefined {
  const myBestContent = myBestMap.get(category.categoryName);

  if (myBestContent === undefined) {
    console.log("myBestContent");
    console.log(myBestContent);
    return undefined;
  }

  return (
    <ContentCard
      key={category.categoryId}
      content={myBestContent}
    />
  );
}

function MyPage(): JSX.Element {
  const [myContents, setMyContents] = useState<Content[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);
  const [likedContents, setLikedContents] = useState<Content[]>([]);
  const [profile, setProfile] = useState<User>();
  const [editProfile, setEditProfile] = useState<User>();
  const [isEditing, setIsEditing] = useState(false);
  const [selectedTab, setSelectedTab] = useState("posts");
  const [isLoading, setIsLoading] = useState<boolean>(true);

  const [myContentsPage, setMyContentsPage] = useState(0);
  const [myContentsTotalPage, setMyContentsTotalPage] = useState(0);

  const [likedContentsPage, setLikedContentsPage] = useState(0);
  const [likedContentsTotalPage, setLikedContentsTotalPage] = useState(0);

  const [mapMyBest, setMapMyBest] = useState<Map<string, Content>>(() => {
    return new Map<string, Content>();
  });

  const api = useApi();

  useEffect((): void => {
    api
      .fetch("/api/me/profile")
      .then((res: Response): Promise<User> => res.json())
      .then((data: User): Promise<Response> => {
        setProfile(data);
        setEditProfile(data);
        return api.fetch(`/api/users/${data.userId}/contents?page=0&size=20`);
      })
      .then((res: Response): Promise<PageResponse<Content> | Content[]> => res.json())
      .then((data: PageResponse<Content> | Content[]): void => {
        if (Array.isArray(data)) {
          setMyContents(data);
          setMyContentsTotalPage(0);
        } else {
          setMyContents(data.content);
          setMyContentsTotalPage(data.totalPages);
        }

        setMyContentsPage(0);
        setIsLoading(false);
        console.log(data);
      })
      .catch((error: Error): void => {
        console.error(error);
        setIsLoading(false);
      });
  }, []);

  const handleChange = (
    event: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>
  ): void => {
    if (!editProfile) {
      return;
    }

    const { name, value } = event.target;
    setEditProfile({
      ...editProfile,
      [name]: value,
    });
  };

  const handleMenuChange = (selectedTabString: string): void => {
    setSelectedTab(selectedTabString);
    setIsLoading(true);

    switch (selectedTabString) {
      case "posts":
        setMyContentsPage(0);

        api
          .fetch(`/api/users/${profile?.userId}/contents?page=0&size=20`)
          .then((res: Response): Promise<PageResponse<Content> | Content[]> =>
            res.json()
          )
          .then((data: PageResponse<Content> | Content[]): void => {
            if (Array.isArray(data)) {
              setMyContents(data);
              setMyContentsTotalPage(0);
            } else {
              setMyContents(data.content);
              setMyContentsTotalPage(data.totalPages);
            }

            setIsLoading(false);
          })
          .catch((error: Error): void => {
            console.error(error);
            setIsLoading(false);
          });
        break;

      case "liked":
        setLikedContentsPage(0);

        api
          .fetch(`/api/users/${profile?.userId}/likedContents?page=0&size=20`)
          .then((res: Response): Promise<PageResponse<Content> | Content[]> =>
            res.json()
          )
          .then((data: PageResponse<Content> | Content[]): void => {
            if (Array.isArray(data)) {
              setLikedContents(data);
              setLikedContentsTotalPage(0);
            } else {
              setLikedContents(data.content);
              setLikedContentsTotalPage(data.totalPages);
            }

            setIsLoading(false);
          })
          .catch((error: Error): void => {
            console.error(error);
            setIsLoading(false);
          });
        break;

      case "mybest":
        Promise.all([
          api.fetch(`/api/myBest/${profile?.userId}`),
          api.fetch("/api/categories"),
        ])
          .then(([myBestRes, categoryRes]: [Response, Response]) =>
            Promise.all([myBestRes.json(), categoryRes.json()])
          )
          .then(([data1, data2]: [Record<string, Content>, Category[]]): void => {
            const myBestData = data1;
            const categoryData = data2;

            setMapMyBest(new Map<string, Content>(Object.entries(myBestData)));
            setCategories(categoryData);
            setIsLoading(false);
          })
          .catch((error: Error): void => {
            console.error(error);
            setIsLoading(false);
          });
        break;

      default:
        setIsLoading(false);
        break;
    }
  };

  const handleMyContentsPageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    const nextPage = value - 1;

    setMyContentsPage(nextPage);
    setIsLoading(true);

    api
      .fetch(`/api/users/${profile?.userId}/contents?page=${nextPage}&size=20`)
      .then((res: Response): Promise<PageResponse<Content> | Content[]> =>
        res.json()
      )
      .then((data: PageResponse<Content> | Content[]): void => {
        if (Array.isArray(data)) {
          setMyContents(data);
          setMyContentsTotalPage(0);
        } else {
          setMyContents(data.content);
          setMyContentsTotalPage(data.totalPages);
        }

        setIsLoading(false);
      })
      .catch((error: Error): void => {
        console.error(error);
        setIsLoading(false);
      });
  };

  const handleLikedContentsPageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    const nextPage = value - 1;

    setLikedContentsPage(nextPage);
    setIsLoading(true);

    api
      .fetch(
        `/api/users/${profile?.userId}/likedContents?page=${nextPage}&size=20`
      )
      .then((res: Response): Promise<PageResponse<Content> | Content[]> =>
        res.json()
      )
      .then((data: PageResponse<Content> | Content[]): void => {
        if (Array.isArray(data)) {
          setLikedContents(data);
          setLikedContentsTotalPage(0);
        } else {
          setLikedContents(data.content);
          setLikedContentsTotalPage(data.totalPages);
        }

        setIsLoading(false);
      })
      .catch((error: Error): void => {
        console.error(error);
        setIsLoading(false);
      });
  };

  const handleEdit = (): void => {
    setEditProfile(profile);
    setIsEditing(true);
  };

  const handleCancel = (): void => {
    setEditProfile(profile);
    setIsEditing(false);
  };

  const handleSave = (): void => {
    if (!editProfile?.name) {
      alert("名前を入力してください。");
      return;
    }

    api
      .sendJson(`/api/users/${profile?.userId}`, editProfile, {
        method: "POST",
        headers: { "Content-Type": "application/json" },
      })
      .then((res: Response): Promise<User> => res.json())
      .then((data: User): void => {
        setProfile(data);
        setEditProfile(data);
        setIsEditing(false);
      })
      .catch((error: Error): void => {
        console.error(error);
      });
  };

  return (
    <Container maxWidth="md" sx={{ mt: 4 }}>
      <Box sx={{ position: "relative", mb: 6 }}>
        <Box
          sx={{
            height: 140,
            borderRadius: 3,
            background: "linear-gradient(135deg, #60A5FA, #A78BFA)",
          }}
        />

        <Avatar
          sx={{
            width: 90,
            height: 90,
            position: "absolute",
            bottom: -45,
            left: 24,
            border: "4px solid white",
            fontSize: 32,
          }}
        >
          {profile?.name?.[0]}
        </Avatar>
      </Box>

      {!isEditing && (
        <Button
          startIcon={<EditIcon />}
          sx={{
            position: "relative",
            color: "grey",
            display: "flex",
            justifyContent: "flex-end",
          }}
          onClick={handleEdit}
        >
          編集
        </Button>
      )}

      <Box sx={{ px: 2, mb: 4 }}>
        {isEditing ? (
          <>
            <Typography
              variant="h6"
              sx={{
                fontWeight: "bold",
                textAlign: "left",
              }}
            >
              名前
            </Typography>

            <TextField
              label="名前"
              name="name"
              slotProps={{
                htmlInput: {
                  maxLength: 10,
                },
              }}
              helperText="10文字まで"
              value={editProfile?.name ?? ""}
              onChange={handleChange}
              fullWidth
              sx={{
                mb: 2,
                textAlign: "left",
                overflowWrap: "anywhere",
                wordBreak: "break-word",
              }}
            />

            <TextField
              label="部署"
              name="department"
              slotProps={{
                htmlInput: {
                  maxLength: 10,
                },
              }}
              helperText="10文字まで"
              value={editProfile?.department ?? ""}
              onChange={handleChange}
              fullWidth
              sx={{
                mb: 2,
                textAlign: "left",
                overflowWrap: "anywhere",
                wordBreak: "break-word",
              }}
            />

            <TextField
              label="自己紹介"
              name="description"
              slotProps={{
                htmlInput: {
                  maxLength: 100,
                },
              }}
              helperText="100文字まで"
              value={editProfile?.description ?? ""}
              onChange={handleChange}
              multiline
              rows={5}
              fullWidth
              sx={{
                textAlign: "left",
                overflowWrap: "anywhere",
                wordBreak: "break-word",
              }}
            />

            <Box sx={{ mt: 2, display: "flex", gap: 2 }}>
              <Button variant="contained" onClick={handleSave}>
                保存
              </Button>
              <Button variant="text" onClick={handleCancel}>
                キャンセル
              </Button>
            </Box>
          </>
        ) : (
          <>
            <Typography
              variant="h6"
              sx={{
                fontWeight: "bold",
                textAlign: "left",
              }}
            >
              名前：{profile?.name}
            </Typography>

            <Typography color="text.secondary" sx={{ textAlign: "left" }}>
              部署：{profile?.department}
            </Typography>

            <Typography
              sx={{
                mt: 1,
                whiteSpace: "pre-wrap",
                textAlign: "left",
              }}
            >
              自己紹介：{profile?.description}
            </Typography>
          </>
        )}
      </Box>

      <Tabs
        value={selectedTab}
        onChange={(_event, value: string): void => handleMenuChange(value)}
        variant="fullWidth"
        sx={{
          borderBottom: 1,
          borderColor: "divider",
          mb: 2,
        }}
      >
        <Tab label="投稿" value="posts" />
        <Tab label="いいね" value="liked" />
        <Tab label="マイベスト" value="mybest" />
      </Tabs>

      {selectedTab === "posts" && (
        <>
          {isLoading ? (
            <CircularProgress aria-label="Loading..." />
          ) : myContents.length === 0 ? (
            <Typography align="center" color="text.secondary">
              投稿はまだありません
            </Typography>
          ) : (
            <>
              <Box
                sx={{
                  display: "grid",
                  gridTemplateColumns: "repeat(2, 450px)",
                  columnGap: 4,
                  rowGap: 6,
                  justifyContent: "center",
                  pb: 6,
                }}
              >
                {myContents.map((content: Content): JSX.Element => (
                  <ContentCard
                    key={content.contentsId}
                    content={content}
                  />
                ))}
              </Box>

              {myContentsTotalPage > 1 && (
                <Box
                  sx={{
                    display: "flex",
                    justifyContent: "center",
                    mt: 2,
                    mb: 4,
                  }}
                >
                  <Pagination
                    count={myContentsTotalPage}
                    page={myContentsPage + 1}
                    onChange={handleMyContentsPageChange}
                    color="primary"
                    shape="rounded"
                  />
                </Box>
              )}
            </>
          )}
        </>
      )}

      {selectedTab === "liked" && (
        <>
          {isLoading ? (
            <CircularProgress aria-label="Loading..." />
          ) : likedContents.length === 0 ? (
            <Typography align="center" color="text.secondary">
              投稿はまだありません
            </Typography>
          ) : (
            <>
              <Box
                sx={{
                  display: "grid",
                  gridTemplateColumns: "repeat(2, 450px)",
                  columnGap: 4,
                  rowGap: 6,
                  justifyContent: "center",
                  pb: 6,
                }}
              >
                {likedContents.map((content: Content): JSX.Element => (
                  <ContentCard
                    key={content.contentsId}
                    content={content}
                  />
                ))}
              </Box>

              {likedContentsTotalPage > 1 && (
                <Box
                  sx={{
                    display: "flex",
                    justifyContent: "center",
                    mt: 2,
                    mb: 4,
                  }}
                >
                  <Pagination
                    count={likedContentsTotalPage}
                    page={likedContentsPage + 1}
                    onChange={handleLikedContentsPageChange}
                    color="primary"
                    shape="rounded"
                  />
                </Box>
              )}
            </>
          )}
        </>
      )}

      {selectedTab === "mybest" && (
        <>
          {isLoading ? (
            <CircularProgress aria-label="Loading..." />
          ) : (
            <Box
              sx={{
                display: "grid",
                alignItems: "start",
                gridTemplateColumns: "repeat(2, 450px)",
                justifyContent: "center",
                columnGap: 8,
                rowGap: 30,
                pb: 8,
              }}
            >
              {categories.map((category: Category): JSX.Element => (
                <Box key={category.categoryId}>
                  <Typography
                    variant="h6"
                    sx={{
                      fontWeight: 500,
                      color: "black",
                      mb: 2,
                      pb: 1.5,
                      borderBottom: "2px solid #dbeafe",
                      letterSpacing: "0.03em",
                    }}
                  >
                    {category.categoryName}のベスト
                  </Typography>

                  {mapMyBest.has(category.categoryName) ? (
                    MyBestSeries(mapMyBest, category)
                  ) : (
                    <Box
                      component={Link}
                      to="/home"
                      sx={{
                        width: "100%",
                        height: "100%",
                        boxSizing: "border-box",
                        display: "flex",
                        flexDirection: "column",
                        justifyContent: "center",
                        alignItems: "center",
                        borderRadius: 10,
                        backgroundColor: "#f0f9ff",
                        border: "1px dashed #93c5fd",
                        textDecoration: "none",
                      }}
                    >
                      <Typography
                        sx={{
                          color: "#2563eb",
                          fontWeight: 700,
                          fontSize: 15,
                        }}
                      >
                        未選択しょう！
                      </Typography>
                    </Box>
                  )}
                </Box>
              ))}
            </Box>
          )}
        </>
      )}
    </Container>
  );
}

export default MyPage;
