1. TipService.java 完全版

写真では createTip() の中で、input.setCategoryId(1) や input.setTipTitle("title") のように、フロントから来た値をサーバー側で上書きしている状態だった。ここが投稿失敗の大きな原因候補。

package simplex.pj148.skippy.server.service;

import jakarta.persistence.EntityNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.TipsCreateDto;
import simplex.pj148.skippy.server.entity.Category;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CategoryRepository;
import simplex.pj148.skippy.server.repository.TipRepository;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class TipService {

    private final TipRepository tipRepository;
    private final CategoryRepository categoryRepository;

    @Transactional
    public void createTip(TipsCreateDto input, User loginUser) {

        /*
         * 修正前はここで input の値を固定値に上書きしていた。
         *
         * input.setCategoryId(1);
         * input.setTipTitle("title");
         * input.setTipDetail("detail");
         * input.setTipURL("url");
         *
         * これをやると、フロントから送った投稿内容が全部消える。
         * なので削除。
         */

        // 修正：フロントから送られてきた categoryId をそのまま使う
        Category category = categoryRepository.findByCategoryId(input.getCategoryId())
                .orElseThrow(() -> new EntityNotFoundException(
                        "カテゴリDBに存在しないカテゴリ：" + input.getCategoryId() + " が選択されています。"
                ));

        LocalDateTime createdTime = LocalDateTime.now();

        // 修正：input の値をそのまま Tip Entity に詰める
        Tip tip = Tip.builder()
                .user(loginUser)
                .category(category)
                .tipTitle(input.getTipTitle())
                .tipDetail(input.getTipDetail())
                .tipURL(input.getTipURL())
                .createdTime(createdTime)
                .deleteFlg(false)
                .build();

        tipRepository.save(tip);
    }
}

5. TipForm.tsx 側で最終的に送る値

categoryId はフォームだと文字列になりやすい。
だから投稿時に Number() で数値に変換してから渡す。

type TipFormValues = {
  categoryId: string | number;
  tipTitle: string;
  tipDetail: string;
  tipURL: string;
};

type TipFormProps = {
  initialValues: TipFormValues;
  onSubmit: (values: {
    categoryId: number;
    tipTitle: string;
    tipDetail: string;
    tipURL: string;
  }) => void;
  submitLabel: string;
  onClose: () => void;
};

function TipForm({
  initialValues,
  onSubmit,
  submitLabel,
  onClose,
}: TipFormProps) {
  const [formData, setFormData] = useState<TipFormValues>(initialValues);

  const handleChange = (
    event: React.ChangeEvent<HTMLInputElement | HTMLTextAreaElement>
  ): void => {
    const { name, value } = event.target;

    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  const handleSubmit = (event: React.FormEvent<HTMLFormElement>): void => {
    event.preventDefault();

    // 修正：categoryId を number に変換してから親へ渡す
    onSubmit({
      categoryId: Number(formData.categoryId),
      tipTitle: formData.tipTitle,
      tipDetail: formData.tipDetail,
      tipURL: formData.tipURL,
    });
  };

  return (
    <form onSubmit={handleSubmit}>
      {/* ここに今ある TextField たちをそのまま置く */}

      <button type="submit">
        {submitLabel}
      </button>

      <button type="button" onClick={onClose}>
        キャンセル
      </button>
    </form>
  );
}

export default TipForm;
