package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.create.NeedEmpathyCreateDto;
import simplex.pj148.skippy.server.entity.Need;
import simplex.pj148.skippy.server.entity.NeedEmpathy;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.NeedEmpathyRepository;
import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

@Service
@RequiredArgsConstructor
public class NeedEmpathyService {

    private final NeedEmpathyRepository needEmpathyRepository;
    private final NeedRepository needRepository;
    private final UserRepository userRepository;

    @Transactional
    public void registerNewNeedEmpathy(NeedEmpathyCreateDto input) {
        Need need = needRepository.findByNeedIdAndDeleteFlgFalse(input.getNeedId())
                .orElseThrow(() -> new RuntimeException("ニーズDBに存在しないニーズ"));

        User user = userRepository.findByUserId(input.getUserId())
                .orElseThrow(() -> new RuntimeException("ユーザーDBに存在しないユーザー"));

        boolean alreadyEmpathized =
                needEmpathyRepository.existsByNeed_NeedIdAndUser_UserId(
                        input.getNeedId(),
                        input.getUserId()
                );

        if (alreadyEmpathized) {
            return;
        }

        NeedEmpathy needEmpathy = NeedEmpathy.builder()
                .need(need)
                .user(user)
                .isLooked(false)
                .build();

        needEmpathyRepository.save(needEmpathy);
    }

    @Transactional
    public void deleteNeedEmpathy(NeedEmpathyCreateDto input) {
        NeedEmpathy needEmpathy =
                needEmpathyRepository.findByUser_UserIdAndNeed_NeedId(
                                input.getUserId(),
                                input.getNeedId()
                        )
                        .orElseThrow(() -> new RuntimeException("共感データが見つかりません"));

        needEmpathyRepository.delete(needEmpathy);
    }
}

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.create.CommentDto;
import simplex.pj148.skippy.server.controller.dto.response.CommentShowResponseDto;
import simplex.pj148.skippy.server.entity.Comment;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CommentRepository;
import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
public class CommentService {

    private final CommentRepository commentRepository;
    private final UserRepository userRepository;
    private final TipRepository tipRepository;

    public List<CommentShowResponseDto> findComments(Long tipId, User loginUser) {
        List<Comment> comments = commentRepository.findByTip_TipIdAndDeleteFlgFalse(tipId);

        return comments.stream()
                .map(comment -> {
                    boolean editFlg = comment.getUser()
                            .getUserId()
                            .equals(loginUser.getUserId());

                    return CommentShowResponseDto.builder()
                            .commentId(comment.getCommentId())
                            .tipId(comment.getTip().getTipId())
                            .userId(comment.getUser().getUserId())
                            .userName(comment.getUser().getUserName())
                            .commentBody(comment.getCommentBody())
                            .editFlg(editFlg)
                            .build();
                })
                .toList();
    }

    @Transactional
    public void createNewComment(CommentDto input, Long tipId) {
        User inputUser = userRepository.findByUserId(input.getUserId())
                .orElseThrow(() -> new RuntimeException("ユーザーDBに存在しないユーザー：" + input.getUserId()));

        Tip inputTip = tipRepository.findByTipIdAndDeleteFlgFalse(tipId)
                .orElseThrow(() -> new RuntimeException("TipsDBに存在しないTips：" + tipId));

        Comment comment = Comment.builder()
                .user(inputUser)
                .tip(inputTip)
                .commentBody(input.getCommentBody())
                .isLooked(false)
                .deleteFlg(false)
                .build();

        commentRepository.save(comment);
    }

    @Transactional
    public void deleteComment(Long commentId, User loginUser) {
        Comment comment = commentRepository.findByCommentIdAndDeleteFlgFalse(commentId)
                .orElseThrow(() -> new RuntimeException("コメントが見つかりません"));

        if (!comment.getUser().getUserId().equals(loginUser.getUserId())) {
            throw new RuntimeException("自分のコメント以外は削除できません");
        }

        comment.setDeleteFlg(true);
        commentRepository.save(comment);
    }
}

    private final NeedEmpathyRepository needEmpathyRepository;

            User loginUser,

            String orderBy
    ) {

import simplex.pj148.skippy.server.service.UserInfoService;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/need-empathy")
public class NeedEmpathyController {

    private final NeedEmpathyService needEmpathyService;
    private final UserInfoService userInfoService;

    @PostMapping
    public ResponseEntity<Void> registerNewNeedEmpathy(
            @RequestBody NeedEmpathyCreateDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        input.setUserId(loginUser.getUserId());

        needEmpathyService.registerNewNeedEmpathy(input);

        return ResponseEntity.ok().build();
    }

    @PutMapping("/delete")
    public ResponseEntity<Void> deleteNeedEmpathy(
            @RequestBody NeedEmpathyCreateDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        input.setUserId(loginUser.getUserId());

        needEmpathyService.deleteNeedEmpathy(input);

        return ResponseEntity.ok().build();
    }
}

CommentController.java
package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.*;
import simplex.pj148.skippy.server.controller.dto.create.CommentDto;
import simplex.pj148.skippy.server.controller.dto.response.CommentShowResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.CommentService;
import simplex.pj148.skippy.server.service.UserInfoService;

import java.util.List;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/tips/{tipId}/comments")
public class CommentController {

    private final CommentService commentService;
    private final UserInfoService userInfoService;

    @GetMapping
    public List<CommentShowResponseDto> findComments(
            @PathVariable Long tipId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return commentService.findComments(tipId, loginUser);
    }

    @PostMapping
    public ResponseEntity<Void> createComment(
            @PathVariable Long tipId,
            @RequestBody CommentDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        input.setUserId(loginUser.getUserId());

        commentService.createNewComment(input, tipId);

        return ResponseEntity.ok().build();
    }

    @PutMapping("/{commentId}/delete")
    public ResponseEntity<Void> deleteComment(
            @PathVariable("commentId") Long commentId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        commentService.deleteComment(commentId, loginUser);

        return ResponseEntity.ok().build();
    }
}

重複防止のユニーク制約も、DBに反映するならこれ。

ALTER TABLE tip_interest
ADD CONSTRAINT uk_tip_interest_user_tip UNIQUE (user_id, tip_id);
