import { useEffect, useState } from "react";
import {
  Alert,
  Box,
  Button,
  CircularProgress,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";

import NeedCard from "../components/NeedCard";
import SubmitForm from "./SubmitForm";
import type { Need, NeedPageResponse } from "../models/Need";
import { useApi } from "../lib/api";

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);
  const [orderBy, setOrderBy] = useState<string>("new");
  const [keyword, setKeyword] = useState<string>("");

  const [affiliatedId, setAffiliatedId] = useState<string>("");
  const [positionId, setPositionId] = useState<string>("");
  const [categoryId, setCategoryId] = useState<string>("");

  const [page, setPage] = useState<number>(0);
  const [totalPage, setTotalPage] = useState<number>(0);

  const [isLoading, setIsLoading] = useState<boolean>(true);
  const [isServerStopping, setIsServerStopping] = useState<boolean>(false);
  const [postOpen, setPostOpen] = useState<boolean>(false);

  const fetchNeeds = (): void => {
    setIsLoading(true);
    setIsServerStopping(false);

    const params = new URLSearchParams();

    params.append("page", String(page));
    params.append("size", "18");
    params.append("orderBy", orderBy);

    if (keyword.trim() !== "") {
      params.append("keyword", keyword.trim());
    }

    if (affiliatedId !== "") {
      params.append("affiliatedId", affiliatedId);
    }

    if (positionId !== "") {
      params.append("positionId", positionId);
    }

    if (categoryId !== "") {
      params.append("categoryId", categoryId);
    }

    api
      .fetch(`/api/needs/search?${params.toString()}`)
      .then((res: Response): Promise<NeedPageResponse> => {
        if (!res.ok) {
          throw new Error("Error fetching needs");
        }
        return res.json();
      })
      .then((data: NeedPageResponse): void => {
        setNeedList(data.content ?? []);
        setTotalPage(data.totalPages ?? 0);
        setIsLoading(false);
      })
      .catch((err: Error): void => {
        console.error(err.message);
        setNeedList([]);
        setTotalPage(0);
        setIsLoading(false);
        setIsServerStopping(true);
      });
  };

  useEffect((): void => {
    fetchNeeds();
  }, [page, orderBy]);

  const handleSearch = (): void => {
    setPage(0);
    fetchNeeds();
  };

  const handleReset = (): void => {
    setKeyword("");
    setAffiliatedId("");
    setPositionId("");
    setCategoryId("");
    setOrderBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  return (
    <>
      <Box
        sx={{
          maxWidth: 1280,
          mx: "auto",
          px: 4,
          py: 4,
          color: "#071a33",
        }}
      >
        <Box
          sx={{
            display: "grid",
            gridTemplateColumns: {
              xs: "1fr",
              md: "1fr 280px",
            },
            columnGap: 3,
            rowGap: 3,
            alignItems: "start",
          }}
        >
          <Box>
            <Box
              sx={{
                position: "relative",
                mb: 3,
              }}
            >
              <Typography
                component="h1"
                sx={{
                  textAlign: "center",
                  fontSize: {
                    xs: 32,
                    md: 42,
                  },
                  fontWeight: 900,
                  letterSpacing: "0.02em",
                  color: "#071a33",
                }}
              >
                お悩み一覧
              </Typography>

              <FormControl
                size="small"
                sx={{
                  position: {
                    xs: "static",
                    md: "absolute",
                  },
                  right: 0,
                  top: 8,
                  minWidth: 140,
                  display: "flex",
                  alignItems: {
                    xs: "flex-end",
                    md: "stretch",
                  },
                  mt: {
                    xs: 2,
                    md: 0,
                  },
                }}
              >
                <Select
                  value={orderBy}
                  onChange={(event) => {
                    setOrderBy(event.target.value);
                    setPage(0);
                  }}
                  sx={{
                    height: 48,
                    borderRadius: 2,
                    backgroundColor: "#fff",
                    fontWeight: 800,
                    fontSize: 16,
                    color: "#071a33",
                    boxShadow: "0 1px 4px rgba(0,0,0,0.06)",
                  }}
                >
                  <MenuItem value="new">新着順</MenuItem>
                  <MenuItem value="empathy">共感数順</MenuItem>
                </Select>
              </FormControl>
            </Box>

            {isServerStopping ? (
              <Alert
                severity="error"
                sx={{
                  mb: 4,
                  justifyContent: "center",
                  maxWidth: 700,
                  mx: "auto",
                }}
              >
                サーバーが起動していません。しばらくたってからお試しください。
              </Alert>
            ) : isLoading ? (
              <CircularProgress
                aria-label="Loading"
                sx={{
                  display: "block",
                  mx: "auto",
                  my: 8,
                }}
              />
            ) : (
              <>
                <Grid container spacing={3}>
                  {needList.map((need: Need, index: number) => (
                    <Grid
                      size={{
                        xs: 12,
                        sm: 6,
                        md: 4,
                      }}
                      key={need.needId ?? index}
                    >
                      <NeedCard need={need} />
                    </Grid>
                  ))}
                </Grid>

                {totalPage > 1 && (
                  <Stack
                    direction="row"
                    sx={{
                      pt: 4,
                      justifyContent: "center",
                      alignItems: "center",
                    }}
                  >
                    <Pagination
                      count={totalPage}
                      page={page + 1}
                      onChange={handlePageChange}
                      color="primary"
                      shape="rounded"
                      sx={{
                        "& .MuiPaginationItem-root": {
                          fontWeight: 700,
                          borderRadius: 2,
                        },
                        "& .Mui-selected": {
                          backgroundColor: "#0054a6",
                          color: "#fff",
                          "&:hover": {
                            backgroundColor: "#004080",
                          },
                        },
                      }}
                    />
                  </Stack>
                )}
              </>
            )}
          </Box>

          <Box
            sx={{
              mt: {
                xs: 0,
                md: 7,
              },
              width: {
                xs: "100%",
                md: 280,
              },
              borderRadius: 3,
              p: 3,
              backgroundColor: "#f4efe6",
              boxShadow: "0 8px 24px rgba(0,0,0,0.04)",
              position: {
                xs: "static",
                md: "sticky",
              },
              top: 100,
            }}
          >
            <Typography
              sx={{
                fontWeight: 900,
                textAlign: "center",
                mb: 3,
                fontSize: 24,
                color: "#071a33",
              }}
            >
              絞り込み
            </Typography>

            <Box
              sx={{
                display: "grid",
                gridTemplateColumns: "72px 1fr",
                rowGap: 2,
                columnGap: 1.5,
                alignItems: "center",
                mb: 3,
              }}
            >
              <Typography sx={{ fontSize: 17, fontWeight: 900 }}>
                所属
              </Typography>

              <FormControl fullWidth size="small">
                <Select
                  value={affiliatedId}
                  displayEmpty
                  onChange={(event) => {
                    setAffiliatedId(event.target.value);
                    setPage(0);
                  }}
                  sx={{
                    height: 48,
                    borderRadius: 2,
                    backgroundColor: "#fff",
                    fontSize: 15,
                    fontWeight: 700,
                  }}
                >
                  <MenuItem value="">指定なし</MenuItem>
                  <MenuItem value="1">SHI Staff</MenuItem>
                  <MenuItem value="2">SHI Engineer</MenuItem>
                  <MenuItem value="3">SHI Manager</MenuItem>
                </Select>
              </FormControl>

              <Typography sx={{ fontSize: 17, fontWeight: 900 }}>
                職位
              </Typography>

              <FormControl fullWidth size="small">
                <Select
                  value={positionId}
                  displayEmpty
                  onChange={(event) => {
                    setPositionId(event.target.value);
                    setPage(0);
                  }}
                  sx={{
                    height: 48,
                    borderRadius: 2,
                    backgroundColor: "#fff",
                    fontSize: 15,
                    fontWeight: 700,
                  }}
                >
                  <MenuItem value="">指定なし</MenuItem>
                  <MenuItem value="1">Staff</MenuItem>
                  <MenuItem value="2">Leader</MenuItem>
                  <MenuItem value="3">Manager</MenuItem>
                </Select>
              </FormControl>

              <Typography
                sx={{
                  fontSize: 17,
                  fontWeight: 900,
                  lineHeight: 1.25,
                }}
              >
                悩み
                <br />
                カテゴリ
              </Typography>

              <FormControl fullWidth size="small">
                <Select
                  value={categoryId}
                  displayEmpty
                  onChange={(event) => {
                    setCategoryId(event.target.value);
                    setPage(0);
                  }}
                  sx={{
                    height: 48,

                    backgroundColor: "#fff",

                  }}
                >
                  <MenuItem value="">指定なし</MenuItem>
                  <MenuItem value="1">睡眠・体調</MenuItem>
                  <MenuItem value="2">仕事・タスク</MenuItem>
                  <MenuItem value="3">人間関係</MenuItem>
                  <MenuItem value="4">キャリア</MenuItem>

                </Select>
              </FormControl>
            </Box>

            <Box
              sx={{
                borderTop: "1px dashed #d5cabb",

            >
              <Typography
                sx={{
                  fontWeight: 900,

                }}
              >
                検索
              </Typography>

              <TextField
                fullWidth
