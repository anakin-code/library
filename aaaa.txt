import { useCallback } from "react";
import {
  type AuthContextProps,
  useAuth,
} from "react-oidc-context";

/**
 * 先頭が「/」になっているAPIパスだけを受け付けます。
 *
 * 例：
 * "/api/needs"
 * "/api/tips/1"
 */
type ApiPath = `/${string}`;

/**
 * API通信時のオプションです。
 *
 * RequestInitにアクセストークンを追加しています。
 */
type ApiFetchOptions = RequestInit & {
  accessToken?: string;
};

/**
 * useApiの利用側ではaccessTokenを直接指定させないため、
 * ApiFetchOptionsからaccessTokenを除外します。
 */
type UseApiFetchOptions = Omit<
  ApiFetchOptions,
  "accessToken"
>;

/**
 * サーバーから返されるエラーレスポンスの型です。
 */
type ApiErrorResponse = {
  status?: number;
  code?: string;
  message?: string;
};

/**
 * REST APIのホストです。
 *
 * 例：
 * VITE_REST_HOST=http://localhost:8080
 */
const restHost: string = import.meta.env.VITE_REST_HOST;

/**
 * APIのURLを作成します。
 */
function buildApiUrl(path: ApiPath): string {
  if (path.startsWith("//")) {
    throw new Error("API path must not start with //.");
  }

  return `${restHost}${path}`;
}

/**
 * ブラウザ標準のfetchをラップします。
 *
 * accessTokenが存在する場合は、
 * Authorizationヘッダーを追加します。
 */
function apiFetch(
  path: ApiPath,
  {
    accessToken,
    headers: initHeaders,
    ...init
  }: ApiFetchOptions = {},
): Promise<Response> {
  const headers = new Headers(initHeaders);

  if (accessToken) {
    headers.set(
      "Authorization",
      `Bearer ${accessToken}`,
    );
  }

  return fetch(buildApiUrl(path), {
    ...init,
    headers,
  });
}

/**
 * JSON形式のボディを送信します。
 */
function apiSendJson<TBody>(
  path: ApiPath,
  body: TBody,
  options: ApiFetchOptions = {},
): Promise<Response> {
  const headers = new Headers(options.headers);

  headers.set(
    "Content-Type",
    "application/json",
  );

  return apiFetch(path, {
    ...options,
    method: options.method ?? "POST",
    headers,
    body: JSON.stringify(body),
  });
}

/**
 * フロントエンドからREST APIへアクセスするための
 * カスタムフックです。
 *
 * 認証済みユーザーのアクセストークンを
 * Authorizationヘッダーへ自動的に設定します。
 */
export function useApi(): {
  fetch: (
    path: ApiPath,
    options?: UseApiFetchOptions,
  ) => Promise<Response>;

  sendJson: <TBody>(
    path: ApiPath,
    body: TBody,
    options?: UseApiFetchOptions,
  ) => Promise<Response>;
} {
  const auth: AuthContextProps = useAuth();

  const accessToken = auth.user?.access_token;
  const removeUser = auth.removeUser;

  /**
   * 403エラーを確認します。
   *
   * 現在は動作確認のため、
   * 403が返された場合はすべてアカウント停止として扱います。
   */
  const checkAccountBanned = useCallback(
    async (
      response: Response,
    ): Promise<Response> => {
      if (response.status !== 403) {
        return response;
      }

      let message =
        "このアカウントは利用停止されています";

      try {
        /*
         * responseを直接json化すると、
         * 呼び出し元で本文を再度読めなくなるため、
         * cloneしたResponseを使用します。
         */
        const errorData =
          (await response.clone().json()) as ApiErrorResponse;

        if (
          typeof errorData.message === "string" &&
          errorData.message.length > 0
        ) {
          message = errorData.message;
        }
      } catch (error) {
        /*
         * サーバーのレスポンスがJSONでなかった場合も、
         * 403であればログアウト処理は続けます。
         */
        console.error(
          "403レスポンスの読み込みに失敗しました。",
          error,
        );
      }

      /*
       * ログアウト後に表示するメッセージを一時保存します。
       */
      sessionStorage.setItem(
        "loginErrorMessage",
        message,
      );

      /*
       * react-oidc-contextが保持している
       * ローカルのユーザー情報を削除します。
       */
      await removeUser();

      /*
       * ログイン前の画面へ戻します。
       *
       * プロジェクトのログイン画面が「/login」なら、
       * "/"を"/login"に変更してください。
       */
      window.location.replace("/");

      return response;
    },
    [removeUser],
  );

  /**
   * GET、DELETEなど、
   * JSONボディを自動生成しないリクエストで使用します。
   */
  const fetch = useCallback(
    async (
      path: ApiPath,
      options: UseApiFetchOptions = {},
    ): Promise<Response> => {
      const response = await apiFetch(path, {
        ...options,
        accessToken,
      });

      return checkAccountBanned(response);
    },
    [
      accessToken,
      checkAccountBanned,
    ],
  );

  /**
   * POST、PUT、PATCHなど、
   * JSONボディを送信するリクエストで使用します。
   */
  const sendJson = useCallback(
    async <TBody>(
      path: ApiPath,
      body: TBody,
      options: UseApiFetchOptions = {},
    ): Promise<Response> => {
      const response = await apiSendJson(
        path,
        body,
        {
          ...options,
          accessToken,
        },
      );

      return checkAccountBanned(response);
    },
    [
      accessToken,
      checkAccountBanned,
    ],
  );

  return {
    /** ボディなしでHTTPリクエストを送る場合に使用します */
    fetch,

    /** JSONボディ付きでHTTPリクエストを送る場合に使用します */
    sendJson,
  };
}
