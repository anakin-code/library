
今回は NeedList.tsx の完全版として出す。
修正箇所には // 修正 を入れている。

主な修正はこれ。

1. Select ではなく Autocomplete にする
2. disablePortal は使わない
3. 候補リストの幅を少し広げる
4. スクロールバーは listbox 側だけにする
5. paper 側に overflow: auto を付けない

投稿画面側では Autocomplete を使って選択肢を出していたので、NeedList の絞り込みもそれに寄せる。

NeedList.tsx 完全版
import { useEffect, useMemo, useState } from "react";
import type { ChangeEvent } from "react";

import {
  Alert,
  Autocomplete,
  Box,
  Button,
  CircularProgress,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import type { SelectChangeEvent } from "@mui/material";

import ChevronLeftIcon from "@mui/icons-material/ChevronLeft";
import ChevronRightIcon from "@mui/icons-material/ChevronRight";

import { useNavigate } from "react-router";

import NeedCard from "../components/NeedCard";
import SubmitNeeds from "../components/SubmitNeeds";
import type { Need, NeedPageResponse } from "../models/Need";
import { useApi } from "../lib/useApi";

type Affiliated = {
  affiliatedId: number;
  affiliatedName: string;
};

type Rank = {
  rankId: number;
  rankName: string;
};

type Category = {
  categoryId: number;
  categoryName: string;
};

export default function NeedList(): JSX.Element {
  const api = useApi();
  const navigate = useNavigate();

  const [needList, setNeedList] = useState<Need[]>([]);

  const [orderBy, setOrderBy] = useState("new");
  const [keyword, setKeyword] = useState("");
  const [appliedKeyword, setAppliedKeyword] = useState("");

  const [affiliatedId, setAffiliatedId] = useState("");
  const [positionId, setPositionId] = useState("");
  const [categoryId, setCategoryId] = useState("");

  const [affiliatedList, setAffiliatedList] = useState<Affiliated[]>([]);
  const [rankList, setRankList] = useState<Rank[]>([]);
  const [categoryList, setCategoryList] = useState<Category[]>([]);

  const [page, setPage] = useState(0);
  const [totalPage, setTotalPage] = useState(0);

  const [isLoading, setIsLoading] = useState(true);
  const [isServerStopping, setIsServerStopping] = useState(false);

  const [isPostDialogOpen, setIsPostDialogOpen] = useState(false);

  const selectedAffiliated =
    affiliatedList.find(
      (affiliated) => String(affiliated.affiliatedId) === affiliatedId
    ) ?? null;

  const selectedPosition =
    rankList.find((rank) => String(rank.rankId) === positionId) ?? null;

  const selectedCategory =
    categoryList.find(
      (category) => String(category.categoryId) === categoryId
    ) ?? null;

  // 修正：api.fetch が `/${string}` 型を要求している場合に合わせる
  const requestUrl = useMemo((): `/${string}` => {
    const params = new URLSearchParams();

    params.append("page", String(page));
    params.append("size", "18");
    params.append("orderBy", orderBy);

    if (appliedKeyword.trim() !== "") {
      params.append("keyword", appliedKeyword.trim());
    }

    if (affiliatedId !== "") {
      params.append("affiliatedId", affiliatedId);
    }

    if (positionId !== "") {
      params.append("positionId", positionId);
    }

    if (categoryId !== "") {
      params.append("categoryId", categoryId);
    }

    return `/api/needs/search?${params.toString()}`;
  }, [page, orderBy, appliedKeyword, affiliatedId, positionId, categoryId]);

  useEffect((): (() => void) => {
    let ignore = false;

    const fetchNeeds = async (): Promise<void> => {
      try {
        setIsLoading(true);
        setIsServerStopping(false);

        const response: Response = await api.fetch(requestUrl);

        if (!response.ok) {
          throw new Error("お悩み一覧の取得に失敗しました");
        }

        const data: NeedPageResponse = await response.json();

        if (ignore) {
          return;
        }

        setNeedList(data.content ?? []);
        setTotalPage(data.totalPages ?? 0);
      } catch (error) {
        if (ignore) {
          return;
        }

        console.error(error);
        setNeedList([]);
        setTotalPage(0);
        setIsServerStopping(true);
      } finally {
        if (!ignore) {
          setIsLoading(false);
        }
      }
    };

    fetchNeeds();

    return (): void => {
      ignore = true;
    };
  }, [api, requestUrl]);

  useEffect((): (() => void) => {
    let ignore = false;

    const fetchMasterData = async (): Promise<void> => {
      try {
        const [affiliatedResponse, rankResponse, categoryResponse] =
          await Promise.all([
            api.fetch("/api/affiliated"),
            api.fetch("/api/rank"),
            api.fetch("/api/categories"),
          ]);

        if (ignore) {
          return;
        }

        if (affiliatedResponse.ok) {
          const data: Affiliated[] = await affiliatedResponse.json();
          setAffiliatedList(data ?? []);
        }

        if (rankResponse.ok) {
          const data: Rank[] = await rankResponse.json();
          setRankList(data ?? []);
        }

        if (categoryResponse.ok) {
          const data: Category[] = await categoryResponse.json();
          setCategoryList(data ?? []);
        }
      } catch (error) {
        console.error(error);
      }
    };

    fetchMasterData();

    return (): void => {
      ignore = true;
    };
  }, [api]);

  const handleSearch = (): void => {
    setPage(0);
    setAppliedKeyword(keyword);
  };

  const handleReset = (): void => {
    setKeyword("");
    setAppliedKeyword("");
    setAffiliatedId("");
    setPositionId("");
    setCategoryId("");
    setOrderBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  const handleCardClick = (need: Need): void => {
    navigate(`/need/detail/${need.needId}`);
  };

  const handleUserClick = (need: Need): void => {
    navigate(`/userpage/${need.userId}`);
  };

  const handlePostButtonClick = (): void => {
    setIsPostDialogOpen(true);
  };

  return (
    <Box
      sx={{
        maxWidth: 1060,
        mx: "auto",
        px: 2,
        py: 2.5,
        color: "#071a33",
      }}
    >
      <Box
        sx={{
          display: "grid",
          gridTemplateColumns: {
            xs: "1fr",
            md: "1fr 190px",
          },
          columnGap: 2,
          alignItems: "start",
        }}
      >
        <Box>
          <Box
            sx={{
              position: "relative",
              mb: 2,
            }}
          >
            <Typography
              component="h1"
              sx={{
                textAlign: "center",
                fontSize: 28,
                fontWeight: 900,
                color: "#071a33",
              }}
            >
              お悩み一覧
            </Typography>
          </Box>

          {isServerStopping ? (
            <Alert
              severity="error"
              sx={{
                mb: 2,
                justifyContent: "center",
                maxWidth: 600,
                mx: "auto",
              }}
            >
              サーバーが起動していません。しばらくたってからお試しください。
            </Alert>
          ) : isLoading ? (
            <CircularProgress
              aria-label="Loading"
              sx={{
                display: "block",
                mx: "auto",
                my: 6,
              }}
            />
          ) : needList.length === 0 ? (
            <Typography
              component="h2"
              sx={{
                textAlign: "center",
                fontSize: 28,
                fontWeight: 900,
                color: "#071a33",
              }}
            >
              条件に一致するお悩みがありませんでした
            </Typography>
          ) : (
            <Grid container spacing={2}>
              {needList.map((need, index) => (
                <Grid
                  key={need.needId ?? index}
                  size={{
                    xs: 12,
                    sm: 6,
                    md: 4,
                  }}
                >
                  <NeedCard
                    need={need}
                    onCardClick={handleCardClick}
                    onUserClick={handleUserClick}
                  />
                </Grid>
              ))}
            </Grid>
          )}

          {totalPage > 1 && (
            <Stack
              direction="row"
              sx={{
                pt: 4,
                borderTop: "1px solid",
                borderColor: "rgba(0, 0, 0, 0.06)",
                justifyContent: "space-between",
                alignItems: "center",
              }}
            >
              <Button
                variant="text"
                startIcon={<ChevronLeftIcon />}
                disabled={page === 0}
                onClick={() => setPage(0)}
                sx={{
                  textTransform: "none",
                  fontWeight: 600,
                  px: 2,
                  color: "text.primary",
                  "&:disabled": {
                    color: "text.disabled",
                  },
                }}
              >
                First
              </Button>

              <Button
                variant="text"
                startIcon={<ChevronLeftIcon />}
                disabled={page === 0}
                onClick={() => setPage(page - 1)}
                sx={{
                  borderRadius: 3,
                  textTransform: "none",
                  fontWeight: 600,
                  px: 2,
                  color: "text.primary",
                  "&:disabled": {
                    color: "text.disabled",
                  },
                }}
              >
                前へ
              </Button>

              <Pagination
                count={totalPage}
                page={page + 1}
                onChange={handlePageChange}
                hidePrevButton
                hideNextButton
                color="primary"
                variant="text"
                shape="rounded"
                sx={{
                  "& .MuiPaginationItem-root": {
                    fontWeight: 600,
                    borderRadius: 2,
                  },
                  "& .Mui-selected": {
                    backgroundColor: "primary.main",
                    color: "#fff",
                    "&:hover": {
                      backgroundColor: "primary.dark",
                    },
                  },
                }}
              />

              <Button
                variant="text"
                endIcon={<ChevronRightIcon />}
                disabled={page + 1 >= totalPage}
                onClick={() => setPage(page + 1)}
                sx={{
                  borderRadius: 3,
                  textTransform: "none",
                  fontWeight: 600,
                  px: 2,
                  color: "text.primary",
                  "&:disabled": {
                    color: "text.disabled",
                  },
                }}
              >
                次へ
              </Button>

              <Button
                variant="text"
                endIcon={<ChevronRightIcon />}
                disabled={page + 1 >= totalPage}
                onClick={() => setPage(totalPage - 1)}
                sx={{
                  borderRadius: 3,
                  textTransform: "none",
                  fontWeight: 600,
                  px: 2,
                  color: "text.primary",
                  "&:disabled": {
                    color: "text.disabled",
                  },
                }}
              >
                Last
              </Button>
            </Stack>
          )}
        </Box>

        <Box
          sx={{
            position: {
              xs: "static",
              md: "sticky",
            },
            top: 105,
            mt: {
              xs: 3,
              md: 0,
            },
          }}
        >
          <Box
            sx={{
              backgroundColor: "#f5f0dc",
              borderRadius: 2,
              p: 2,
              mb: 2,
            }}
          >
            <Typography
              sx={{
                fontSize: 18,
                fontWeight: 900,
                textAlign: "center",
                mb: 2,
                color: "#111",
              }}
            >
              絞り込み
            </Typography>

            <Stack spacing={1.6}>
              <TextField
                size="small"
                fullWidth
                placeholder="入力キーワード"
                value={keyword}
                onChange={(event): void => {
                  setKeyword(event.target.value);
                }}
                sx={{
                  backgroundColor: "#fff",
                  borderRadius: 1,
                  "& .MuiOutlinedInput-root": {
                    height: 34,
                    borderRadius: 1,
                    fontSize: 12,
                    fontWeight: 700,
                  },
                }}
              />

              <Box>
                <Typography
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    lineHeight: 1.2,
                    mb: 0.5,
                    color: "#111",
                  }}
                >
                  所属
                </Typography>

                {/* 修正：SelectからAutocompleteへ変更 */}
                {/* 修正：disablePortalは使わない */}
                {/* 修正：paperではなくlistboxだけスクロールさせる */}
                <Autocomplete
                  size="small"
                  options={affiliatedList}
                  value={selectedAffiliated}
                  getOptionLabel={(option: Affiliated): string =>
                    option.affiliatedName
                  }
                  isOptionEqualToValue={(option, value): boolean =>
                    option.affiliatedId === value.affiliatedId
                  }
                  onChange={(_, value: Affiliated | null): void => {
                    setAffiliatedId(value ? String(value.affiliatedId) : "");
                    setPage(0);
                  }}
                  noOptionsText="候補がありません"
                  renderInput={(params) => (
                    <TextField
                      {...params}
                      placeholder="指定なし"
                      sx={{
                        backgroundColor: "#fff",
                        borderRadius: 1,
                        "& .MuiOutlinedInput-root": {
                          minHeight: 34,
                          borderRadius: 1,
                          fontSize: 12,
                          fontWeight: 700,
                        },
                      }}
                    />
                  )}
                  slotProps={{
                    popper: {
                      placement: "bottom-start",
                      sx: {
                        width: "220px !important",
                        zIndex: 1300,
                      },
                    },
                    paper: {
                      sx: {
                        mt: 0.5,
                        borderRadius: 1,

                        // 修正：ここに overflow: "auto" を入れない
                        overflow: "visible",
                      },
                    },
                    listbox: {
                      sx: {
                        // 修正：スクロールは listbox だけに持たせる
                        maxHeight: 260,
                        overflowY: "auto",
                        overflowX: "hidden",
                        py: 0.5,
                        "& .MuiAutocomplete-option": {
                          fontSize: 13,
                          fontWeight: 700,
                          py: 1,
                          whiteSpace: "normal",
                          wordBreak: "keep-all",
                        },
                      },
                    },
                  }}
                />
              </Box>

              <Box>
                <Typography
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    lineHeight: 1.2,
                    mb: 0.5,
                    color: "#111",
                  }}
                >
                  職位
                </Typography>

                {/* 修正：SelectからAutocompleteへ変更 */}
                {/* 修正：二重スクロール対策済み */}
                <Autocomplete
                  size="small"
                  options={rankList}
                  value={selectedPosition}
                  getOptionLabel={(option: Rank): string => option.rankName}
                  isOptionEqualToValue={(option, value): boolean =>
                    option.rankId === value.rankId
                  }
                  onChange={(_, value: Rank | null): void => {
                    setPositionId(value ? String(value.rankId) : "");
                    setPage(0);
                  }}
                  noOptionsText="候補がありません"
                  renderInput={(params) => (
                    <TextField
                      {...params}
                      placeholder="指定なし"
                      sx={{
                        backgroundColor: "#fff",
                        borderRadius: 1,
                        "& .MuiOutlinedInput-root": {
                          minHeight: 34,
                          borderRadius: 1,
                          fontSize: 12,
                          fontWeight: 700,
                        },
                      }}
                    />
                  )}
                  slotProps={{
                    popper: {
                      placement: "bottom-start",
                      sx: {
                        width: "220px !important",
                        zIndex: 1300,
                      },
                    },
                    paper: {
                      sx: {
                        mt: 0.5,
                        borderRadius: 1,
                        overflow: "visible",
                      },
                    },
                    listbox: {
                      sx: {
                        maxHeight: 260,
                        overflowY: "auto",
                        overflowX: "hidden",
                        py: 0.5,
                        "& .MuiAutocomplete-option": {
                          fontSize: 13,
                          fontWeight: 700,
                          py: 1,
                          whiteSpace: "normal",
                          wordBreak: "keep-all",
                        },
                      },
                    },
                  }}
                />
              </Box>

              <Box>
                <Typography
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    lineHeight: 1.2,
                    mb: 0.5,
                    color: "#111",
                  }}
                >
                  悩み
                  <br />
                  カテゴリ
                </Typography>

                {/* 修正：SelectからAutocompleteへ変更 */}
                {/* 修正：候補リストだけ広げる */}
                {/* 修正：二重スクロールを防ぐ */}
                <Autocomplete
                  size="small"
                  options={categoryList}
                  value={selectedCategory}
                  getOptionLabel={(option: Category): string =>
                    option.categoryName
                  }
                  isOptionEqualToValue={(option, value): boolean =>
                    option.categoryId === value.categoryId
                  }
                  onChange={(_, value: Category | null): void => {
                    setCategoryId(value ? String(value.categoryId) : "");
                    setPage(0);
                  }}
                  noOptionsText="候補がありません"
                  renderInput={(params) => (
                    <TextField
                      {...params}
                      placeholder="指定なし"
                      sx={{
                        backgroundColor: "#fff",
                        borderRadius: 1,
                        "& .MuiOutlinedInput-root": {
                          minHeight: 34,
                          borderRadius: 1,
                          fontSize: 12,
                          fontWeight: 700,
                        },
                      }}
                    />
                  )}
                  slotProps={{
                    popper: {
                      placement: "bottom-start",
                      sx: {
                        width: "220px !important",
                        zIndex: 1300,
                      },
                    },
                    paper: {
                      sx: {
                        mt: 0.5,
                        borderRadius: 1,

                        // 修正：paper側はスクロールさせない
                        overflow: "visible",
                      },
                    },
                    listbox: {
                      sx: {
                        // 修正：スクロールはここだけ
                        maxHeight: 260,
                        overflowY: "auto",
                        overflowX: "hidden",
                        py: 0.5,
                        "& .MuiAutocomplete-option": {
                          fontSize: 13,
                          fontWeight: 700,
                          py: 1,
                          whiteSpace: "normal",
                          wordBreak: "keep-all",
                        },
                      },
                    },
                  }}
                />
              </Box>

              <Stack direction="row" spacing={1}>
                <Button
                  fullWidth
                  variant="outlined"
                  onClick={handleReset}
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    color: "#111",
                    borderColor: "#d6c8a8",
                    backgroundColor: "#fff8e8",
                    "&:hover": {
                      borderColor: "#d6c8a8",
                      backgroundColor: "#fff8e8",
                    },
                  }}
                >
                  リセット
                </Button>

                <Button
                  fullWidth
                  variant="contained"
                  onClick={handleSearch}
                  sx={{
                    fontSize: 12,
                    fontWeight: 900,
                    backgroundColor: "#071a33",
                    color: "#fff",
                    "&:hover": {
                      backgroundColor: "#071a33",
                    },
                  }}
                >
                  実行
                </Button>
              </Stack>
            </Stack>
          </Box>

          <Box sx={{ mb: 2 }}>
            <FormControl fullWidth size="small">
              <Select
                value={orderBy}
                displayEmpty
                onChange={(event: SelectChangeEvent<string>): void => {
                  setOrderBy(event.target.value);
                  setPage(0);
                }}
                sx={{
                  height: 34,
                  borderRadius: 1.3,
                  backgroundColor: "#fff",
                  fontWeight: 800,
                  fontSize: 12,
                  color: "#071a33",
                }}
              >
                <MenuItem value="new">新着順</MenuItem>
                <MenuItem value="empathy">共感数順</MenuItem>
              </Select>
            </FormControl>
          </Box>

          <Button
            fullWidth
            variant="contained"
            onClick={handlePostButtonClick}
            sx={{
              borderRadius: 999,
              backgroundColor: "#071a33",
              color: "#fff",
              fontSize: 16,
              fontWeight: 900,
              py: 1.2,
              lineHeight: 1.2,
              "&:hover": {
                backgroundColor: "#071a33",
              },
            }}
          >
            お悩み
            <br />
            投稿
          </Button>
        </Box>
      </Box>

      {isPostDialogOpen && (
        <SubmitNeeds
          onCancel={() => {
            setIsPostDialogOpen(false);
          }}
        />
      )}
    </Box>
  );
}
今回の修正箇所だけ抜き出すと
1. disablePortal を消す
// 修正前
<Autocomplete
  disablePortal
  ...
/>
// 修正後
<Autocomplete
  ...
/>
2. paper 側に overflow: auto を入れない
// 修正前
paper: {
  sx: {
    maxHeight: 260,
    overflow: "auto",
  },
},
// 修正後
paper: {
  sx: {
    mt: 0.5,
    borderRadius: 1,
    overflow: "visible",
  },
},
3. スクロールは listbox だけ
listbox: {
  sx: {
    maxHeight: 260,
    overflowY: "auto",
    overflowX: "hidden",
  },
},
4. 候補リストの幅を広げる
popper: {
  placement: "bottom-start",
  sx: {
    width: "220px !important",
    zIndex: 1300,
  },
},

これで、候補リストが狭すぎて文字が変に折り返される問題と、スクロールバーが二重に見える問題を抑えられる。
