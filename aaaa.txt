import { useCallback } from "react";
import {
  type AuthContextProps,
  useAuth,
} from "react-oidc-context";

/**
 * 先頭が「/」のAPIパスだけを受け付けます。
 *
 * 例：
 * "/api/needs"
 * "/api/tips/1"
 */
type ApiPath = `/${string}`;

/**
 * API通信に渡すオプションです。
 *
 * RequestInitに、Cognitoのアクセストークンを追加しています。
 */
type ApiFetchOptions = RequestInit & {
  accessToken?: string;
};

/**
 * useApiを使用する画面側では、
 * accessTokenを直接渡さないため除外します。
 */
type UseApiFetchOptions = Omit<ApiFetchOptions, "accessToken">;

/**
 * サーバーから返されるエラーレスポンスです。
 */
type ApiErrorResponse = {
  status?: number;
  code?: string;
  message?: string;
};

/**
 * 環境変数に設定されたREST APIのホスト名です。
 *
 * 例：
 * http://localhost:8080
 */
const restHost: string = import.meta.env.VITE_REST_HOST;

/**
 * REST APIのURLを生成します。
 */
function buildApiUrl(path: ApiPath): string {
  if (path.startsWith("//")) {
    throw new Error("API path must not start with //.");
  }

  return `${restHost}${path}`;
}

/**
 * ブラウザ標準のfetchをラップし、
 * Authorizationヘッダーにアクセストークンを設定します。
 */
function apiFetch(
  path: ApiPath,
  {
    accessToken,
    headers: initHeaders,
    ...init
  }: ApiFetchOptions = {},
): Promise<Response> {
  const headers = new Headers(initHeaders);

  if (accessToken) {
    headers.set("Authorization", `Bearer ${accessToken}`);
  }

  return fetch(buildApiUrl(path), {
    ...init,
    headers,
  });
}

/**
 * JSON形式のボディを送信します。
 */
function apiSendJson<TBody>(
  path: ApiPath,
  body: TBody,
  options: ApiFetchOptions = {},
): Promise<Response> {
  const headers = new Headers(options.headers);

  headers.set("Content-Type", "application/json");

  return apiFetch(path, {
    ...options,
    method: options.method ?? "POST",
    headers,
    body: JSON.stringify(body),
  });
}

/**
 * フロントエンドからREST APIへアクセスするための
 * カスタムフックです。
 *
 * Cognitoから取得したアクセストークンを、
 * Authorizationヘッダーに自動で設定します。
 *
 * また、サーバーからアカウント停止エラーが返された場合は、
 * ローカルの認証情報を削除してログイン前の画面に戻します。
 */
export function useApi(): {
  fetch: (
    path: ApiPath,
    options?: UseApiFetchOptions,
  ) => Promise<Response>;

  sendJson: <TBody>(
    path: ApiPath,
    body: TBody,
    options?: UseApiFetchOptions,
  ) => Promise<Response>;
} {
  const auth: AuthContextProps = useAuth();

  const accessToken = auth.user?.access_token;
  const removeUser = auth.removeUser;

  /**
   * アカウント停止レスポンスを確認します。
   *
   * 通常のレスポンスはそのまま返します。
   * ACCOUNT_BANNEDの場合はログアウトさせます。
   */
  const checkAccountBanned = useCallback(
    async (response: Response): Promise<Response> => {
      if (response.status !== 403) {
        return response;
      }

      try {
        /*
         * response.json()を直接実行すると、
         * 呼び出し元でレスポンス本文を読めなくなるため、
         * clone()したレスポンスを使用します。
         */
        const errorData =
          (await response.clone().json()) as ApiErrorResponse;

        if (errorData.code !== "ACCOUNT_BANNED") {
          return response;
        }

        sessionStorage.setItem(
          "loginErrorMessage",
          errorData.message ??
            "このアカウントは利用停止されています",
        );

        /*
         * react-oidc-contextが保持している
         * ローカルのログイン情報を削除します。
         */
        await removeUser();

        /*
         * 認証前の画面へ戻します。
         *
         * 「/」を開いた際に認証処理が動く現在の構成を想定しています。
         */
        window.location.replace("/");

        return response;
      } catch (error) {
        console.error(
          "アカウント停止レスポンスの解析に失敗しました。",
          error,
        );

        return response;
      }
    },
    [removeUser],
  );

  /**
   * ボディなし、またはRequestInitを直接指定して
   * HTTPリクエストを送信します。
   */
  const fetch = useCallback(
    async (
      path: ApiPath,
      options: UseApiFetchOptions = {},
    ): Promise<Response> => {
      const response = await apiFetch(path, {
        ...options,
        accessToken,
      });

      return checkAccountBanned(response);
    },
    [accessToken, checkAccountBanned],
  );

  /**
   * JSON形式のボディ付きで
   * HTTPリクエストを送信します。
   */
  const sendJson = useCallback(
    async <TBody>(
      path: ApiPath,
      body: TBody,
      options: UseApiFetchOptions = {},
    ): Promise<Response> => {
      const response = await apiSendJson(path, body, {
        ...options,
        accessToken,
      });

      return checkAccountBanned(response);
    },
    [accessToken, checkAccountBanned],
  );

  return {
    /** ボディなしでHTTPリクエストを送信するときに利用します */
    fetch,

    /** ボディ付きでHTTPリクエストを送信するときに利用します */
    sendJson,
  };
}
