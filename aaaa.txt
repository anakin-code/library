import { useEffect, useMemo, useState } from "react";
import dagre from "dagre";
import {
  Background,
  Controls,
  ReactFlow,
  Position,
  type Edge,
  type Node,
} from "@xyflow/react";
import "@xyflow/react/dist/style.css";
import api from "../../api/api";

type SkillNode = {
  id: number;
  label: string;
  months: number;
  normalizedSize: number;
};

type SkillEdge = {
  source: number;
  target: number;
  months: number;
  normalizedWeight: number;
};

type SkillNetworkResponse = {
  nodes: SkillNode[];
  edges: SkillEdge[];
};

type Props = {
  hrid: string;
};

const nodeWidth = 120;
const nodeHeight = 120;

function layoutNodesAndEdges(
  nodes: Node[],
  edges: Edge[]
) {
  const graph = new dagre.graphlib.Graph();

  graph.setDefaultEdgeLabel(() => ({}));

  graph.setGraph({
    rankdir: "LR",
    ranksep: 120,
    nodesep: 80,
  });

  nodes.forEach((node) => {
    graph.setNode(node.id, {
      width: nodeWidth,
      height: nodeHeight,
    });
  });

  edges.forEach((edge) => {
    graph.setEdge(edge.source, edge.target);
  });

  dagre.layout(graph);

  const layoutedNodes = nodes.map((node) => {
    const position = graph.node(node.id);

    return {
      ...node,
      position: {
        x: position.x - nodeWidth / 2,
        y: position.y - nodeHeight / 2,
      },
      targetPosition: Position.Left,
      sourcePosition: Position.Right,
    };
  });

  return {
    nodes: layoutedNodes,
    edges,
  };
}

export default function PersonalSkillNetwork({ hrid }: Props) {
  const [data, setData] = useState<SkillNetworkResponse>({
    nodes: [],
    edges: [],
  });

  useEffect(() => {
    if (!hrid) {
      return;
    }

    api.get<SkillNetworkResponse>(`/user/${hrid}/skill-network`)
      .then((response) => {
        setData(response.data);
      })
      .catch((error) => {
        console.error(error);
        setData({
          nodes: [],
          edges: [],
        });
      });
  }, [hrid]);

  const rawNodes: Node[] = useMemo(() => {
    return data.nodes.map((skill) => {
      const size = 60 + Math.sqrt(skill.months) * 10;

      return {
        id: String(skill.id),
        position: {
          x: 0,
          y: 0,
        },
        data: {
          label: `${skill.label}\n${skill.months}ヶ月`,
        },
        style: {
          width: size,
          height: size,
          background: "#d7f3e3",
          border: "2px solid #56b68b",
          borderRadius: "50%",
          color: "#222",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          textAlign: "center",
          fontSize: 12,
          fontWeight: 600,
          whiteSpace: "pre-line",
          boxShadow: "0 2px 8px rgba(0,0,0,0.15)",
        },
      };
    });
  }, [data]);

  const rawEdges: Edge[] = useMemo(() => {
    return data.edges.map((edge) => ({
      id: `${edge.source}-${edge.target}`,
      source: String(edge.source),
      target: String(edge.target),
      type: "straight",
      label: `${edge.months}ヶ月`,
      style: {
        stroke: "#334155",
        strokeWidth: 1.5 + edge.normalizedWeight * 4,
      },
      labelStyle: {
        fontSize: 11,
        fontWeight: 600,
        fill: "#333",
      },
    }));
  }, [data]);

  const layouted = useMemo(() => {
    return layoutNodesAndEdges(rawNodes, rawEdges);
  }, [rawNodes, rawEdges]);

  return (
    <div style={{ width: "100%", height: "600px" }}>
      <ReactFlow
        nodes={layouted.nodes}
        edges={layouted.edges}
        fitView
        fitViewOptions={{
          padding: 0.25,
        }}
        minZoom={0.3}
        maxZoom={2}
      >
        <Background color="#e5e7eb" gap={24} />
        <Controls />
      </ReactFlow>
    </div>
  );
}
