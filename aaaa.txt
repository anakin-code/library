const getServerErrorMessage = (error: unknown): string => {
    if (axios.isAxiosError(error)) {
        const errorData = error.response?.data;

        if (typeof errorData === "string") {
            return errorData;
        }

        if (
            typeof errorData === "object" &&
            errorData !== null &&
            "message" in errorData
        ) {
            return String((errorData as { message?: string }).message);
        }
    }

    return "処理に失敗しました。";
};

// ===== スキル削除処理 追加ここから =====
const handleDeleteSkill = async (): Promise<void> => {
    if (!deleteSkillId) {
        setErrorMessage("削除するスキルを選択してください。");
        return;
    }

    try {
        setDeleteSkillLoading(true);
        setErrorMessage(null);
        setSuccessMessage(null);

        await api.delete(`/skill/${deleteSkillId}`);

        setAllAvailableSkills((prev) =>
            prev.filter((skill) => String(skill.skillId) !== String(deleteSkillId))
        );

        setSkillExperiences((prev) =>
            prev.filter((skill) => String(skill.skillId) !== String(deleteSkillId))
        );

        setCareers((prev) =>
            prev.map((career) => ({
                ...career,
                careerSkills: career.careerSkills.filter(
                    (skill) => String(skill.skillId) !== String(deleteSkillId)
                ),
            }))
        );

        setSuccessMessage("スキルを削除しました。");
        setShowDeleteSkillModal(false);
        setDeleteSkillId("");
    } catch (error: unknown) {
        setErrorMessage(getServerErrorMessage(error));
    } finally {
        setDeleteSkillLoading(false);
    }
};
// ===== スキル削除処理 追加ここまで =====
