package simplex.pj148.skippy.server.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import simplex.pj148.skippy.server.entity.Tip;

public interface TipRepository extends JpaRepository<Tip, Long> {

    @Query(
            value = """
                    SELECT t
                    FROM Tip t
                    LEFT JOIN Tipinterest ti ON ti.tip = t
                    LEFT JOIN Comment c ON c.tip = t AND c.deleteFlg = false
                    WHERE t.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(t.tipTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.tipDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.user.userName) LIKE LOWER(CONCAT('%', :keyword, '%'))
                          )
                      AND (:affiliatedId IS NULL OR t.user.affiliated.affiliatedId = :affiliatedId)
                      AND (:rankId IS NULL OR t.user.rank.rankId = :rankId)
                      AND (:categoryId IS NULL OR t.category.categoryId = :categoryId)
                    GROUP BY t
                    ORDER BY
                      CASE WHEN :orderBy = 'interestCount' THEN COUNT(DISTINCT ti) END DESC,
                      CASE WHEN :orderBy = 'commentCount' THEN COUNT(DISTINCT c) END DESC,
                      t.createdTime DESC
                    """,
            countQuery = """
                    SELECT COUNT(t)
                    FROM Tip t
                    WHERE t.deleteFlg = false
                      AND (
                            :keyword IS NULL
                            OR :keyword = ''
                            OR LOWER(t.tipTitle) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.tipDetail) LIKE LOWER(CONCAT('%', :keyword, '%'))
                            OR LOWER(t.user.userName) LIKE LOWER(CONCAT('%', :keyword, '%'))
                          )
                      AND (:affiliatedId IS NULL OR t.user.affiliated.affiliatedId = :affiliatedId)
                      AND (:rankId IS NULL OR t.user.rank.rankId = :rankId)
                      AND (:categoryId IS NULL OR t.category.categoryId = :categoryId)
                    """
    )
    Page<Tip> searchTips(
            @Param("keyword") String keyword,
            @Param("affiliatedId") Long affiliatedId,
            @Param("rankId") Long rankId,
            @Param("categoryId") Long categoryId,
            @Param("orderBy") String orderBy,
            Pageable pageable
    );
}

5. URL例

全部なしでもOK。

GET /api/tips
GET /api/needs

キーワードだけでもOK。

GET /api/tips?keyword=睡眠
GET /api/needs?keyword=睡眠

絞り込みだけでもOK。

GET /api/tips?affiliatedId=1
GET /api/tips?rankId=2
GET /api/tips?categoryId=3

全部指定もOK。

GET /api/tips?keyword=睡眠&affiliatedId=1&rankId=2&categoryId=3&orderBy=interestCount&page=0&size=20

検索条件の意味はこれ。

Tip:
(tipTitle OR tipDetail OR userName に keyword を含む)
AND affiliatedId
AND rankId
AND categoryId

Need:
(needTitle OR needDetail OR userName に keyword を含む)
AND affiliatedId
AND rankId
AND categoryId

keyword・affiliatedId・rankId・categoryId がない場合は、その条件は無視される。

2. fetch関数を作る
ノウハウ検索
export const fetchTips = async (params: TipSearchParams = {}) => {
  const searchParams = new URLSearchParams();

  if (params.keyword && params.keyword.trim() !== "") {
    searchParams.append("keyword", params.keyword.trim());
  }

  if (params.affiliatedId != null) {
    searchParams.append("affiliatedId", String(params.affiliatedId));
  }

  if (params.rankId != null) {
    searchParams.append("rankId", String(params.rankId));
  }

  if (params.categoryId != null) {
    searchParams.append("categoryId", String(params.categoryId));
  }

  if (params.orderBy && params.orderBy !== "") {
    searchParams.append("orderBy", params.orderBy);
  }

  if (params.page != null) {
    searchParams.append("page", String(params.page));
  }

  if (params.size != null) {
    searchParams.append("size", String(params.size));
  }

  const query = searchParams.toString();
  const url = query ? `/api/tips?${query}` : "/api/tips";

  const response = await fetch(url);

  if (!response.ok) {
    throw new Error("ノウハウ一覧の取得に失敗しました");
  }

  return response.json();
};

7. 並び替え

Server 側の orderBy に合わせるなら、client 側はこういう値を送る。

ノウハウなら、

"createdTime"
"interestCount"
"commentCount"

お悩みなら、

"createdTime"
"empathyCount"
"answerCount"

たとえば、

setOrderBy("interestCount");

として検索実行すると、

GET /api/tips?orderBy=interestCount&page=0&size=20

になる。

8. client が最終的にやること一覧

まとめると、client側はこれ。

1. positionId を rankId に直す
2. 検索欄 keyword の state を作る
3. 所属 affiliatedId の state を作る
4. 職位 rankId の state を作る
5. カテゴリ categoryId の state を作る
6. orderBy の state を作る
7. page / size の state を作る
8. URLSearchParams で、値があるものだけ query に付ける
9. GET /api/tips または GET /api/needs を叩く
10. response の data.content を一覧カードに表示する
