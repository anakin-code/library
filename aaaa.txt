型を追加：

type SkillCircleNodeData = {
  label: string;
  style: React.CSSProperties;
};

SkillCircleNode をこれに変更：

function SkillCircleNode({
  data,
}: NodeProps<Node<SkillCircleNodeData>>) {
  return (
    <div style={{ ...data.style, position: "relative" }}>
      <Handle
        type="source"
        position={Position.Top}
        style={{
          left: "50%",
          top: "50%",
          transform: "translate(-50%, -50%)",
          opacity: 0,
          pointerEvents: "none",
        }}
      />

      <Handle
        type="target"
        position={Position.Top}
        style={{
          left: "50%",
          top: "50%",
          transform: "translate(-50%, -50%)",
          opacity: 0,
          pointerEvents: "none",
        }}
      />

      {data.label}
    </div>
  );
}
