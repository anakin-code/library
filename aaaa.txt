7. CareerService.java の重要部分
@Service
public class CareerService {

    private final CareerRepository careerRepository;
    private final ProfileRepository profileRepository;
    private final ProjectRepository projectRepository;
    private final SkillRepository skillRepository;
    private final CareerSkillRepository careerSkillRepository;
    private final ProjectRoleRepository projectRoleRepository;

    public CareerService(
            CareerRepository careerRepository,
            ProfileRepository profileRepository,
            ProjectRepository projectRepository,
            SkillRepository skillRepository,
            CareerSkillRepository careerSkillRepository,
            ProjectRoleRepository projectRoleRepository
    ) {
        this.careerRepository = careerRepository;
        this.profileRepository = profileRepository;
        this.projectRepository = projectRepository;
        this.skillRepository = skillRepository;
        this.careerSkillRepository = careerSkillRepository;
        this.projectRoleRepository = projectRoleRepository;
    }

    public CareerResponse createCareer(CreateCareerRequest request) {

        Profile profile = profileRepository.findById(request.getHrid())
                .orElseThrow(() -> new IllegalArgumentException("プロフィールが見つかりません: " + request.getHrid()));

        Project project = projectRepository.findById(request.getProjectId())
                .orElseThrow(() -> new IllegalArgumentException("プロジェクトが見つかりません: " + request.getProjectId()));

        ProjectRole role = projectRoleRepository.findByRoleName(request.getRoleName())
                .orElseThrow(() -> new IllegalArgumentException("ロールが見つかりません: " + request.getRoleName()));

        Career career = new Career();
        career.setProfile(profile);
        career.setProject(project);
        career.setProjectRole(role);
        career.setStartTime(request.getStartTime());
        career.setEndTime(request.getEndTime());

        Career savedCareer = careerRepository.save(career);

        if (request.getSkillIds() != null) {
            for (Long skillId : request.getSkillIds()) {
                Skill skill = skillRepository.findById(skillId)
                        .orElseThrow(() -> new IllegalArgumentException("スキルが見つかりません: " + skillId));

                CareerSkill careerSkill = new CareerSkill();
                careerSkill.setCareer(savedCareer);
                careerSkill.setSkill(skill);

                careerSkillRepository.save(careerSkill);
            }
        }

        return convertToResponse(savedCareer);
    }

    public CareerResponse updateCareer(Long careerId, UpdateCareerRequest request) {

        Career career = careerRepository.findById(careerId)
                .orElseThrow(() -> new IllegalArgumentException("経歴が見つかりません: " + careerId));

        if (request.getProjectId() != null) {
            Project project = projectRepository.findById(request.getProjectId())
                    .orElseThrow(() -> new IllegalArgumentException("プロジェクトが見つかりません: " + request.getProjectId()));
            career.setProject(project);
        }

        if (request.getRoleName() != null) {
            ProjectRole role = projectRoleRepository.findByRoleName(request.getRoleName())
                    .orElseThrow(() -> new IllegalArgumentException("ロールが見つかりません: " + request.getRoleName()));
            career.setProjectRole(role);
        }

        if (request.getStartTime() != null) {
            career.setStartTime(request.getStartTime());
        }

        if (request.getEndTime() != null) {
            career.setEndTime(request.getEndTime());
        }

        Career savedCareer = careerRepository.save(career);

        return convertToResponse(savedCareer);
    }

    private CareerResponse convertToResponse(Career career) {

        CareerResponse response = new CareerResponse();

        response.setCareerId(career.getCareerId());

        response.setHrid(
                career.getProfile() != null
                        ? career.getProfile().getHrid()
                        : null
        );

        response.setProjectName(
                career.getProject() != null
                        ? career.getProject().getProjectName()
                        : null
        );

        response.setRoleName(
                career.getProjectRole() != null
                        ? career.getProjectRole().getRoleName()
                        : null
        );

        response.setStartTime(career.getStartTime());
        response.setEndTime(career.getEndTime());

        if (career.getCareerSkills() != null) {
            List<String> skillNames = career.getCareerSkills().stream()
                    .map(cs -> cs.getSkill().getSkillName())
                    .toList();

            response.setSkillNames(skillNames);
        }

        return response;
    }
}
