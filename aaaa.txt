package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import simplex.pj148.skippy.server.controller.dto.response.NeedEmpathyResponseDto;
import simplex.pj148.skippy.server.entity.Need;
import simplex.pj148.skippy.server.entity.NeedEmpathy;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.NeedEmpathyRepository;
import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

@Service
@RequiredArgsConstructor
public class NeedEmpathyService {

    private final NeedEmpathyRepository needEmpathyRepository;

    private final NeedRepository needRepository;

    private final UserRepository userRepository;

    @Transactional
    public NeedEmpathyResponseDto createNeedEmpathy(Long needId, Long loginUserId) {
        Need need = needRepository.findByNeedIdAndDeleteFlgFalse(needId)
                .orElseThrow(() -> new RuntimeException("お悩みが存在しません"));

        User user = userRepository.findByUserId(loginUserId)
                .orElseThrow(() -> new RuntimeException("ユーザーが存在しません"));

        boolean alreadyEmpathy = needEmpathyRepository
                .existsByUser_UserIdAndNeed_NeedId(loginUserId, needId);

        if (!alreadyEmpathy) {
            NeedEmpathy needEmpathy = NeedEmpathy.builder()
                    .need(need)
                    .user(user)
                    .isLookedByReactor(true)
                    .isLookedByPoster(false)
                    .build();

            needEmpathyRepository.save(needEmpathy);
        }

        return buildNeedEmpathyResponse(needId, loginUserId);
    }

    @Transactional
    public NeedEmpathyResponseDto deleteNeedEmpathy(Long needId, Long loginUserId) {
        NeedEmpathy needEmpathy = needEmpathyRepository
                .findByUser_UserIdAndNeed_NeedId(loginUserId, needId)
                .orElse(null);

        if (needEmpathy != null) {
            needEmpathyRepository.delete(needEmpathy);
        }

        return buildNeedEmpathyResponse(needId, loginUserId);
    }

    @Transactional(readOnly = true)
    public NeedEmpathyResponseDto buildNeedEmpathyResponse(Long needId, Long loginUserId) {
        Long empathyCount = needEmpathyRepository.countByNeed_NeedId(needId);

        Boolean empathyByCurrentUser = needEmpathyRepository
                .existsByUser_UserIdAndNeed_NeedId(loginUserId, needId);

        return NeedEmpathyResponseDto.builder()
                .needId(needId)
                .empathyCount(empathyCount)
                .empathyByCurrentUser(empathyByCurrentUser)
                .build();
    }
}

package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;

import simplex.pj148.skippy.server.controller.dto.response.NeedEmpathyResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.NeedEmpathyService;
import simplex.pj148.skippy.server.service.UserInfoService;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/needs")
public class NeedEmpathyController {

    private final NeedEmpathyService needEmpathyService;

    private final UserInfoService userInfoService;

    @PostMapping("/{needId}/empathy")
    public NeedEmpathyResponseDto createNeedEmpathy(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long needId
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return needEmpathyService.createNeedEmpathy(
                needId,
                loginUser.getUserId()
        );
    }

    @DeleteMapping("/{needId}/empathy")
    public NeedEmpathyResponseDto deleteNeedEmpathy(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long needId
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return needEmpathyService.deleteNeedEmpathy(
                needId,
                loginUser.getUserId()
        );
    }
}
