9. TipRepository.java

旧 ContentRepository の置き換え。
deleteFlg なしなので privateFlg=false, deleteFlg=false みたいな条件は全部なし。

package simplex.pj148.skippy.server.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

import simplex.pj148.skippy.server.entity.Tip;

import java.util.List;
import java.util.Optional;

public interface TipRepository extends JpaRepository<Tip, Long> {

    Optional<Tip> findByTipId(Long tipId);

    Page<Tip> findAllByOrderByCreatedTimeDesc(Pageable pageable);

    Page<Tip> findByUser_UserIdOrderByCreatedTimeDesc(
            Long userId,
            Pageable pageable
    );

    Page<Tip> findByCategory_CategoryIdOrderByCreatedTimeDesc(
            Long categoryId,
            Pageable pageable
    );

    List<Tip> findByNeed_NeedsId(Long needsId);

    Page<Tip> findByTipTitleContainingIgnoreCaseOrderByCreatedTimeDesc(
            String tipTitle,
            Pageable pageable
    );

    @Query("""
            SELECT t
            FROM Tip t
            WHERE (:categoryId = 0 OR t.category.categoryId = :categoryId)
            ORDER BY t.createdTime DESC
            """)
    Page<Tip> findSelectedTipsSortedByCreatedTime(
            Pageable pageable,
            @Param("categoryId") Long categoryId
    );

    @Query("""
            SELECT t
            FROM Tip t
            LEFT JOIN TipInterest ti ON ti.tip = t
            WHERE (:categoryId = 0 OR t.category.categoryId = :categoryId)
            GROUP BY t
            ORDER BY COUNT(ti) DESC, t.createdTime DESC
            """)
    Page<Tip> findSelectedTipsSortedByInterestCount(
            Pageable pageable,
            @Param("categoryId") Long categoryId
    );

    @Query("""
            SELECT DISTINCT t
            FROM Tip t
            LEFT JOIN t.user u
            LEFT JOIN t.category c
            LEFT JOIN TipsTag tt ON tt.tip = t
            LEFT JOIN tt.tag tag
            WHERE (:context = 'title' AND LOWER(t.tipTitle) LIKE LOWER(CONCAT('%', :keyword, '%')))
               OR (:context = 'user' AND LOWER(u.userName) LIKE LOWER(CONCAT('%', :keyword, '%')))
               OR (:context = 'category' AND LOWER(c.categoryName) LIKE LOWER(CONCAT('%', :keyword, '%')))
               OR (:context = 'tag' AND LOWER(tag.tagName) LIKE LOWER(CONCAT('%', :keyword, '%')))
            ORDER BY t.createdTime DESC
            """)
    Page<Tip> searchTips(
            @Param("keyword") String keyword,
            @Param("context") String context,
            Pageable pageable
    );
}

ここでは検索をRepositoryに寄せた版にしている。
ただし、前に話した「空白区切り複数キーワード検索」までやるなら、この @Query ではなく Custom Repository / Criteria API の方がよい。今回はRepository完全版として、まず1文字列検索の形にしている。

package simplex.pj148.skippy.server.repository;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;

public interface TipInterestRepository extends JpaRepository<TipInterest, Long> {

            Long tipId
    );

    List<TipInterest> findByUser_UserId(Long userId);

    List<TipInterest> findByTip_TipId(Long tipId);

    Long countByTip_TipId(Long tipId);

    @Query("""
            SELECT ti.tip.tipId, COUNT(ti)
            FROM TipInterest ti
            WHERE ti.tip.tipId IN :tipIds
            GROUP BY ti.tip.tipId
            """)
    List<Object[]> countInterestsByTipIds(
            @Param("tipIds") List<Long> tipIds
    );

    List<TipInterest> findByUser_UserIdAndTip_TipIdIn(

13. NotificationRepository.java

通知設定用。
deleteFlg ではなく、通知ON/OFFは isActive にするのが自然。

package simplex.pj148.skippy.server.repository;

import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.stereotype.Repository;
import simplex.pj148.skippy.server.entity.Notification;

import java.time.LocalTime;
import java.util.List;
import java.util.Optional;

@Repository
public interface NotificationRepository extends JpaRepository<Notification, Long> {

    Optional<Notification> findByNotificationId(Long notificationId);

    Optional<Notification> findByTipInterest_TipInterestId(Long tipInterestId);

    boolean existsByTipInterest_TipInterestId(Long tipInterestId);

    List<Notification> findByTipInterest_User_UserId(Long userId);

    List<Notification> findByTipInterest_User_UserIdAndIsActiveTrue(Long userId);

    List<Notification> findByFrequency_FrequencyNameAndNotificationTimeAndIsActiveTrue(
            String frequencyName,
            LocalTime notificationTime
    );

    List<Notification> findByFrequency_FrequencyNameAndNotificationWeekAndNotificationTimeAndIsActiveTrue(
            String frequencyName,
            Integer notificationWeek,
            LocalTime notificationTime
    );

    List<Notification> findByFrequency_FrequencyNameAndNotificationDayAndNotificationTimeAndIsActiveTrue(
            String frequencyName,
            Integer notificationDay,
            LocalTime notificationTime
    );

    void deleteByTipInterest_TipInterestId(Long tipInterestId);
}

通知を1つの tip_interest に複数設定できる仕様にするなら、

Optional<Notification> findByTipInterest_TipInterestId(...)

ではなく、

List<Notification> findByTipInterest_TipInterestId(Long tipInterestId);

にした方がいい。

今回はER図的には notification_id が別であり、tip_interest_id にuniqueはなさそうだから、本当は複数通知も許せる。
でも画面仕様が「1つの気になるTipにつき通知設定1つ」なら Optional でOK。
