package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.data.web.PageableDefault;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import simplex.pj148.skippy.server.controller.dto.response.ContentShowResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.ContentService;
import simplex.pj148.skippy.server.service.LikeService;
import simplex.pj148.skippy.server.service.UserInfoService;

@RestController
@RequiredArgsConstructor
@RequestMapping("/api/users")
public class UserContentController {

    private final ContentService contentService;
    private final UserInfoService userInfoService;
    private final LikeService likeService;

    // 任意のユーザーが作ったコンテンツを一覧で返すとき用
    @GetMapping("/{userId}/contents")
    public Page<ContentShowResponseDto> listContentsByUser(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long userId,
            @PageableDefault(
                    size = 20,
                    sort = "createdTime",
                    direction = Sort.Direction.DESC
            )
            Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return contentService.findPublicContentByUser(
                userId,
                loginUser.getUserId(),
                pageable
        );
    }

    // 任意のユーザーのいいねしたコンテンツを一覧で返すとき用
    @GetMapping("/{userId}/likedContents")
    public Page<ContentShowResponseDto> listLikedContentsByUser(
            @AuthenticationPrincipal Jwt jwt,
            @PathVariable Long userId,
            @PageableDefault(
                    size = 20,
                    sort = "createdTime",
                    direction = Sort.Direction.DESC
            )
            Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return contentService.findLikedContentsByUser(
                userId,
                loginUser,
                pageable
        );
    }
}

package simplex.pj148.skippy.server.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import org.springframework.data.repository.query.Param;
import simplex.pj148.skippy.server.entity.Content;

public interface ContentRepository extends JpaRepository<Content, Long> {

    Page<Content> findByUser_userIdAndPrivateFlgFalseAndDeleteFlgFalse(
            Long userId,
            Pageable pageable
    );

    @Query("""
            SELECT c
            FROM Like l
            JOIN l.content c
            WHERE l.user.userId = :userId
              AND c.privateFlg = false
              AND c.deleteFlg = false
            ORDER BY l.createdTime DESC
            """)
    Page<Content> findLikedContentsByUserId(
            @Param("userId") Long userId,
            Pageable pageable
    );
}

今はクライアントがこれしか呼んでない。

/api/users/${data.userId}/contents

だから、サーバー側の @PageableDefault(size = 20) により、常に

page=0
size=20

扱いになる。

上のコードではこうなる。

/api/users/${userId}/contents?page=${page}&size=20

だから、Paginationの2ページ目を押すと、

/api/users/1/contents?page=1&size=20

が呼ばれる。
