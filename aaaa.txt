3. CirculationRecordRepository.java に追加
一番下に追加。




public Optional<CollectionItem> findOne(Integer circulationId) {
    List<CollectionItem> list = jdbcTemplate.query(
            """
            select
                cr.id as circulation_id,
                c.serial_number,
                pt.title,
                m.name as borrower_name,
                case
                    when cr.actual_return_date is null then true
                    else false
                end as is_on_loan
            from circulation_record cr
            join collection c on cr.collection_id = c.id
            join published_title pt on c.published_title_id = pt.id
            join member m on cr.member_id = m.id
            where cr.id = ?
            """,
            (rs, rowNum) -> new CollectionItem(
                    rs.getInt("circulation_id"),
                    rs.getInt("serial_number"),
                    rs.getString("title"),
                    rs.getString("borrower_name"),
                    rs.getBoolean("is_on_loan")
            ),
            circulationId
    );

    return list.stream().findFirst();
}

public boolean isAvailable(Integer collectionId) {
    Boolean available = jdbcTemplate.queryForObject(
            """
            select not exists(
                select 1
                from circulation_record
                where collection_id = ?
                  and actual_return_date is null
            )
            """,
            Boolean.class,
            collectionId
    );

    return Boolean.TRUE.equals(available);
}
