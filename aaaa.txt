package simplex.pj148.skippy.server.controller;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.domain.Sort;
import org.springframework.data.web.PageableDefault;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.*;
import simplex.pj148.skippy.server.controller.dto.create.NeedDto;
import simplex.pj148.skippy.server.controller.dto.response.NeedResponseDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.NeedService;
import simplex.pj148.skippy.server.service.UserInfoService;

@RequestMapping("/api/needs")
@RequiredArgsConstructor
@RestController
public class NeedsController {

    private final NeedService needService;
    private final UserInfoService userInfoService;

    @GetMapping("")
    public Page<NeedResponseDto> getAllNeedsInfo(
            @AuthenticationPrincipal Jwt jwt,
            @RequestParam(defaultValue = "created_desc") String orderBy,
            @PageableDefault(
                    size = 20,
                    sort = "createdTime",
                    direction = Sort.Direction.DESC
            )
            Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);
        return needService.findAllNeed(loginUser, pageable, orderBy);
    }

    @PostMapping("")
    @ResponseStatus(HttpStatus.CREATED)
    public void createNeed(
            @RequestBody @Valid NeedDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        Long userId = userInfoService.getUserFromJwt(jwt).getUserId();
        needService.createNewNeed(input, userId);
    }

    @PutMapping("/{needsId}")
    public ResponseEntity<Void> updateNeed(
            @PathVariable Long needsId,
            @RequestBody @Valid NeedDto input
    ) {
        needService.updateNeeds(needsId, input);
        return ResponseEntity.ok().build();
    }

    @PutMapping("/{needsId}/delete")
    public ResponseEntity<Void> deleteNeed(
            @PathVariable Long needsId
    ) {
        needService.deleteNeed(needsId);
        return ResponseEntity.ok().build();
    }
}

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.create.NeedDto;
import simplex.pj148.skippy.server.controller.dto.response.NeedResponseDto;
import simplex.pj148.skippy.server.entity.Need;
import simplex.pj148.skippy.server.entity.NeedsTag;
import simplex.pj148.skippy.server.entity.Tag;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.exception.EntityNotFoundException;
import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.NeedsTagRepository;
import simplex.pj148.skippy.server.repository.TagRepository;
import simplex.pj148.skippy.server.repository.UserRepository;
import simplex.pj148.skippy.server.repository.WantRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
public class NeedService {

    private final NeedRepository needsRepository;
    private final NeedsTagRepository needsTagRepository;
    private final UserRepository userRepository;
    private final TagRepository tagRepository;
    private final WantRepository wantRepository;

    // findAllNeed内で使うタグ取得
    private List<String> makeTagListFromNeed(Need need) {
        Long needsId = need.getNeedsId();

        return needsTagRepository.findByNeed_needsId(needsId)
                .stream()
                .map(needsTag -> needsTag.getTag().getTagName())
                .toList();
    }

    // GET ニーズ一覧
    public Page<NeedResponseDto> findAllNeed(
            User loginUser,
            Pageable pageable,
            String orderBy
    ) {
        Page<Need> needPage;

        if ("want_desc".equals(orderBy)) {
            needPage = needsRepository.findSelectedNeedsSortedWantCount(pageable);
        } else {
            needPage = needsRepository.findByDeleteFlgFalse(pageable);
        }

        return toNeedResponseDtoPage(needPage, loginUser);
    }

    // Page<Need> を Page<NeedResponseDto> に変える関数
    public Page<NeedResponseDto> toNeedResponseDtoPage(
            Page<Need> needPage,
            User loginUser
    ) {
        return needPage.map(need -> {
            Long needsId = need.getNeedsId();

            boolean editFlg = loginUser.equals(need.getUser());

            boolean wantedByCurrentUser =
                    wantRepository.existsByNeed_needsIdAndUser_userId(
                            needsId,
                            loginUser.getUserId()
                    );

            Long wantCount =
                    wantRepository.countByNeed_needsId(needsId);

            return NeedResponseDto.builder()
                    .needsId(needsId)
                    .needsBody(need.getNeedsBody())
                    .userId(need.getUser().getUserId())
                    .userName(need.getUser().getName())
                    .editFlg(editFlg)
                    .wantedByCurrentUser(wantedByCurrentUser)
                    .wantCount(wantCount)
                    .tagList(makeTagListFromNeed(need))
                    .build();
        });
    }

    // POST ニーズ作成
    @Transactional
    public void createNewNeed(NeedDto input, Long userId) {
        User user = userRepository.findByUserId(userId)
                .orElseThrow(() -> new EntityNotFoundException("ユーザーDBに存在しないユーザー"));

        Need need = Need.builder()
                .needsBody(input.getNeedsBody())
                .user(user)
                .deleteFlg(false)
                .build();

        needsRepository.save(need);

        for (Long tagId : input.getTagIdList()) {
            Tag tag = tagRepository.findByTagId(tagId)
                    .orElseThrow(() -> new EntityNotFoundException("タグDBに存在しないタグ"));

            NeedsTag needsTag = NeedsTag.builder()
                    .need(need)
                    .tag(tag)
                    .build();

            needsTagRepository.save(needsTag);
        }
    }

    // ニーズ更新
    @Transactional
    public void updateNeeds(Long needsId, NeedDto input) {
        Need need = needsRepository.findByNeedsId(needsId)
                .orElseThrow(() -> new RuntimeException("投稿が見つかりません"));

        List<Tag> inputTags = input.getTagIdList()
                .stream()
                .map(tagId -> tagRepository.findByTagId(tagId)
                        .orElseThrow(() -> new RuntimeException("タグが見つかりません：" + tagId)))
                .toList();

        need.setNeedsBody(input.getNeedsBody());
        needsRepository.save(need);

        List<NeedsTag> nowNeedsTags =
                needsTagRepository.findByNeed_needsId(need.getNeedsId());

        List<Long> nowNeedsTagIds = nowNeedsTags
                .stream()
                .map(needsTag -> needsTag.getTag().getTagId())
                .toList();

        for (NeedsTag nowNeedsTag : nowNeedsTags) {
            if (!input.getTagIdList().contains(nowNeedsTag.getTag().getTagId())) {
                needsTagRepository.delete(nowNeedsTag);
            }
        }

        for (Long inputTagId : input.getTagIdList()) {
            if (!nowNeedsTagIds.contains(inputTagId)) {
                NeedsTag needsTag = NeedsTag.builder()
                        .need(need)
                        .tag(tagRepository.getReferenceById(inputTagId))
                        .build();

                needsTagRepository.save(needsTag);
            }
        }
    }

    // ニーズ削除
    @Transactional
    public void deleteNeed(Long needsId) {
        Need need = needsRepository.findByNeedsId(needsId)
                .orElseThrow(() -> new RuntimeException("投稿が見つかりません"));

        need.setDeleteFlg(true);
        needsRepository.save(need);
    }
}

package simplex.pj148.skippy.server.repository;

import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.data.jpa.repository.JpaRepository;
import org.springframework.data.jpa.repository.Query;
import simplex.pj148.skippy.server.entity.Need;

import java.util.Optional;

public interface NeedRepository extends JpaRepository<Need, Long> {

    Optional<Need> findByNeedsId(Long needsId);

    Page<Need> findByDeleteFlgFalse(Pageable pageable);

    @Query("""
            SELECT n
            FROM Need n
            LEFT JOIN Want w ON w.need = n
            WHERE n.deleteFlg = false
            GROUP BY n
            ORDER BY COUNT(w) DESC, n.createdTime DESC
            """)
    Page<Need> findSelectedNeedsSortedWantCount(Pageable pageable);
}

修正後のAPI
GET /api/needs?page=0&size=20

orderBy を残すなら、

GET /api/needs?page=0&size=20&orderBy=created_desc

今の NeedList.tsx は、api.fetch("/api/needs?orderBy=${sortBy}") で取得して、返ってきた data をそのまま setNeedList(data) している。つまり 配列が返ってくる前提。
でもサーバー側を Page<NeedResponseDto> に変えると、返却は配列ではなく、

{
  content: Need[];
  totalPages: number;
  totalElements: number;
  number: number;
  size: number;
}

みたいな形になる。

だから client 側は、

setNeedList(data);

ではなく、

setNeedList(data.content);
setTotalPage(data.totalPages);

に変える。

Home.tsx はすでに page, totalPage, handlePageChange, Pagination を持っているので、この形に寄せればいい。
