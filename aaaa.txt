@Transactional(readOnly = true)
public Page<ContentShowResponseDto> findAllPublicContent(
        User loginUser,
        Pageable pageable,
        String orderBy,
        Long categoryId
) {
    Specification<Content> specification = (root, query, criteriaBuilder) -> {
        List<Predicate> predicates = new ArrayList<>();

        // 削除済みコンテンツは常に除外する
        predicates.add(criteriaBuilder.equal(root.get("deleteFlg"), false));

        // カテゴリ指定がある場合だけカテゴリ条件を追加する
        if (categoryId != null) {
            predicates.add(
                    criteriaBuilder.equal(
                            root.get("category").get("categoryId"),
                            categoryId
                    )
            );
        }

        return criteriaBuilder.and(predicates.toArray(new Predicate[0]));
    };

    Page<Content> contents = contentRepository.findAll(specification, pageable);

    return contents.map(content -> toContentShowResponseDto(content, loginUser));
}
