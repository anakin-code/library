
function layoutNodesAndEdges(nodes: Node[], edges: Edge[]): {
  nodes: Node[];
  edges: Edge[];
} {
  const centerX = 360;
  const centerY = 260;
  const radius = 180;

  const scale = 0.75;
  const offsetX = 100;
  const offsetY = 80;

  const layoutedNodes = nodes.map((node, index) => {
    const angle = (2 * Math.PI * index) / nodes.length - Math.PI / 2;

    const width =
      typeof node.style?.width === "number"
        ? node.style.width
        : 120;

    const height =
      typeof node.style?.height === "number"
        ? node.style.height
        : 120;

    const x = centerX + radius * Math.cos(angle) - width / 2;
    const y = centerY + radius * Math.sin(angle) - height / 2;

    return {
      ...node,
      position: {
        x: x * scale + offsetX,
        y: y * scale + offsetY,
      },
      sourcePosition: Position.Right,
      targetPosition: Position.Left,
    };
  });

  return {
    nodes: layoutedNodes,
    edges,
  };
}
