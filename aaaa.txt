private int mergedMonths(List<CareerSkill> careerSkills) {
    List<CareerPeriod> periods = careerSkills.stream()
            .map(CareerSkill::getCareer)
            .filter(career -> career.getStartTime() != null)
            .map(career -> new CareerPeriod(
                    career.getStartTime(),
                    career.getEndTime() == null
                            ? LocalDate.now()
                            : career.getEndTime()
            ))
            .sorted(Comparator.comparing(CareerPeriod::startTime))
            .toList();

    if (periods.isEmpty()) {
        return 0;
    }

    List<CareerPeriod> mergedPeriods = new ArrayList<>();

    LocalDate currentStart = periods.get(0).startTime();
    LocalDate currentEnd = periods.get(0).endTime();

    for (int i = 1; i < periods.size(); i++) {
        CareerPeriod next = periods.get(i);

        if (!next.startTime().isAfter(currentEnd.plusDays(1))) {
            if (next.endTime().isAfter(currentEnd)) {
                currentEnd = next.endTime();
            }
        } else {
            mergedPeriods.add(new CareerPeriod(currentStart, currentEnd));
            currentStart = next.startTime();
            currentEnd = next.endTime();
        }
    }

    mergedPeriods.add(new CareerPeriod(currentStart, currentEnd));

    int totalMonths = 0;

    for (CareerPeriod period : mergedPeriods) {
        totalMonths += countMonths(
                period.startTime(),
                period.endTime()
        );
    }

    return totalMonths;
}

private int countMonths(LocalDate start, LocalDate end) {
    Period period = Period.between(start, end.plusDays(1));

    return period.getYears() * 12 + period.getMonths();
}

private record CareerPeriod(
        LocalDate startTime,
        LocalDate endTime
) {
}

4. getPersonalSkillNetwork の修正版
public PersonalSkillGraphResponse getPersonalSkillNetwork(String hrid) {
    List<Career> careers = careerRepository.findByProfileHrid(hrid);

    if (careers.isEmpty()) {
        profileRepository.findById(hrid)
                .orElseThrow(() ->
                        new IllegalArgumentException("社員が見つかりません: " + hrid)
                );
    }

    List<CareerSkill> careerSkills = getCareerSkills(careers);

    Map<Long, String> skillNameMap = new HashMap<>();
    Map<Long, Integer> totalMonthsMap = new HashMap<>();
    Map<String, Integer> edgeMonthsMap = new HashMap<>();

    Map<Long, List<CareerSkill>> bySkillId =
            careerSkills.stream()
                    .collect(Collectors.groupingBy(
                            cs -> (long) cs.getSkill().getSkillId()
                    ));

    for (Map.Entry<Long, List<CareerSkill>> entry : bySkillId.entrySet()) {
        Long skillId = entry.getKey();
        List<CareerSkill> list = entry.getValue();

        skillNameMap.put(
                skillId,
                list.get(0).getSkill().getSkillName()
        );

        totalMonthsMap.put(
                skillId,
                mergedMonths(list)
        );
    }

    Map<Long, List<CareerSkill>> byCareerId =
            careerSkills.stream()
                    .collect(Collectors.groupingBy(
                            cs -> cs.getCareer().getCareerId()
                    ));

    for (List<CareerSkill> list : byCareerId.values()) {
        for (int i = 0; i < list.size(); i++) {
            for (int j = i + 1; j < list.size(); j++) {
                CareerSkill a = list.get(i);
                CareerSkill b = list.get(j);

                Long idA = (long) a.getSkill().getSkillId();
                Long idB = (long) b.getSkill().getSkillId();

                if (idA.equals(idB)) {
                    continue;
                }

                Long source = Math.min(idA, idB);
                Long target = Math.max(idA, idB);

                int coMonths = months(a);

                edgeMonthsMap.merge(
                        source + "-" + target,
                        coMonths,
                        Integer::sum
                );
            }
        }
    }

    int maxMonths = totalMonthsMap.values()
            .stream()
            .max(Integer::compareTo)
            .orElse(1);

    List<PersonalSkillNodeResponse> nodes =
            totalMonthsMap.entrySet()
                    .stream()
                    .map(entry -> new PersonalSkillNodeResponse(
                            entry.getKey(),
                            skillNameMap.get(entry.getKey()),
                            entry.getValue(),
                            maxMonths == 0
                                    ? 0.0
                                    : (double) entry.getValue() / maxMonths
                    ))
                    .toList();

    List<PersonalSkillEdgeResponse> edges =
            edgeMonthsMap.entrySet()
                    .stream()
                    .map(entry -> {
                        String[] ids = entry.getKey().split("-");
                        Long source = Long.valueOf(ids[0]);
                        Long target = Long.valueOf(ids[1]);

                        int totalMonths = entry.getValue();

                        int denominator = Math.min(
                                totalMonthsMap.getOrDefault(source, 0),
                                totalMonthsMap.getOrDefault(target, 0)
                        );

                        double normalizedWeight =
                                denominator == 0
                                        ? 0.0
                                        : (double) totalMonths / denominator;

                        return new PersonalSkillEdgeResponse(
                                source,
                                target,
                                totalMonths,
                                normalizedWeight
                        );
                    })
                    .toList();

    return new PersonalSkillGraphResponse(nodes, edges);
}
