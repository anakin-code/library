
PersonalSkillNetwork.tsx はこれに丸ごと差し替え。

import { useEffect, useMemo, useState } from "react";
import ReactFlow, {
  Background,
  Controls,
  type Edge,
  type Node,
} from "reactflow";
import "reactflow/dist/style.css";
import api from "../../api/api";

type SkillNode = {
  id: number;
  label: string;
  months: number;
  normalizedSize: number;
};

type SkillEdge = {
  source: number;
  target: number;
  months: number;
  normalizedWeight: number;
};

type SkillNetworkResponse = {
  nodes: SkillNode[];
  edges: SkillEdge[];
};

type Props = {
  hrid: string;
};

export default function PersonalSkillNetwork({ hrid }: Props) {
  const [data, setData] = useState<SkillNetworkResponse>({
    nodes: [],
    edges: [],
  });

  useEffect(() => {
    if (!hrid) {
      return;
    }

    api.get<SkillNetworkResponse>(`/user/${hrid}/skill-network`)
      .then((response) => {
        setData(response.data);
      })
      .catch((error) => {
        console.error(error);
        setData({
          nodes: [],
          edges: [],
        });
      });
  }, [hrid]);

  const nodes: Node[] = useMemo(() => {
    if (data.nodes.length === 0) {
      return [];
    }

    const centerX = 420;
    const centerY = 230;
    const radius = 170;

    return data.nodes.map((skill, index) => {
      const angle = (2 * Math.PI * index) / data.nodes.length;
      const size = 65 + Math.sqrt(skill.months) * 7;

      return {
        id: String(skill.id),
        position: {
          x: centerX + Math.cos(angle) * radius,
          y: centerY + Math.sin(angle) * radius,
        },
        data: {
          label: `${skill.label}\n${skill.months}ヶ月`,
        },
        style: {
          width: size,
          height: size,
          background: "#d7f3e3",
          border: "2px solid #56b68b",
          borderRadius: "50%",
          color: "#222",
          display: "flex",
          alignItems: "center",
          justifyContent: "center",
          textAlign: "center",
          fontSize: 12,
          fontWeight: 600,
          boxShadow: "0 2px 8px rgba(0,0,0,0.15)",
          whiteSpace: "pre-line",
        },
      };
    });
  }, [data]);

  const edges: Edge[] = useMemo(() => {
    return data.edges.map((edge) => ({
      id: `${edge.source}-${edge.target}`,
      source: String(edge.source),
      target: String(edge.target),
      label: `${edge.months}ヶ月`,
      style: {
        stroke: "#556070",
        strokeWidth: 2 + edge.normalizedWeight * 6,
      },
      labelStyle: {
        fontSize: 11,
        fontWeight: 600,
        fill: "#333",
      },
    }));
  }, [data]);

  return (
    <div style={{ width: "100%", height: "520px" }}>
      <ReactFlow
        nodes={nodes}
        edges={edges}
        fitView
        fitViewOptions={{ padding: 0.25 }}
        minZoom={0.3}
        maxZoom={2}
      >
        <Background color="#e5e7eb" gap={24} />
        <Controls />
      </ReactFlow>
    </div>
  );
}
