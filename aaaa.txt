2. TipService.java

needId がある場合だけ NeedRepository でNeedを取得して、Tip にセットする。

package simplex.pj148.skippy.server.service;

import jakarta.persistence.EntityNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.TipsCreateDto;
import simplex.pj148.skippy.server.entity.Category;
import simplex.pj148.skippy.server.entity.Need;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CategoryRepository;
import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.TipRepository;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class TipService {

    private final TipRepository tipRepository;
    private final CategoryRepository categoryRepository;
    private final NeedRepository needRepository;

    @Transactional
    public void createTip(TipsCreateDto input, User loginUser) {

        Category category = categoryRepository.findByCategoryId(input.getCategoryId())
                .orElseThrow(() -> new EntityNotFoundException(
                        "カテゴリDBに存在しないカテゴリ：" + input.getCategoryId() + " が選択されています。"
                ));

        Need need = null;

        if (input.getNeedId() != null) {
            need = needRepository.findByNeedIdAndDeleteFlgFalse(input.getNeedId())
                    .orElseThrow(() -> new EntityNotFoundException(
                            "お悩みDBに存在しないお悩み：" + input.getNeedId() + " が選択されています。"
                    ));
        }

        LocalDateTime createdTime = LocalDateTime.now();

        Tip tip = Tip.builder()
                .user(loginUser)
                .need(need)
                .category(category)
                .tipTitle(input.getTipTitle())
                .tipDetail(input.getTipDetail())
                .tipURL(input.getTipURL())
                .createdTime(createdTime)
                .deleteFlg(false)
                .build();

        tipRepository.save(tip);
    }
}

4. Tip.java

Tip Entityに Need との紐づきが必要。

Needに紐づかない通常投稿もあるなら、optional = true にする。
DB的にも need_id は nullable にする。

@JoinColumn(name = "need_id")

つまり、nullable = false は付けない。

ダメな例：

@ManyToOne(fetch = FetchType.LAZY)
@JoinColumn(name = "need_id", nullable = false)

これだと、通常Tip投稿で needId がないと保存できない。

7. TipForm.tsx は needId を持たなくていい

ここ大事。

TipForm はただの入力フォームだから、基本は

categoryId
tipTitle
tipDetail
tipURL

だけでいい。

needId を入れる責任は、フォームではなく 呼び出し元のページ に置いた方がきれい。

つまり、

Home画面
onSubmit={(values) => {
  api.sendJson("/api/tips", values, ...);
}}
Need詳細画面
onSubmit={(values) => {
  api.sendJson("/api/tips", {
    ...values,
    needId: needId,
  }, ...);
}}

この方が自然。

結論

君が needId をDTOに入れていたのは正しい。
ただし、needId は 任意項目 にする。

設計としてはこれ。

TipsCreateDto
  needId      任意
  categoryId 必須
  tipTitle   必須
  tipDetail  必須
  tipURL     任意

Serviceではこう。

needId == null
  → 通常Tip投稿

needId != null
  → Needを取得して、Needに紐づいたTip投稿

この設計なら、Home投稿とNeed詳細投稿の両方に対応できる。

import { useState } from "react";
import type { ChangeEvent, FormEvent } from "react";
import {
  Box,
  Button,
  MenuItem,
  Modal,
  Stack,
  TextField,
  Typography,
} from "@mui/material";

import type { Category } from "../models/category";
import type { TipFormValues } from "../models/tip";

type TipFormInputValues = {
  categoryId: string | number;
  tipTitle: string;
  tipDetail: string;
  tipURL: string;
};

type TipFormProps = {
  initialValues: TipFormInputValues;
  categories: Category[];
  onSubmit: (values: TipFormValues) => void;
  submitLabel: string;
  onClose: () => void;
};

function TipForm({
  initialValues,
  categories,
  onSubmit,
  submitLabel,
  onClose,
}: TipFormProps) {
  const [formData, setFormData] = useState<TipFormInputValues>(initialValues);

  const handleChange = (
    event: ChangeEvent<HTMLInputElement | HTMLTextAreaElement>
  ): void => {
    const { name, value } = event.target;

    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  const handleSubmit = (event: FormEvent<HTMLFormElement>): void => {
    event.preventDefault();

    onSubmit({
      categoryId: Number(formData.categoryId),
      tipTitle: formData.tipTitle,
      tipDetail: formData.tipDetail,
      tipURL: formData.tipURL,
    });
  };

  const selectableCategories = categories.filter(
    (category) => category.categoryId !== 0
  );

  return (
    <Modal open onClose={onClose}>
      <Box
        component="form"
        onSubmit={handleSubmit}
        sx={{
          position: "absolute",
          top: "50%",
          left: "50%",
          transform: "translate(-50%, -50%)",
          width: {
            xs: "90%",
            sm: 520,
          },
          bgcolor: "background.paper",
          borderRadius: 3,
          boxShadow: 24,
          p: 4,
        }}
      >
        <Typography
          component="h2"
          variant="h5"
          sx={{
            fontWeight: 700,
            mb: 3,
          }}
        >
          投稿フォーム
        </Typography>

        <Stack spacing={2}>
          <TextField
            required
            select
            fullWidth
            label="カテゴリ"
            name="categoryId"
            value={formData.categoryId}
            onChange={handleChange}
          >
            <MenuItem value="" disabled>
              カテゴリを選択
            </MenuItem>

            {selectableCategories.map((category) => (
              <MenuItem key={category.categoryId} value={category.categoryId}>
                {category.categoryName}
              </MenuItem>
            ))}
          </TextField>

          <TextField
            required
            fullWidth
            label="タイトル"
            name="tipTitle"
            value={formData.tipTitle}
            onChange={handleChange}
            slotProps={{
              htmlInput: {
                maxLength: 50,
              },
            }}
          />

          <TextField
            required
            fullWidth
            multiline
            minRows={4}
            label="投稿内容"
            name="tipDetail"
            value={formData.tipDetail}
            onChange={handleChange}
            slotProps={{
              htmlInput: {
                maxLength: 500,
              },
            }}
          />

          <TextField
            fullWidth
            label="URL"
            name="tipURL"
            value={formData.tipURL}
            onChange={handleChange}
          />

            direction="row"
            spacing={2}
            sx={{
              justifyContent: "flex-end",

            }}
          >
            <Button variant="outlined" onClick={onClose}>
              キャンセル
            </Button>

            <Button type="submit" variant="contained">
              {submitLabel}
            </Button>
          </Stack>
        </Stack>
      </Box>
    </Modal>
  );
}

3. Home.tsx 側

Home画面では、Needに紐づかない投稿だから needId は付けない。

import { useState, useEffect } from "react";
import { Container, Box, Typography, Fab } from "@mui/material";
import AddIcon from "@mui/icons-material/Add";

import TipForm from "../components/TipForm";
import type { Category } from "../models/category";
import type { TipFormValues, TipCreateRequest } from "../models/tip";
import { useApi } from "../hooks/useApi";

function Home() {
  const [categories, setCategories] = useState<Category[]>([]);
  const [isPostFormOpen, setIsPostFormOpen] = useState(false);

  const api = useApi();

  useEffect((): void => {
    api
      .fetch("/api/categories")
      .then((res: Response) => res.json())
      .then((categoryList: Category[]): void => {
        setCategories(categoryList);
      })
      .catch((error: unknown): void => {
        console.error(error);
      });
  }, [api]);

  const handleSubmit = (values: TipFormValues): void => {
    const requestBody: TipCreateRequest = {
      categoryId: values.categoryId,
      tipTitle: values.tipTitle,
      tipDetail: values.tipDetail,
      tipURL: values.tipURL,
    };

    api
      .sendJson("/api/tips", requestBody, {
        method: "POST",
        headers: {
          "Content-type": "application/json",
        },
      })
      .then((res: Response): void => {
        if (!res.ok) {
          throw new Error("投稿に失敗しました");
        }

        setIsPostFormOpen(false);
      })
      .catch((error: unknown): void => {
        console.error(error);
        alert("投稿に失敗しました。");
      });
  };

  return (
    <Container maxWidth={false}>
      <Box>
        <Typography
          component="h1"
          variant="h3"
          sx={{
            color: "black",
            fontWeight: 800,
            textAlign: "center",
            mb: 5,
          }}
        >
          投稿
        </Typography>

        <Fab
          color="primary"
          aria-label="add"
          onClick={() => setIsPostFormOpen(true)}
          sx={{
            position: "fixed",
            right: 32,
            bottom: 32,
          }}
        >
          <AddIcon />
        </Fab>

        {isPostFormOpen && (
          <TipForm
            initialValues={{
              categoryId: "",
              tipTitle: "",
              tipDetail: "",
              tipURL: "",
            }}
            categories={categories}
            onSubmit={handleSubmit}
            submitLabel="投稿する"
            onClose={() => setIsPostFormOpen(false)}
          />
        )}
      </Box>
    </Container>
  );
}

export default Home;
