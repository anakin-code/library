package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.stereotype.Service;
import simplex.pj148.skippy.server.controller.dto.response.TipShowResponseDto;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.TipInterest;
import simplex.pj148.skippy.server.entity.TipsTag;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.TipInterestRepository;
import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.TipsTagRepository;

import java.util.HashMap;
import java.util.HashSet;
import java.util.List;
import java.util.Map;
import java.util.Set;

@Service
@RequiredArgsConstructor
public class TipService {

    private final TipRepository tipRepository;
    private final TipInterestRepository tipInterestRepository;
    private final TipsTagRepository tipsTagRepository;

    public Page<TipShowResponseDto> findAllTips(
            User loginUser,
            Pageable pageable,
            String orderBy,
            Long categoryId
    ) {
        Page<Tip> tipPage;

        if ("interestCount".equals(orderBy)) {
            tipPage = tipRepository.findSelectedTipsSortedByInterestCount(pageable, categoryId);
        } else {
            tipPage = tipRepository.findSelectedTipsSortedByCreatedTime(pageable, categoryId);
        }

        return toTipShowResponseDtoPage(tipPage, loginUser);
    }

    public Page<TipShowResponseDto> searchTips(
            String keyword,
            String context,
            User loginUser,
            Pageable pageable
    ) {
        if (keyword == null || keyword.trim().isEmpty()) {
            return findAllTips(loginUser, pageable, "createdTime", 0L);
        }

        Page<Tip> tipPage = tipRepository.searchTips(
                keyword.trim(),
                context,
                pageable
        );

        return toTipShowResponseDtoPage(tipPage, loginUser);
    }

    public Page<TipShowResponseDto> findTipsByUser(
            Long userId,
            User loginUser,
            Pageable pageable
    ) {
        Page<Tip> tipPage = tipRepository.findByUser_UserIdOrderByCreatedTimeDesc(
                userId,
                pageable
        );

        return toTipShowResponseDtoPage(tipPage, loginUser);
    }

    private Page<TipShowResponseDto> toTipShowResponseDtoPage(
            Page<Tip> tipPage,
            User loginUser
    ) {
        List<Long> tipIds = tipPage.getContent()
                .stream()
                .map(Tip::getTipId)
                .toList();

        if (tipIds.isEmpty()) {
            return tipPage.map(tip -> toTipShowResponseDto(tip, false, 0L));
        }

        List<TipInterest> userInterests =
                tipInterestRepository.findByUser_UserIdAndTip_TipIdIn(
                        loginUser.getUserId(),
                        tipIds
                );

        Set<Long> interestedTipIdSet = new HashSet<>(
                userInterests.stream()
                        .map(tipInterest -> tipInterest.getTip().getTipId())
                        .toList()
        );

        List<Object[]> interestCounts =
                tipInterestRepository.countInterestsByTipIds(tipIds);

        Map<Long, Long> interestCountMap = new HashMap<>();

        for (Object[] row : interestCounts) {
            Long tipId = (Long) row[0];
            Long count = (Long) row[1];
            interestCountMap.put(tipId, count);
        }

        return tipPage.map(tip -> {
            boolean interested = interestedTipIdSet.contains(tip.getTipId());
            Long interestCount = interestCountMap.getOrDefault(tip.getTipId(), 0L);

            return toTipShowResponseDto(tip, interested, interestCount);
        });
    }

    private TipShowResponseDto toTipShowResponseDto(
            Tip tip,
            boolean interested,
            Long interestCount
    ) {
        List<TipsTag> tipsTags = tipsTagRepository.findByTip_TipId(tip.getTipId());

        List<String> tagNameList = tipsTags.stream()
                .map(tipsTag -> tipsTag.getTag().getTagName())
                .toList();

        return TipShowResponseDto.builder()
                .tipId(tip.getTipId())
                .userId(tip.getUser().getUserId())
                .userName(tip.getUser().getUserName())
                .tipTitle(tip.getTipTitle())
                .tipUrl(tip.getTipUrl())
                .tipDetail(tip.getTipDetail())
                .categoryId(tip.getCategory().getCategoryId())
                .categoryName(tip.getCategory().getCategoryName())
                .tagNameList(tagNameList)
                .interestedByCurrentUser(interested)
                .interestCount(interestCount)
                .createdTime(tip.getCreatedTime())
                .updatedTime(tip.getUpdatedTime())
                .build();
    }
}

    private final TipRepository tipRepository;
    private final TipsTagRepository tipsTagRepository;
    private final TagRepository tagRepository;
    private final CategoryRepository categoryRepository;
    private final NeedRepository needRepository;
    private final UserRepository userRepository;

    @Transactional
    public void registerNewTip(TipDto input) {
        User user = userRepository.findByUserId(input.getUserId())
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Category category = categoryRepository.findByCategoryId(input.getCategoryId())

                    .orElseThrow(() -> new RuntimeException("ニーズが見つかりません"));

                .user(user)
                .tipTitle(input.getTipTitle())
                .tipUrl(input.getTipUrl())
                .tipDetail(input.getTipDetail())
                .category(category)
                .need(need)
                .build();

        tipRepository.save(tip);

        for (Long tagId : input.getTagIdList()) {
            Tag tag = tagRepository.findByTagId(tagId)
                    .orElseThrow(() -> new RuntimeException("タグが見つかりません"));

            TipsTag tipsTag = TipsTag.builder()
                    .tip(tip)
                    .tag(tag)
                    .build();

            tipsTagRepository.save(tipsTag);
        }
    }

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;

import java.util.List;
import java.util.Optional;

@Service
@RequiredArgsConstructor
public class TipDetailService {

    private final TipRepository tipRepository;
    private final TipsTagRepository tipsTagRepository;
    private final CommentRepository commentRepository;
    private final TipInterestRepository tipInterestRepository;
    private final NotificationRepository notificationRepository;

    public TipDetailResponseDto getTipDetail(
            User loginUser,
            Long tipId
    ) {
        Tip tip = tipRepository.findByTipId(tipId)
                .orElseThrow(() -> new RuntimeException("Tipが見つかりません"));

        boolean editFlg = loginUser.getUserId().equals(tip.getUser().getUserId());

        boolean interestedByCurrentUser =
                tipInterestRepository.existsByUser_UserIdAndTip_TipId(
                        loginUser.getUserId(),
                        tipId
                );

        Long interestCount = tipInterestRepository.countByTip_TipId(tipId);

        boolean mybestByCurrentUser =
                tipInterestRepository
                        .findByUser_UserIdAndTip_TipIdAndIsRegisterMybestTrue(
                                loginUser.getUserId(),
                                tipId
                        )
                        .isPresent();

        Long mybestCount =
                tipInterestRepository.countByTip_TipIdAndIsRegisterMybestTrue(tipId);

        List<String> tagNameList = tipsTagRepository.findByTip_TipId(tipId)
                .stream()
                .map(tipsTag -> tipsTag.getTag().getTagName())
                .toList();

        List<CommentShowResponseDto> comments =
                commentRepository.findByTip_TipIdOrderByCreatedTimeAsc(tipId)
                        .stream()
                        .map(this::toCommentShowResponseDto)
                        .toList();

        Optional<Notification> notification =
                tipInterestRepository.findByUser_UserIdAndTip_TipId(
                                loginUser.getUserId(),
                                tipId
                        )
                        .flatMap(tipInterest ->
                                notificationRepository.findByTipInterest_TipInterestId(
                                        tipInterest.getTipInterestId()
                                )
                        );

        return TipDetailResponseDto.builder()
                .tipId(tip.getTipId())
                .userId(tip.getUser().getUserId())
                .userName(tip.getUser().getUserName())
                .tipTitle(tip.getTipTitle())
                .tipUrl(tip.getTipUrl())
                .tipDetail(tip.getTipDetail())
                .categoryId(tip.getCategory().getCategoryId())
                .categoryName(tip.getCategory().getCategoryName())
                .needId(tip.getNeed() == null ? null : tip.getNeed().getNeedsId())
                .tagNameList(tagNameList)
                .editFlg(editFlg)
                .interestedByCurrentUser(interestedByCurrentUser)
                .interestCount(interestCount)
                .mybestByCurrentUser(mybestByCurrentUser)
                .mybestCount(mybestCount)
                .comments(comments)
                .hasNotification(notification.isPresent())
                .createdTime(tip.getCreatedTime())
                .updatedTime(tip.getUpdatedTime())
                .build();
    }

    private CommentShowResponseDto toCommentShowResponseDto(Comment comment) {
        return CommentShowResponseDto.builder()
                .commentId(comment.getCommentId())
                .tipId(comment.getTip().getTipId())
                .userId(comment.getUser().getUserId())
                .userName(comment.getUser().getUserName())
                .commentBody(comment.getCommentBody())
                .createdTime(comment.getCreatedTime())
                .build();
    }
}

import lombok.RequiredArgsConstructor;

import simplex.pj148.skippy.server.repository.TipsTagRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
public class TipUpdateService {

    private final TipRepository tipRepository;
    private final CategoryRepository categoryRepository;
    private final NeedRepository needRepository;
    private final TagRepository tagRepository;
    private final TipsTagRepository tipsTagRepository;

    public TipDto getUpdateTip(Long tipId) {
        Tip tip = tipRepository.findByTipId(tipId)
                .orElseThrow(() -> new RuntimeException("Tipが見つかりません"));

        List<Long> tagIdList = tipsTagRepository.findByTip_TipId(tipId)
                .stream()
                .map(tipsTag -> tipsTag.getTag().getTagId())
                .toList();

        return TipDto.builder()
                .userId(tip.getUser().getUserId())
                .tipTitle(tip.getTipTitle())
                .tipUrl(tip.getTipUrl())
                .tipDetail(tip.getTipDetail())
                .categoryId(tip.getCategory().getCategoryId())
                .needId(tip.getNeed() == null ? null : tip.getNeed().getNeedsId())
                .tagIdList(tagIdList)
                .build();
    }

    @Transactional
    public void updateTip(
            Long tipId,
            TipDto input
    ) {
        Tip tip = tipRepository.findByTipId(tipId)
                .orElseThrow(() -> new RuntimeException("Tipが見つかりません"));

        Category category = categoryRepository.findByCategoryId(input.getCategoryId())
                .orElseThrow(() -> new RuntimeException("カテゴリが見つかりません"));

        Need need = null;

        if (input.getNeedId() != null) {
            need = needRepository.findByNeedsId(input.getNeedId())
                    .orElseThrow(() -> new RuntimeException("ニーズが見つかりません"));
        }

        tip.setTipTitle(input.getTipTitle());
        tip.setTipUrl(input.getTipUrl());
        tip.setTipDetail(input.getTipDetail());
        tip.setCategory(category);
        tip.setNeed(need);

        tipRepository.save(tip);

        updateTipTags(tip, input.getTagIdList());
    }

    private void updateTipTags(
            Tip tip,
            List<Long> inputTagIdList
    ) {
        List<TipsTag> currentTipsTags =
                tipsTagRepository.findByTip_TipId(tip.getTipId());

        List<Long> currentTagIdList = currentTipsTags.stream()
                .map(tipsTag -> tipsTag.getTag().getTagId())
                .toList();

        for (TipsTag currentTipsTag : currentTipsTags) {
            Long currentTagId = currentTipsTag.getTag().getTagId();

            if (!inputTagIdList.contains(currentTagId)) {
                tipsTagRepository.delete(currentTipsTag);
            }
        }

        for (Long inputTagId : inputTagIdList) {
            if (!currentTagIdList.contains(inputTagId)) {
                Tag tag = tagRepository.findByTagId(inputTagId)
                        .orElseThrow(() -> new RuntimeException("タグが見つかりません"));

                TipsTag tipsTag = TipsTag.builder()
                        .tip(tip)
                        .tag(tag)
                        .build();

                tipsTagRepository.save(tipsTag);
            }
        }
    }
}

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.entity.TipInterest;
import simplex.pj148.skippy.server.repository.CommentRepository;

import simplex.pj148.skippy.server.repository.TipInterestRepository;
import simplex.pj148.skippy.server.repository.TipRepository;
import simplex.pj148.skippy.server.repository.TipsTagRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
public class TipDeleteService {

    private final TipRepository tipRepository;
    private final TipsTagRepository tipsTagRepository;
    private final CommentRepository commentRepository;
    private final TipInterestRepository tipInterestRepository;
    private final NotificationRepository notificationRepository;

    @Transactional
    public void deleteTip(Long tipId) {
        if (!tipRepository.existsById(tipId)) {
            throw new RuntimeException("Tipが見つかりません");
        }

        List<TipInterest> tipInterests =
                tipInterestRepository.findByTip_TipId(tipId);

        for (TipInterest tipInterest : tipInterests) {
            notificationRepository.deleteByTipInterest_TipInterestId(
                    tipInterest.getTipInterestId()
            );
        }

        tipInterestRepository.deleteByTip_TipId(tipId);

        commentRepository.deleteByTip_TipId(tipId);

        tipsTagRepository.deleteByTip_TipId(tipId);

        tipRepository.deleteById(tipId);
    }
}

@Service
@RequiredArgsConstructor
public class TipInterestService {

    private final TipInterestRepository tipInterestRepository;
    private final TipRepository tipRepository;
    private final UserRepository userRepository;
    private final NotificationRepository notificationRepository;

    @Transactional
    public void registerInterest(
            Long userId,
            Long tipId
    ) {
        if (tipInterestRepository.existsByUser_UserIdAndTip_TipId(userId, tipId)) {
            return;
        }

        User user = userRepository.findByUserId(userId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Tip tip = tipRepository.findByTipId(tipId)
                .orElseThrow(() -> new RuntimeException("Tipが見つかりません"));

        TipInterest tipInterest = TipInterest.builder()
                .user(user)
                .tip(tip)
                .isLooked(false)
                .isRegisterMybest(false)
                .build();

        tipInterestRepository.save(tipInterest);
    }

    @Transactional
    public void cancelInterest(
            Long userId,
            Long tipId
    ) {
        TipInterest tipInterest =
                tipInterestRepository.findByUser_UserIdAndTip_TipId(userId, tipId)
                        .orElseThrow(() -> new RuntimeException("気になる登録がありません"));

        notificationRepository.deleteByTipInterest_TipInterestId(
                tipInterest.getTipInterestId()
        );

        tipInterestRepository.delete(tipInterest);
    }

    @Transactional
    public void registerMybest(
            Long userId,
            Long tipId
    ) {
        TipInterest tipInterest =
                tipInterestRepository.findByUser_UserIdAndTip_TipId(userId, tipId)
                        .orElseGet(() -> createTipInterest(userId, tipId));

        tipInterest.setIsRegisterMybest(true);

        tipInterestRepository.save(tipInterest);
    }

    @Transactional
    public void cancelMybest(
            Long userId,
            Long tipId
    ) {
        TipInterest tipInterest =
                tipInterestRepository.findByUser_UserIdAndTip_TipId(userId, tipId)
                        .orElseThrow(() -> new RuntimeException("気になる登録がありません"));

        tipInterest.setIsRegisterMybest(false);

        tipInterestRepository.save(tipInterest);
    }

    private TipInterest createTipInterest(
            Long userId,
            Long tipId
    ) {
        User user = userRepository.findByUserId(userId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Tip tip = tipRepository.findByTipId(tipId)
                .orElseThrow(() -> new RuntimeException("Tipが見つかりません"));

        TipInterest tipInterest = TipInterest.builder()
                .user(user)
                .tip(tip)
                .isLooked(false)
                .isRegisterMybest(false)
                .build();

        return tipInterestRepository.save(tipInterest);
    }
}

package simplex.pj148.skippy.server.service;

    public void deleteNeed(Long needsId) {
        Need need = needRepository.findByNeedsId(needsId)
                .orElseThrow(() -> new RuntimeException("ニーズが見つかりません"));

        needEmpathyRepository.findByNeed_NeedsId(needsId)
                .forEach(needEmpathyRepository::delete);

        needRepository.delete(need);
    }

    private NeedResponseDto toNeedResponseDto(Need need) {
        Long empathyCount = needEmpathyRepository.countByNeed_NeedsId(
                need.getNeedsId()
        );

        return NeedResponseDto.builder()
                .needsId(need.getNeedsId())
                .userId(need.getUser().getUserId())
                .userName(need.getUser().getUserName())
                .needTitle(need.getNeedTitle())
                .needsDetail(need.getNeedsDetail())
                .categoryId(need.getCategory().getCategoryId())
                .categoryName(need.getCategory().getCategoryName())
                .empathyCount(empathyCount)
                .createdTime(need.getCreatedTime())
                .updatedTime(need.getUpdatedTime())
                .build();
    }
}

import simplex.pj148.skippy.server.repository.NeedRepository;
import simplex.pj148.skippy.server.repository.UserRepository;

@Service
@RequiredArgsConstructor
public class NeedEmpathyService {

    private final NeedEmpathyRepository needEmpathyRepository;
    private final NeedRepository needRepository;
    private final UserRepository userRepository;

    @Transactional
    public void registerEmpathy(
            Long userId,
            Long needsId
    ) {
        if (needEmpathyRepository.existsByUser_UserIdAndNeed_NeedsId(userId, needsId)) {
            return;
        }

        User user = userRepository.findByUserId(userId)
                .orElseThrow(() -> new RuntimeException("ユーザーが見つかりません"));

        Need need = needRepository.findByNeedsId(needsId)
                .orElseThrow(() -> new RuntimeException("ニーズが見つかりません"));

        NeedEmpathy empathy = NeedEmpathy.builder()
                .user(user)
                .need(need)
                .isLooked(false)
                .build();

        needEmpathyRepository.save(empathy);
    }

    @Transactional
    public void cancelEmpathy(
            Long userId,
            Long needsId
    ) {
        NeedEmpathy empathy =
                needEmpathyRepository.findByUser_UserIdAndNeed_NeedsId(userId, needsId)
                        .orElseThrow(() -> new RuntimeException("共感登録がありません"));

        needEmpathyRepository.delete(empathy);
    }
}
10. NotificationService

通知設定を作成・更新・削除する。

package simplex.pj148.skippy.server.service;

import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.create.NotificationDto;
import simplex.pj148.skippy.server.entity.Frequency;
import simplex.pj148.skippy.server.entity.Notification;
import simplex.pj148.skippy.server.entity.TipInterest;
import simplex.pj148.skippy.server.repository.FrequencyRepository;
import simplex.pj148.skippy.server.repository.NotificationRepository;
import simplex.pj148.skippy.server.repository.TipInterestRepository;

import java.util.List;

@Service
@RequiredArgsConstructor
public class NotificationService {

    private final NotificationRepository notificationRepository;
    private final TipInterestRepository tipInterestRepository;
    private final FrequencyRepository frequencyRepository;

    public List<Notification> findMyNotifications(Long userId) {
        return notificationRepository.findByTipInterest_User_UserId(userId);
    }

    @Transactional
    public void createNotification(NotificationDto input) {
        TipInterest tipInterest =
                tipInterestRepository.findByTipInterestId(input.getTipInterestId())
                        .orElseThrow(() -> new RuntimeException("気になる登録が見つかりません"));

        Frequency frequency =
                frequencyRepository.findByFrequencyId(input.getFrequencyId())
                        .orElseThrow(() -> new RuntimeException("通知頻度が見つかりません"));

        validateNotificationRule(
                frequency.getFrequencyName(),
                input.getNotificationWeek(),
                input.getNotificationDay()
        );

        Notification notification = Notification.builder()
                .tipInterest(tipInterest)
                .frequency(frequency)
                .notificationWeek(input.getNotificationWeek())
                .notificationDay(input.getNotificationDay())
                .notificationTime(input.getNotificationTime())
                .isActive(true)
                .build();

        notificationRepository.save(notification);
    }

    @Transactional
    public void updateNotification(
            Long notificationId,
            NotificationDto input
    ) {
        Notification notification =
                notificationRepository.findByNotificationId(notificationId)
                        .orElseThrow(() -> new RuntimeException("通知設定が見つかりません"));

        Frequency frequency =
                frequencyRepository.findByFrequencyId(input.getFrequencyId())
                        .orElseThrow(() -> new RuntimeException("通知頻度が見つかりません"));

        validateNotificationRule(
                frequency.getFrequencyName(),
                input.getNotificationWeek(),
                input.getNotificationDay()
        );

        notification.setFrequency(frequency);
        notification.setNotificationWeek(input.getNotificationWeek());
        notification.setNotificationDay(input.getNotificationDay());
        notification.setNotificationTime(input.getNotificationTime());

        notificationRepository.save(notification);
    }

    @Transactional
    public void deleteNotification(Long notificationId) {
        if (!notificationRepository.existsById(notificationId)) {
            throw new RuntimeException("通知設定が見つかりません");
        }

        notificationRepository.deleteById(notificationId);
    }

    @Transactional
    public void changeActive(
            Long notificationId,
            boolean active
    ) {
        Notification notification =
                notificationRepository.findByNotificationId(notificationId)
                        .orElseThrow(() -> new RuntimeException("通知設定が見つかりません"));

        notification.setIsActive(active);

        notificationRepository.save(notification);
    }

    private void validateNotificationRule(
            String frequencyName,
            Integer notificationWeek,
            Integer notificationDay
    ) {
        if ("DAILY".equals(frequencyName)) {
            if (notificationWeek != null || notificationDay != null) {
                throw new RuntimeException("毎日通知では曜日・日付を指定できません");
            }
            return;
        }

        if ("WEEKLY".equals(frequencyName)) {
            if (notificationWeek == null) {
                throw new RuntimeException("毎週通知では曜日指定が必要です");
            }

            if (notificationDay != null) {
                throw new RuntimeException("毎週通知では日付指定はできません");
            }

            if (notificationWeek < 1 || notificationWeek > 7) {
                throw new RuntimeException("曜日は1〜7で指定してください");
            }

            return;
        }

        if ("MONTHLY".equals(frequencyName)) {
            if (notificationDay == null) {
                throw new RuntimeException("毎月通知では日付指定が必要です");
            }

            if (notificationWeek != null) {
                throw new RuntimeException("毎月通知では曜日指定はできません");
            }

            if (notificationDay < 1 || notificationDay > 31) {
                throw new RuntimeException("日付は1〜31で指定してください");
            }

            return;
        }

        throw new RuntimeException("不正な通知頻度です");
    }
}
