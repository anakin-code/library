package simplex.pj148.skippy.server.controller;

import lombok.RequiredArgsConstructor;
import org.springframework.data.domain.Page;
import org.springframework.data.domain.Pageable;
import org.springframework.security.core.annotation.AuthenticationPrincipal;
import org.springframework.security.oauth2.jwt.Jwt;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RestController;
import simplex.pj148.skippy.server.controller.dto.request.TipCommentCreateRequestDto;
import simplex.pj148.skippy.server.controller.dto.request.TipCreateRequestDto;
import simplex.pj148.skippy.server.controller.dto.request.TipUpdateRequestDto;
import simplex.pj148.skippy.server.controller.dto.response.TipDetailResponseDto;
import simplex.pj148.skippy.server.controller.dto.response.TipListResponseDto;
import simplex.pj148.skippy.server.controller.dto.search.TipSearchConditionDto;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.service.TipService;
import simplex.pj148.skippy.server.service.UserInfoService;

@RestController
@RequestMapping("/api/tips")
@RequiredArgsConstructor
public class TipController {

    private final TipService tipService;

    private final UserInfoService userInfoService;

    @GetMapping
    public Page<TipListResponseDto> searchTips(
            @AuthenticationPrincipal Jwt jwt,
            @RequestParam(required = false) String keyword,
            @RequestParam(required = false) Integer affiliatedId,
            @RequestParam(required = false) Integer positionId,
            @RequestParam(required = false) Integer categoryId,
            @RequestParam(required = false) String orderBy,
            Pageable pageable
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        TipSearchConditionDto condition = new TipSearchConditionDto(
                keyword,
                affiliatedId,
                positionId,
                categoryId,
                orderBy
        );

        return tipService.searchTips(loginUser.getUserId(), condition, pageable);
    }

    @GetMapping("/{tipId}")
    public TipDetailResponseDto findTipDetail(
            @PathVariable Long tipId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        return tipService.findTipDetail(tipId, loginUser.getUserId());
    }

    @PostMapping
    public void createTip(
            @RequestBody TipCreateRequestDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.createTip(input, loginUser.getUserId());
    }

    @PutMapping("/{tipId}")
    public void updateTip(
            @PathVariable Long tipId,
            @RequestBody TipUpdateRequestDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.updateTip(tipId, input, loginUser.getUserId());
    }

    @DeleteMapping("/{tipId}")
    public void deleteTip(
            @PathVariable Long tipId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.deleteTip(tipId, loginUser.getUserId());
    }

    @PostMapping("/{tipId}/interest")
    public void createTipInterest(
            @PathVariable Long tipId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.createTipInterest(tipId, loginUser.getUserId());
    }

    @DeleteMapping("/{tipId}/interest")
    public void deleteTipInterest(
            @PathVariable Long tipId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.deleteTipInterest(tipId, loginUser.getUserId());
    }

    @PostMapping("/{tipId}/comments")
    public void createComment(
            @PathVariable Long tipId,
            @RequestBody TipCommentCreateRequestDto input,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.createComment(tipId, input, loginUser.getUserId());
    }

    @DeleteMapping("/{tipId}/comments/{commentId}")
    public void deleteComment(
            @PathVariable Long tipId,
            @PathVariable Long commentId,
            @AuthenticationPrincipal Jwt jwt
    ) {
        User loginUser = userInfoService.getUserFromJwt(jwt);

        tipService.deleteComment(tipId, commentId, loginUser.getUserId());
    }

検索URLの扱い

全部指定するならこれ。

GET /api/tips?keyword=睡眠&affiliatedId=1&positionId=2&categoryId=3&orderBy=interestCount&page=0&size=20

でも、実際は条件なしでもOK。

GET /api/tips
GET /api/tips?keyword=睡眠
GET /api/tips?affiliatedId=1
GET /api/tips?categoryId=3
GET /api/tips?keyword=睡眠&categoryId=3

Controllerで全部 required = false にしているから、検索欄なし・所属なし・職位なし・カテゴリなしでも動く。
