import {
  Box,
  Button,
  MenuItem,
  TextField,
  Typography,
} from "@mui/material";
import { useState } from "react";
import { useApi } from "../lib/useApi";
import type { Category } from "../models/Category";
import type { NeedInput } from "../models/Need";

type Props = {
  categories: Category[];
  onClose: () => void;
  onCreated: () => void;
};

const initialValues: NeedInput = {
  categoryId: "",
  needTitle: "",
  needDetail: "",
};

export default function SubmitNeeds({
  categories,
  onClose,
  onCreated,
}: Props): JSX.Element {
  const api = useApi();

  const [formData, setFormData] = useState<NeedInput>(initialValues);
  const [isSubmitting, setIsSubmitting] = useState(false);

  const titleCount = formData.needTitle.length;
  const detailCount = formData.needDetail.length;

  const canSubmit =
    formData.categoryId !== "" &&
    formData.needTitle.trim() !== "" &&
    formData.needDetail.trim() !== "" &&
    titleCount <= 30 &&
    detailCount <= 1000 &&
    !isSubmitting;

  const resetForm = (): void => {
    setFormData(initialValues);
  };

  const handleCancel = (): void => {
    if (isSubmitting) return;
    resetForm();
    onClose();
  };

  const handleSubmit = async (
    event: React.FormEvent<HTMLFormElement>
  ): Promise<void> => {
    event.preventDefault();

    if (formData.categoryId === "") {
      alert("悩みカテゴリを選択してください。");
      return;
    }

    if (!formData.needTitle.trim()) {
      alert("タイトルを入力してください。");
      return;
    }

    if (!formData.needDetail.trim()) {
      alert("詳細を入力してください。");
      return;
    }

    try {
      setIsSubmitting(true);

      const response = await api.fetch("/api/needs", {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
        },
        body: JSON.stringify({
          categoryId: Number(formData.categoryId),
          needTitle: formData.needTitle.trim(),
          needDetail: formData.needDetail.trim(),
        }),
      });

      if (!response.ok) {
        throw new Error("投稿に失敗しました");
      }

      resetForm();
      onCreated();
      onClose();
    } catch (error) {
      console.error(error);
      alert("投稿に失敗しました。入力内容やサーバーを確認してください。");
    } finally {
      setIsSubmitting(false);
    }
  };

  return (
    <Box component="form" onSubmit={handleSubmit}>
      <Typography
        component="h2"
        sx={{
          fontSize: 24,
          fontWeight: 800,
          textAlign: "center",
          mb: 3,
          color: "black",
        }}
      >
        お悩み投稿
      </Typography>

      <Box
        sx={{
          display: "flex",
          flexDirection: "column",
          gap: 2.5,
        }}
      >
        <Box>
          <Typography sx={{ fontWeight: 700, mb: 0.5 }}>*</Typography>
          <TextField
            select
            fullWidth
            size="small"
            value={formData.categoryId}
            onChange={(event) =>
              setFormData((prev) => ({
                ...prev,
                categoryId:
                  event.target.value === "" ? "" : Number(event.target.value),
              }))
            }
            SelectProps={{
              displayEmpty: true,
            }}
            sx={{
              backgroundColor: "white",
              borderRadius: 2,
              "& .MuiOutlinedInput-root": {
                borderRadius: 2,
              },
            }}
          >
            <MenuItem value="" disabled>
              悩みカテゴリ
            </MenuItem>

            {categories.map((category) => (
              <MenuItem
                key={category.categoryId}
                value={category.categoryId}
              >
                {category.categoryName}
              </MenuItem>
            ))}
          </TextField>
        </Box>

        <Box>
          <Typography sx={{ fontWeight: 700, mb: 0.5 }}>*</Typography>
          <TextField
            fullWidth
            size="small"
            placeholder="タイトル"
            value={formData.needTitle}
            onChange={(event) =>
              setFormData((prev) => ({
                ...prev,
                needTitle: event.target.value,
              }))
            }
            inputProps={{
              maxLength: 30,
            }}
            sx={{
              backgroundColor: "white",
              borderRadius: 2,
              "& .MuiOutlinedInput-root": {
                borderRadius: 2,
              },
            }}
          />

          <Typography
            sx={{
              mt: 0.3,
              textAlign: "right",
              fontSize: 12,
              color: titleCount > 30 ? "error.main" : "text.secondary",
            }}
          >
            {titleCount}/30
          </Typography>
        </Box>

        <Box>
          <TextField
            fullWidth
            multiline
            minRows={4}
            placeholder="詳細"
            value={formData.needDetail}
            onChange={(event) =>
              setFormData((prev) => ({
                ...prev,
                needDetail: event.target.value,
              }))
            }
            inputProps={{
              maxLength: 1000,
            }}
            sx={{
              backgroundColor: "white",
              borderRadius: 2,
              "& .MuiOutlinedInput-root": {
                borderRadius: 2,
              },
            }}
          />

          <Typography
            sx={{
              mt: 0.3,
              textAlign: "right",
              fontSize: 12,
              color: detailCount > 1000 ? "error.main" : "text.secondary",
            }}
          >
            {detailCount}/1000
          </Typography>

          <Typography
            sx={{
              mt: 0.5,
              textAlign: "center",
              fontSize: 11,
              color: "text.secondary",
            }}
          >
            ※投稿後の編集はできません。
          </Typography>
        </Box>

        <Box
          sx={{
            display: "flex",
            justifyContent: "center",
            gap: 2,
            mt: 1,
          }}
        >
          <Button
            type="button"
            variant="contained"
            onClick={handleCancel}
            disabled={isSubmitting}
            sx={{
              backgroundColor: "#ead9b8",
              color: "black",
              borderRadius: 3,
              px: 4,
              fontWeight: 700,
              "&:hover": {
                backgroundColor: "#dfc99d",
              },
            }}
          >
            キャンセル
          </Button>

          <Button
            type="submit"
            variant="contained"
            disabled={!canSubmit}
            sx={{
              backgroundColor: "#0068a8",

              borderRadius: 3,

                backgroundColor: "#005486",

            }}
          >
            投稿する
          </Button>
        </Box>
      </Box>
    </Box>
  );
}

import {
  Alert,
  Box,
  Button,
  CircularProgress,
  Dialog,
  DialogContent,
  Fab,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import type { SelectChangeEvent } from "@mui/material";
import { useCallback, useEffect, useMemo, useState } from "react";
import NeedCard from "../components/NeedCard";
import SubmitNeeds from "../components/SubmitNeeds";
import { useApi } from "../lib/useApi";
import type { Affiliated } from "../models/Affiliated";
import type { Category } from "../models/Category";
import type { Need, NeedPageResponse } from "../models/Need";
import type { Rank } from "../models/Rank";

type SortBy = "new" | "empathy";

type SearchCondition = {
  keyword: string;
  affiliatedId: string;
  positionId: string;
  categoryId: string;
};

const emptySearchCondition: SearchCondition = {
  keyword: "",
  affiliatedId: "",
  positionId: "",
  categoryId: "",
};

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);
  const [postOpen, setPostOpen] = useState(false);

  const [sortBy, setSortBy] = useState<SortBy>("new");
  const [page, setPage] = useState(0);
  const [totalPage, setTotalPage] = useState(0);
  const [totalElements, setTotalElements] = useState(0);

  const [searchCondition, setSearchCondition] =
    useState<SearchCondition>(emptySearchCondition);

  const [appliedCondition, setAppliedCondition] =
    useState<SearchCondition>(emptySearchCondition);

  const [affiliateds, setAffiliateds] = useState<Affiliated[]>([]);
  const [ranks, setRanks] = useState<Rank[]>([]);
  const [categories, setCategories] = useState<Category[]>([]);

  const [isLoading, setIsLoading] = useState(true);
  const [isMasterLoading, setIsMasterLoading] = useState(true);
  const [isServerStopping, setIsServerStopping] = useState(false);

  const size = 18;

  const requestUrl = useMemo(() => {
    const params = new URLSearchParams();

    params.set("page", String(page));
    params.set("size", String(size));
    params.set("orderBy", sortBy);

    if (appliedCondition.keyword.trim() !== "") {
      params.set("keyword", appliedCondition.keyword.trim());
    }

    if (appliedCondition.affiliatedId !== "") {
      params.set("affiliatedId", appliedCondition.affiliatedId);
    }

    if (appliedCondition.positionId !== "") {
      params.set("positionId", appliedCondition.positionId);
    }

    if (appliedCondition.categoryId !== "") {
      params.set("categoryId", appliedCondition.categoryId);
    }

    return `/api/needs/search?${params.toString()}`;
  }, [page, sortBy, appliedCondition]);

  const fetchNeeds = useCallback(async (): Promise<void> => {
    try {
      setIsLoading(true);
      setIsServerStopping(false);

      const response = await api.fetch(requestUrl);

      if (!response.ok) {
        throw new Error("お悩み一覧の取得に失敗しました");
      }

      const data: NeedPageResponse = await response.json();

      setNeedList(data.content ?? []);
      setTotalPage(data.totalPages ?? 0);
      setTotalElements(data.totalElements ?? 0);
    } catch (error) {
      console.error(error);
      setNeedList([]);
      setTotalPage(0);
      setTotalElements(0);
      setIsServerStopping(true);
    } finally {
      setIsLoading(false);
    }
  }, [api, requestUrl]);

  const fetchMasters = useCallback(async (): Promise<void> => {
    try {
      setIsMasterLoading(true);

      const [affiliatedResponse, rankResponse, categoryResponse] =
        await Promise.all([
          api.fetch("/api/affiliateds"),
          api.fetch("/api/ranks"),
          api.fetch("/api/categories"),
        ]);

      if (
        !affiliatedResponse.ok ||
        !rankResponse.ok ||
        !categoryResponse.ok
      ) {
        throw new Error("マスタ取得に失敗しました");
      }

      const affiliatedData: Affiliated[] = await affiliatedResponse.json();
      const rankData: Rank[] = await rankResponse.json();
      const categoryData: Category[] = await categoryResponse.json();

      setAffiliateds(affiliatedData);
      setRanks(rankData);
      setCategories(categoryData);
    } catch (error) {
      console.error(error);
      setAffiliateds([]);
      setRanks([]);
      setCategories([]);
    } finally {
      setIsMasterLoading(false);
    }
  }, [api]);

  useEffect(() => {
    fetchMasters();
  }, [fetchMasters]);

  useEffect(() => {
    fetchNeeds();
  }, [fetchNeeds]);

  const handleSortChange = (event: SelectChangeEvent): void => {
    setSortBy(event.target.value as SortBy);
    setPage(0);
  };

  const handleKeywordChange = (
    event: React.ChangeEvent<HTMLInputElement>
  ): void => {
    setSearchCondition((prev) => ({
      ...prev,
      keyword: event.target.value,
    }));
  };

  const handleSelectSearchChange =
    (field: keyof SearchCondition) =>
    (event: SelectChangeEvent): void => {
      setSearchCondition((prev) => ({
        ...prev,
        [field]: event.target.value,
      }));
    };

  const handleSearchSubmit = (): void => {
    setAppliedCondition(searchCondition);
    setPage(0);
  };

  const handleReset = (): void => {
    setSearchCondition(emptySearchCondition);
    setAppliedCondition(emptySearchCondition);
    setSortBy("new");
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  const handleCreated = (): void => {
    setPage(0);
    fetchNeeds();
  };

  return (
    <>
      <Box
        sx={{
          width: "100%",
          minHeight: "100vh",
          px: { xs: 2, md: 5 },
          py: 3,
          backgroundColor: "#fff7e8",
        }}
      >
        <Box
          sx={{
            display: "grid",
            gridTemplateColumns: { xs: "1fr", md: "1fr 230px" },
            gap: 3,
            alignItems: "start",
          }}
        >
          <Box>
            <Box
              sx={{
                display: "flex",
                justifyContent: "center",
                alignItems: "center",
                position: "relative",
                mb: 3,
              }}
            >
              <Typography
                component="h1"
                sx={{
                  fontSize: 30,
                  fontWeight: 800,
                  color: "black",
                  textAlign: "center",
                }}
              >
                お悩み一覧
              </Typography>

              <FormControl
                size="small"
                sx={{
                  position: { xs: "static", md: "absolute" },
                  right: 0,
                  minWidth: 110,
                  ml: { xs: 2, md: 0 },
                }}
              >
                <Select
                  value={sortBy}
                  onChange={handleSortChange}
                  sx={{
                    backgroundColor: "white",
                    borderRadius: 1,
                    fontSize: 13,
                    fontWeight: 700,
                  }}
                >
                  <MenuItem value="new">新着順</MenuItem>
                  <MenuItem value="empathy">共感数順</MenuItem>
                </Select>
              </FormControl>
            </Box>

            {isServerStopping && (
              <Alert
                severity="error"
                sx={{
                  mb: 3,
                  justifyContent: "center",
                  maxWidth: 600,
                  mx: "auto",
                }}
              >
                サーバーが起動していません。しばらくたってからお試しください。
              </Alert>
            )}

            {isLoading ? (
              <CircularProgress
                aria-label="Loading"
                sx={{
                  display: "block",
                  mx: "auto",
                  my: 8,
                }}
              />
            ) : (
              <>
                {needList.length === 0 ? (
                  <Box
                    sx={{
                      textAlign: "center",
                      py: 8,
                      color: "text.secondary",
                    }}
                  >
                    <Typography variant="h6" sx={{ fontWeight: 700 }}>
                      お悩みがありません
                    </Typography>
                  </Box>
                ) : (
                  <Grid container spacing={2}>
                    {needList.map((need, index) => (
                      <Grid
                        item

                        sm={6}
                        md={4}
                        key={need.needId ?? index}
                      >
                        <NeedCard need={need} />

                    ))}
                  </Grid>
                )}

                {totalPage > 1 && (

                      mt: 4,

          "&:hover": {
            backgroundColor: "#004a7c",
          },
        }}
        disabled={isServerStopping}
      >
        お悩み
        <br />
        投稿
      </Fab>

      <Dialog
        open={postOpen}
        onClose={() => setPostOpen(false)}
        fullWidth
        maxWidth="sm"
        slotProps={{
          paper: {
            sx: {

        <DialogContent

          }}
        >
          <SubmitNeeds
            categories={categories}
            onClose={() => setPostOpen(false)}
            onCreated={handleCreated}
          />
        </DialogContent>
      </Dialog>
    </>
  );
