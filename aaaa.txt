
import { useEffect, useState } from "react";
import type { Element } from "react";
import { useNavigate, useParams } from "react-router-dom";
import type { NavigateFunction } from "react-router-dom";
import type { AxiosResponse } from "axios";

import api from "../../api/client";
import type { Project } from "../../models/Affiliation";
import type { Profile } from "../../models/profile";
import ProjectDetail from "../../components/project/ProjectDetail";
import ErrorMessage from "../../components/common/ErrorMessage";

export default function ProjectDetailPage(): Element {
    const [project, setProject] = useState<Project>();
    const [errorMessage, setErrorMessage] = useState<string>("");
    const { projectId } = useParams();
    const [profile, setProfile] = useState<Profile[]>([]);

    const navigate: NavigateFunction = useNavigate();

    useEffect((): void => {
        api
            .get<Project>(`/project/${projectId}`)
            .then((res: AxiosResponse<Project>): void => setProject(res.data))
            .catch((err: any): void => {
                setErrorMessage("プロジェクトの取得に失敗しました。");
                console.error("プロジェクトの取得に失敗しました。", err);
            });
    }, [projectId]);

    useEffect((): void => {
        if (!projectId) return;

        api
            .get<Profile[]>(`/user/searchByProject/${projectId}`)
            .then((res: AxiosResponse<Profile[]>): void => {
                setProfile(res.data || []);
            })
            .catch((err: any): void => {
                setErrorMessage("プロフィールの取得に失敗しました。");
                console.error("プロフィールの取得に失敗しました。", err);
            });
    }, [projectId]);

    // ===== 追加ここから =====
    // endTime が null のキャリアを持つ人を「所属中」として先頭に並べる
    const sortedProfiles: Profile[] = [...profile].sort((a, b) => {
        const aIsCurrent =
            a.careers?.some((career) => career.endTime == null) ?? false;

        const bIsCurrent =
            b.careers?.some((career) => career.endTime == null) ?? false;

        if (aIsCurrent === bIsCurrent) {
            return 0;
        }

        return aIsCurrent ? -1 : 1;
    });
    // ===== 追加ここまで =====

    return (
        <div className="min-h-screen bg-zinc-50 px-6 py-5">
            <button
                onClick={(): void => navigate(`/project/update/${projectId}`)}
                className="px-12 py-2.5 mx-16 bg-blue-600 hover:bg-blue-700 text-white font-semibold rounded-xl shadow-lg shadow-blue-600/20"
            >
                編集
            </button>

            <div>
                <ErrorMessage message={errorMessage} />
            </div>

            {project && <ProjectDetail project={project} />}

            <div className="border border-zinc-200 bg-white">
                <div className="overflow-x-auto mt-[-50px]">
                    <h2>所属経歴メンバー一覧</h2>

                    {profile.length > 0 ? (
                        <table className="w-full text-sm">
                            <thead className="bg-zinc-50 text-left text-xs font-semibold uppercase tracking-wide text-zinc-500">
                                <tr>
                                    <th className="border border-gray-300 px-4 py-2">
                                        HRID
                                    </th>
                                    <th className="border border-gray-300 px-4 py-2">
                                        名前
                                    </th>
                                    <th className="border border-gray-300 px-4 py-2">
                                        メール
                                    </th>
                                    <th className="border border-gray-300 px-4 py-2">
                                        ランク
                                    </th>

                                    {/* ===== 追加ここから ===== */}
                                    <th className="border border-gray-300 px-4 py-2">
                                        所属状況
                                    </th>
                                    {/* ===== 追加ここまで ===== */}
                                </tr>
                            </thead>

                            <tbody>
                                {/* ===== 変更: profile.map → sortedProfiles.map ===== */}
                                {sortedProfiles.map((p: Profile): Element => {
                                    // ===== 追加ここから =====
                                    // endTime が null のキャリアがあれば現在も所属中
                                    const isCurrent =
                                        p.careers?.some(
                                            (career) => career.endTime == null
                                        ) ?? false;
                                    // ===== 追加ここまで =====

                                    return (
                                        <tr
                                            key={p.hrid}
                                            className="transition hover:bg-zinc-50 hover:cursor-pointer"
                                            onClick={(): void =>
                                                navigate(`/user/${p.hrid}`)
                                            }
                                        >
                                            <td className="border border-gray-300 px-4 py-2">
                                                {p.hrid}
                                            </td>

                                            <td className="border border-gray-300 px-4 py-2">
                                                {p.name}
                                            </td>

                                            <td className="border border-gray-300 px-4 py-2">
                                                {p.mailAddress}
                                            </td>

                                            <td className="border border-gray-300 px-4 py-2">
                                                {p.rankName}
                                            </td>

                                            {/* ===== 追加ここから ===== */}
                                            <td className="border border-gray-300 px-4 py-2">
                                                <span
                                                    className={
                                                        isCurrent
                                                            ? "rounded-full bg-green-100 px-2 py-1 text-xs font-semibold text-green-700"
                                                            : "rounded-full bg-gray-100 px-2 py-1 text-xs font-semibold text-gray-600"
                                                    }
                                                >
                                                    {isCurrent
                                                        ? "所属中"
                                                        : "離任済み"}
                                                </span>
                                            </td>
                                            {/* ===== 追加ここまで ===== */}
                                        </tr>
                                    );
                                })}
                            </tbody>
                        </table>
                    ) : (
                        <p>プロフィールがありません</p>
                    )}
                </div>
            </div>
        </div>
    );
}
