3. src/pages/NeedsList.tsx

これは、昔の NeedsList.tsx を今の仕様に合わせた完全版。

ポイントはここ。

/api/needs/search
orderBy=new | empathy
page=0始まり
size=18
import {
  Alert,
  Box,
  Button,
  CircularProgress,
  Fab,
  FormControl,
  Grid,
  MenuItem,
  Pagination,
  Select,
  Stack,
  Typography,
} from "@mui/material";
import type { SelectChangeEvent } from "@mui/material";
import { useEffect, useMemo, useState } from "react";
import NeedCard from "../components/NeedCard";
import SubmitForm from "./SubmitForm";
import type { Need, NeedPageResponse } from "../models/Need";
import { useApi } from "../lib/api";

type SortBy = "new" | "empathy";

export default function NeedsList(): JSX.Element {
  const api = useApi();

  const [needList, setNeedList] = useState<Need[]>([]);
  const [postOpen, setPostOpen] = useState(false);

  const [sortBy, setSortBy] = useState<SortBy>("new");
  const [page, setPage] = useState(0);
  const [totalPage, setTotalPage] = useState(0);
  const [totalElements, setTotalElements] = useState(0);

  const [isLoading, setIsLoading] = useState(true);
  const [isServerStopping, setIsServerStopping] = useState(false);

  const size = 18;

  const requestUrl = useMemo(() => {
    const params = new URLSearchParams();

    params.set("page", String(page));
    params.set("size", String(size));
    params.set("orderBy", sortBy);

    return `/api/needs/search?${params.toString()}`;
  }, [page, sortBy]);

  useEffect(() => {
    let ignore = false;

    const fetchNeeds = async (): Promise<void> => {
      try {
        setIsLoading(true);
        setIsServerStopping(false);

        const res: Response = await api.fetch(requestUrl);

        if (!res.ok) {
          throw new Error("Error fetching needs");
        }

        const data: NeedPageResponse = await res.json();

        if (ignore) return;

        setNeedList(data.content ?? []);
        setTotalPage(data.totalPages ?? 0);
        setTotalElements(data.totalElements ?? 0);
      } catch (err) {
        console.error(err);

        if (!ignore) {
          setNeedList([]);
          setTotalPage(0);
          setTotalElements(0);
          setIsServerStopping(true);
        }
      } finally {
        if (!ignore) {
          setIsLoading(false);
        }
      }
    };

    fetchNeeds();

    return () => {
      ignore = true;
    };
  }, [api, requestUrl]);

  const handleSortChange = (event: SelectChangeEvent): void => {
    setSortBy(event.target.value as SortBy);
    setPage(0);
  };

  const handlePageChange = (
    _event: React.ChangeEvent<unknown>,
    value: number
  ): void => {
    setPage(value - 1);
  };

  const handleCreated = (): void => {
    setPostOpen(false);
    setPage(0);

    // 投稿後に再取得したいので、sortByが同じでもuseEffectを動かすための簡易対応
    // 一番きれいにするなら fetchNeeds を useCallback で外に出す。
    window.location.reload();
  };

  return (
    <>
      <Box
        sx={{
          width: "100%",
          minHeight: "100vh",
          px: { xs: 2, sm: 4, md: 6 },
          py: 4,
          backgroundColor: "#f8fbff",
        }}
      >
        <Box
          sx={{
            display: "flex",
            justifyContent: "space-between",
            alignItems: "flex-end",
            mb: 4,
            gap: 2,
            flexWrap: "wrap",
          }}
        >
          <Box>
            <Typography
              variant="caption"
              sx={{
                color: "primary.main",
                fontWeight: 700,
                textTransform: "uppercase",
                letterSpacing: "0.15rem",
              }}
            >
              worries
            </Typography>

            <Typography
              component="h1"
              variant="h3"
              sx={{
                color: "#1a1a1a",
                fontWeight: 800,
                textAlign: "left",
                letterSpacing: "-0.03em",
                fontSize: { xs: "2rem", sm: "2.5rem" },
                mt: 0.5,
              }}
            >
              お悩み一覧
            </Typography>

            <Typography
              variant="body2"
              sx={{
                mt: 1,
                color: "text.secondary",
              }}
            >
              {totalElements}件のお悩み
            </Typography>
          </Box>

          <FormControl size="small" sx={{ minWidth: 140 }}>
            <Select
              value={sortBy}
              onChange={handleSortChange}
              displayEmpty
              sx={{
                borderRadius: "10px",
                fontWeight: 600,
                fontSize: "0.9rem",
                backgroundColor: "#fff",
                boxShadow: "0 1px 3px rgba(0,0,0,0.05)",
                "& .MuiOutlinedInput-notchedOutline": {
                  borderColor: "rgba(0, 0, 0, 0.08)",
                },
                "&:hover .MuiOutlinedInput-notchedOutline": {
                  borderColor: "rgba(0, 0, 0, 0.15)",
                },
                "&.Mui-focused .MuiOutlinedInput-notchedOutline": {
                  borderColor: "primary.main",
                },
              }}
            >
              <MenuItem value="new">新着順</MenuItem>
              <MenuItem value="empathy">共感数順</MenuItem>
            </Select>
          </FormControl>
        </Box>

        {isServerStopping && (
          <Alert
            severity="error"
            sx={{
              mb: 4,
              justifyContent: "center",
              maxWidth: 600,
              mx: "auto",
            }}
          >
            サーバーが起動していません。しばらくたってからお試しください。
          </Alert>
        )}

        {isLoading ? (
          <CircularProgress
            aria-label="Loading"
            sx={{
              display: "block",
              mx: "auto",
              my: 8,
            }}
          />
        ) : (
          <>
            {needList.length === 0 ? (
              <Box
                sx={{
                  textAlign: "center",
                  py: 8,
                  color: "text.secondary",
                }}
              >
                <Typography variant="h6" sx={{ fontWeight: 700 }}>
                  お悩みがありません
                </Typography>
                <Typography variant="body2" sx={{ mt: 1 }}>
                  条件を変えるか、最初のお悩みを投稿してみてください。
                </Typography>
              </Box>
            ) : (
              <Grid
                container
                spacing={2}
                sx={{
                  mb: 5,
                  mx: "auto",
                  justifyContent: "flex-start",
                }}
              >
                {needList.map((need, index) => (
                  <Grid
                    item
                    xs={12}
                    sm={6}
                    md={4}
                    key={need.needId ?? index}
                  >
                    <NeedCard need={need} />
                  </Grid>
                ))}
              </Grid>
            )}

            {totalPage > 1 && (
              <Stack
                direction="row"
                sx={{
                  pt: 2,
                  borderTop: "1px solid",
                  borderColor: "rgba(0, 0, 0, 0.08)",
                  justifyContent: "space-between",
                  alignItems: "center",
                  gap: 2,
                  flexWrap: "wrap",
                }}
              >
                <Button
                  variant="text"
                  disabled={page <= 0}
                  onClick={() => setPage(0)}
                  sx={{
                    borderRadius: 3,
                    textTransform: "none",
                    fontWeight: 600,
                    px: 2,
                    color: "text.primary",
                    "&:disabled": {
                      color: "text.disabled",
                    },
                  }}
                >
                  First
                </Button>

                <Button
                  variant="text"
                  disabled={page <= 0}
                  onClick={() => setPage(page - 1)}
                  sx={{
                    borderRadius: 3,
                    textTransform: "none",
                    fontWeight: 600,
                    px: 2,
                    color: "text.primary",
                    "&:disabled": {
                      color: "text.disabled",
                    },
                  }}
                >
                  前へ
                </Button>

                <Pagination
                  count={totalPage}
                  page={page + 1}
                  onChange={handlePageChange}
                  hidePrevButton
                  hideNextButton
                  color="primary"
                  variant="text"
                  shape="rounded"
                  sx={{
                    "& .MuiPaginationItem-root": {
                      fontWeight: 600,
                      borderRadius: 2,
                    },
                    "& .Mui-selected": {
                      backgroundColor: "primary.main",
                      color: "white",
                      "&:hover": {
                        backgroundColor: "primary.dark",
                      },
                    },
                  }}
                />

                <Button
                  variant="text"
                  disabled={totalPage <= 1 || page >= totalPage - 1}
                  onClick={() => setPage(page + 1)}
                  sx={{
                    borderRadius: 3,
                    textTransform: "none",
                    fontWeight: 600,
                    px: 2,
                    color: "text.primary",
                    "&:disabled": {
                      color: "text.disabled",
                    },
                  }}
                >
                  次へ
                </Button>

                <Button
                  variant="text"
                  disabled={totalPage <= 1 || page >= totalPage - 1}
                  onClick={() => setPage(totalPage - 1)}
                  sx={{
                    borderRadius: 3,
                    textTransform: "none",
                    fontWeight: 600,
                    px: 2,
                    color: "text.primary",
                    "&:disabled": {
                      color: "text.disabled",
                    },
                  }}
                >
                  Last
                </Button>
              </Stack>
            )}
          </>
        )}
      </Box>

      <Fab
        color="primary"
        onClick={() => setPostOpen(true)}
        sx={{
          position: "fixed",
          right: 32,
          bottom: 32,
          zIndex: 1000,
          fontSize: 28,
          fontWeight: 700,
        }}
        disabled={isServerStopping}
      >
        +
      </Fab>

      <SubmitForm
        open={postOpen}
        onClose={() => setPostOpen(false)}
        onCreated={handleCreated}
      />
    </>
  );
}
