3. TipService.java

ここが一番大事。
input.setCategoryId(1) みたいな固定値代入は消す。
Entity側では Tip が category_id で Category に紐づく形なので、送られてきた categoryId から Category を取って保存する。

package simplex.pj148.skippy.server.service;

import jakarta.persistence.EntityNotFoundException;
import lombok.RequiredArgsConstructor;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;
import simplex.pj148.skippy.server.controller.dto.request.TipsCreateDto;
import simplex.pj148.skippy.server.entity.Category;
import simplex.pj148.skippy.server.entity.Tip;
import simplex.pj148.skippy.server.entity.User;
import simplex.pj148.skippy.server.repository.CategoryRepository;
import simplex.pj148.skippy.server.repository.TipRepository;

import java.time.LocalDateTime;

@Service
@RequiredArgsConstructor
public class TipService {

    private final TipRepository tipRepository;
    private final CategoryRepository categoryRepository;

    @Transactional
    public void createTip(TipsCreateDto input, User loginUser) {

        Category category = categoryRepository.findByCategoryId(input.getCategoryId())
                .orElseThrow(() -> new EntityNotFoundException(
                        "カテゴリDBに存在しないカテゴリ：" + input.getCategoryId() + " が選択されています。"
                ));

        LocalDateTime createdTime = LocalDateTime.now();

        Tip tip = Tip.builder()
                .user(loginUser)
                .category(category)
                .tipTitle(input.getTipTitle())
                .tipDetail(input.getTipDetail())
                .tipURL(input.getTipURL())
                .createdTime(createdTime)
                .deleteFlg(false)
                .build();

        tipRepository.save(tip);
    }
}

import { useState } from "react";
import type { ChangeEvent, FormEvent } from "react";
import {
  Box,
  Button,
  MenuItem,
  Modal,
  Stack,
  TextField,
  Typography,
} from "@mui/material";
import type { Category } from "../models/category";

type TipFormInputValues = {
  categoryId: string | number;
  tipTitle: string;
  tipDetail: string;
  tipURL: string;
};

type TipFormSubmitValues = {
  categoryId: number;
  tipTitle: string;
  tipDetail: string;
  tipURL: string;
};

type TipFormProps = {
  initialValues: TipFormInputValues;
  categories: Category[];
  onSubmit: (values: TipFormSubmitValues) => void;
  submitLabel: string;
  onClose: () => void;
};

function TipForm({
  initialValues,
  categories,
  onSubmit,
  submitLabel,
  onClose,
}: TipFormProps) {
  const [formData, setFormData] = useState<TipFormInputValues>(initialValues);

  const handleChange = (
    event: ChangeEvent<HTMLInputElement | HTMLTextAreaElement>
  ): void => {
    const { name, value } = event.target;

    setFormData((prev) => ({
      ...prev,
      [name]: value,
    }));
  };

  const handleSubmit = (event: FormEvent<HTMLFormElement>): void => {
    event.preventDefault();

    onSubmit({
      categoryId: Number(formData.categoryId),
      tipTitle: formData.tipTitle,
      tipDetail: formData.tipDetail,
      tipURL: formData.tipURL,
    });
  };

  const selectableCategories = categories.filter(
    (category) => category.categoryId !== 0
  );

  return (
    <Modal open onClose={onClose}>
      <Box
        component="form"
        onSubmit={handleSubmit}
        sx={{
          position: "absolute",
          top: "50%",
          left: "50%",
          transform: "translate(-50%, -50%)",
          width: {
            xs: "90%",
            sm: 520,
          },
          bgcolor: "background.paper",
          borderRadius: 3,
          boxShadow: 24,
          p: 4,
        }}
      >
        <Typography
          component="h2"
          variant="h5"
          sx={{
            fontWeight: 700,
            mb: 3,
          }}
        >
          投稿フォーム
        </Typography>

        <Stack spacing={2}>
          <TextField
            required
            select
            fullWidth
            label="カテゴリ"
            name="categoryId"
            value={formData.categoryId}
            onChange={handleChange}
          >
            <MenuItem value="" disabled>
              カテゴリを選択
            </MenuItem>

            {selectableCategories.map((category) => (
              <MenuItem key={category.categoryId} value={category.categoryId}>
                {category.categoryName}
              </MenuItem>
            ))}
          </TextField>

          <TextField
            required
            fullWidth
            label="タイトル"
            name="tipTitle"
            value={formData.tipTitle}
            onChange={handleChange}
            slotProps={{
              htmlInput: {
                maxLength: 50,
              },
            }}
          />

          <TextField
            required
            fullWidth
            multiline
            minRows={4}
            label="投稿内容"
            name="tipDetail"
            value={formData.tipDetail}
            onChange={handleChange}
            slotProps={{
              htmlInput: {
                maxLength: 500,
              },
            }}
          />

          <TextField
            fullWidth
            label="URL"
            name="tipURL"
            value={formData.tipURL}
            onChange={handleChange}
          />

          <Stack
            direction="row"
            spacing={2}
            sx={{
              justifyContent: "flex-end",
              pt: 2,
            }}
          >
            <Button variant="outlined" onClick={onClose}>
              キャンセル
            </Button>

            <Button type="submit" variant="contained">
              {submitLabel}
            </Button>
          </Stack>
        </Stack>
      </Box>
    </Modal>
  );
}

export default TipForm;

    categoryId: 0,
    categoryName: "すべて",
  });
  const [sortBy, setSortBy] = useState<string>("createdTime,desc");
  const [isLoading, setIsLoading] = useState<boolean>(true);
  const [isServerStopping, setIsServerStopping] = useState<boolean>(false);
  const [isPostFormOpen, setIsPostFormOpen] = useState<boolean>(false);

  useEffect((): void => {
    api
      .fetch("/api/categories")
      .then((res: Response) => res.json())
      .then((categoryList: Category[]): void => {

          {
            categoryId: 0,
            categoryName: "すべて",
          },
          ...categoryList,
        ]);
      })
      .catch((error: unknown): void => {
        console.error(error);
      });

  useEffect((): void => {
    fetchTips();
  }, [page, selectedCategory, sortBy]);

    setIsLoading(true);
    setIsServerStopping(false);

      selectedCategory.categoryId === 0
        ? ""
        : `&categoryId=${selectedCategory.categoryId}`;

    api
      .fetch(`/api/tips?page=${page}&sort=${sortBy}${categoryQuery}`)
      .then((res: Response) => {
        if (!res.ok) {
          throw new Error("投稿一覧の取得に失敗しました");

        setTotalPage(data.totalPages ?? 0);
      })

        console.error(error);
        setContents([]);
        setTotalPage(0);
        setIsServerStopping(true);

      .finally((): void => {
        setIsLoading(false);
      });

    tipTitle: string;
    tipDetail: string;
    tipURL: string;

        headers: {
          "Content-type": "application/json",
        },
      })

          throw new Error("投稿に失敗しました");

    setSortBy(event.target.value);
    setPage(0);
  };

  const handlePageChange = (

    value: number

  return (

            color: "primary.main",
            fontWeight: 700,

            textTransform: "uppercase",

            fontSize: {
              xs: "2rem",

          }}

        <Box>
          <Box

              overflowX: "auto",
              whiteSpace: "nowrap",

              },
              msOverflowStyle: "none",
              scrollbarWidth: "none",
            }}
          >
            {categories.map((category) => {
              const isSelected =
                selectedCategory.categoryName === category.categoryName;

              return (
                <Chip
                  key={category.categoryId}
                  label={category.categoryName}
                  clickable
                  onClick={() =>
                    handleCategoryChange(
                      category.categoryId,
                      category.categoryName
                    )
                  }
                  variant={isSelected ? "filled" : "outlined"}
                  color={isSelected ? "primary" : "default"}
                  sx={{
                    fontWeight: isSelected ? 600 : 400,
                    fontSize: "0.85rem",
                    borderRadius: "10px",
                    px: 0.5,
                  }}
                />
              );
            })}
          </Box>

          <Box
            sx={{
              width: "100%",
              display: "flex",
              justifyContent: "flex-end",
            }}
          >
            <FormControl size="small" sx={{ minWidth: 130 }}>
              <Select
                value={sortBy}
                onChange={handleSortChange}
                displayEmpty
                startAdornment={
                  <SortIcon
                    fontSize="small"
                    sx={{
                      mr: 0.5,
